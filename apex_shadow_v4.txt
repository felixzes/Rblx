local MODES = {"NEAREST", "LOW HP", "LOCK"}
local VOID_MODES = {"FLOOR", "TELEPORT", "BOTH"}

local S = ENV.ApexShadowSettings
if type(S) ~= "table" then
    S = {dist = 3, mode = 1, wallSafe = true, teamSafe = true, hitSize = 20, hitAll = true,
         voidOn = true, voidY = -100, voidMode = 1, voidPredict = true, flingGuard = false,
         voidSize = 180, voidLife = 8, ux = 8, uy = 64}
    ENV.ApexShadowSettings = S

S.catches, S.lastCatch = 0, "" -- a fresh count every run, same as the rest of Apex's stats

local function GetRoot(p)
    local c = p and p.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end
local function GetHum(p)
    local c = p and p.Character
    return c and c:FindFirstChildOfClass("Humanoid")
end
local function Targetable(p)
    if p == LP then return false end
    local hum, root = GetHum(p), GetRoot(p)
    if not (hum and root) or hum.Health <= 0 then return false end
    -- team check only means something when the game really has 2+ teams
    if S.teamSafe and p.Team ~= nil and p.Team == LP.Team and #Teams:GetTeams() > 1 then return false end
    return true
end

-- ════════════════════════════════════════════════════════════════
-- SHADOW FOLLOW
-- ════════════════════════════════════════════════════════════════
local Follow = {on = false, target = nil, lastPos = nil, lastT = nil, vel = Vector3.zero, scanT = 0}
local RayParams = RaycastParams.new()
RayParams.FilterType = Enum.RaycastFilterType.Exclude

-- every targetable player, nearest first
local function SortedTargets()
    local my = GetRoot(LP)
    local list = {}
    if not my then return list end
    for _, p in ipairs(Players:GetPlayers()) do
        if Targetable(p) then
            list[#list + 1] = {p = p, d = (GetRoot(p).Position - my.Position).Magnitude}
        end
    end
    table.sort(list, function(a, b) return a.d < b.d end)
    local out = {}
    for i, e in ipairs(list) do out[i] = e.p end
    return out
end

local function PickTarget()
    local list = SortedTargets()
    if MODES[S.mode] == "LOW HP" then
        local best, bh
        for _, p in ipairs(list) do
            local h = GetHum(p).Health
            if not bh or h < bh then best, bh = p, h end
        end
        return best
    end
    return list[1]
end

local function ResetMotion()
    Follow.lastPos, Follow.lastT, Follow.vel = nil, nil, Vector3.zero
end
local function ToggleFollow()
    Follow.on = not Follow.on
    ResetMotion()
    Follow.target = Follow.on and PickTarget() or nil
end
local function NextTarget()
    local list = SortedTargets()
    if #list == 0 then return end
    local idx = 0
    for i, p in ipairs(list) do
        if p == Follow.target then idx = i end
    end
    Follow.target = list[idx % #list + 1]
    ResetMotion()
end

local function FollowStep(dt)
    if not Follow.on then return end
    local my = GetRoot(LP)
    if not my then return end
    local t = Follow.target
    local tr = t and Targetable(t) and GetRoot(t)
    if not tr then
        Follow.target = PickTarget()
        ResetMotion()
        return
    end
    local now = os.clock()

    -- re-pick now and then (NEAREST / LOW HP), with hysteresis so it doesn't flip-flop
    if MODES[S.mode] ~= "LOCK" and now - Follow.scanT > 1 then
        Follow.scanT = now
        local cand = PickTarget()
        if cand and cand ~= t then
            local cr = GetRoot(cand)
            local better
            if MODES[S.mode] == "LOW HP" then
                better = GetHum(cand).Health < GetHum(t).Health - 5
            else
                better = (cr.Position - my.Position).Magnitude + 6 < (tr.Position - my.Position).Magnitude
            end
            if better then Follow.target = cand; ResetMotion(); return end
        end
    end

    -- their velocity, estimated from position changes (once per frame, however many times we're called)
    local pos = tr.Position
    if Follow.lastT == nil then
        Follow.lastPos, Follow.lastT = pos, now
    elseif now - Follow.lastT > 0.006 then
        Follow.vel = Follow.vel:Lerp((pos - Follow.lastPos) / (now - Follow.lastT), 0.5)
        Follow.lastPos, Follow.lastT = pos, now
    end
    local flatV = Vector3.new(Follow.vel.X, 0, Follow.vel.Z)
    local look = Vector3.new(tr.CFrame.LookVector.X, 0, tr.CFrame.LookVector.Z)
    if look.Magnitude < 0.05 then look = Vector3.new(0, 0, -1) end
    -- behind their MOVEMENT direction when they're moving, behind where they face when they're not
    local dir = (flatV.Magnitude > 3) and flatV.Unit or look.Unit

    -- DISTANCE is always exactly S.dist studs from their LIVE position, never a predicted one,
    -- so it reads as the same distance every time. WALL SAFE only shortens it when a wall is
    -- actually behind them, so you're never teleported through geometry.
    local dist = S.dist
    if S.wallSafe then
        RayParams.FilterDescendantsInstances = {LP.Character, t.Character}
        local hit = Workspace:Raycast(pos, -dir * (dist + 1), RayParams)
        if hit then dist = math.max(1.5, hit.Distance - 1) end
    end
    local spot = pos - dir * dist
    local y = pos.Y
    local targetCF = CFrame.lookAt(Vector3.new(spot.X, y, spot.Z), Vector3.new(pos.X, y, pos.Z))

    -- SMOOTH: ease toward the target instead of snapping to it. The rate is high enough that it
    -- closes the gap within a couple of frames — fast enough that the distance still reads as a
    -- flat S.dist, gentle enough that the motion doesn't judder on every physics tick.
    local a = 1 - math.exp(-math.clamp(dt, 1 / 240, 0.1) * 30)
    my.CFrame = my.CFrame:Lerp(targetCF, a)
    -- move WITH them, so the server sees you travelling at their speed instead of stuttering
    my.AssemblyLinearVelocity = my.AssemblyLinearVelocity:Lerp(Vector3.new(Follow.vel.X, 0, Follow.vel.Z), a)
    my.AssemblyAngularVelocity = Vector3.zero
end
Track(RunService.Stepped:Connect(function(_, dt) FollowStep(dt) end))   -- before physics
Track(RunService.Heartbeat:Connect(function(dt) FollowStep(dt) end))   -- after physics

-- ════════════════════════════════════════════════════════════════
-- HITBOX EXTEND (local view only)
-- ════════════════════════════════════════════════════════════════
local Hit = {on = false, orig = setmetatable({}, {__mode = "k"})}

local function HitApplyPlayer(p)
    local c = p.Character
    if not c then return end
    local sz = Vector3.new(S.hitSize, S.hitSize, S.hitSize)
    for _, part in ipairs(c:GetDescendants()) do
        if part:IsA("BasePart") and (S.hitAll or part.Name == "HumanoidRootPart") then
            if not Hit.orig[part] then Hit.orig[part] = {size = part.Size, collide = part.CanCollide} end
            if part.Size ~= sz then
                -- non-colliding, so the big boxes can't shove you around
                pcall(function() part.Size = sz; part.CanCollide = false end)
            end
        end
    end
end
local function HitRestore()
    for part, info in pairs(Hit.orig) do
        if part.Parent then pcall(function() part.Size = info.size; part.CanCollide = info.collide end) end
        Hit.orig[part] = nil
    end
end
local function ToggleHitbox()
    Hit.on = not Hit.on
    if not Hit.on then HitRestore() end
end

local hitLoop = task.spawn(function()
    while true do
        task.wait(0.25)
        if Hit.on then
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LP then HitApplyPlayer(p) end
            end
        end
    end
end)
table.insert(ENV.ApexShadowLoops, {Disconnect = function() task.cancel(hitLoop) end})

-- ════════════════════════════════════════════════════════════════
-- ANTI-VOID
-- Three layers, each optional. SAFE POINT is remembered continuously (the last spot you
-- stood on real ground). CATCH decides what firing does: FLOOR drops a platform under you,
-- TELEPORT sends you back to the safe point (falling back to a floor if there isn't one yet),
-- BOTH does both. PREDICT fires the catch a little early, from how fast you're already
-- falling, instead of waiting until you've actually crossed the line. FLING GUARD is a
-- separate, opt-in safety net for a single hit that launches you hard enough to clear the map.
-- ════════════════════════════════════════════════════════════════
local Void = {part = nil, safeCF = nil, lastVelY = 0, pulse = nil}

local function VoidPulse(frac)
    if not Void.pulse then return end
    Void.pulse.BackgroundTransparency = 1 - math.clamp(frac, 0, 1) * 0.35
end

local function DoCatch(root, reason)
    local fy = S.voidY + 20
    local mode = VOID_MODES[S.voidMode]
    local useSafe = (mode ~= "FLOOR") and Void.safeCF ~= nil
    local needFloor = (mode == "FLOOR") or (mode == "BOTH") or (mode == "TELEPORT" and not useSafe)

    if needFloor then
        if Void.part and Void.part.Parent then Void.part:Destroy() end
        local p = Instance.new("Part")
        p.Name = "VoidFloor"
        p.Anchored = true
        p.CanCollide = true
        p.Size = Vector3.new(S.voidSize, 2, S.voidSize)
        p.Position = Vector3.new(root.Position.X, fy, root.Position.Z)
        p.BrickColor = BrickColor.new("Dark stone grey")
        p.Material = Enum.Material.SmoothPlastic
        p.Transparency = 0.3
        p.Parent = Workspace
        Void.part = p
        task.delay(S.voidLife, function()
            if Void.part == p then p:Destroy(); Void.part = nil end
        end)
    end

    if useSafe then
        root.CFrame = Void.safeCF + Vector3.new(0, 3, 0)
    else
        root.CFrame = CFrame.new(root.Position.X, fy + 3, root.Position.Z)
    end
    root.AssemblyLinearVelocity = Vector3.zero
    S.catches = S.catches + 1
    S.lastCatch = mode .. "  ·  " .. reason
end

local function VoidStep()
    local root = GetRoot(LP)
    local hum = GetHum(LP)
    if not (root and hum) then return end

    if not S.voidOn then
        VoidPulse(0)
        if Void.part and Void.part.Parent then Void.part:Destroy(); Void.part = nil end
        return
    end

    -- remember the last spot you stood on real, non-air ground
    if hum.FloorMaterial ~= Enum.Material.Air and hum.Health > 0 then
        Void.safeCF = root.CFrame
    end

    -- FLING GUARD (off by default): a velocity spike far bigger than the frame before gets
    -- capped, so one launch hit can't throw you clear across the map. A normal dash ramps up
    -- over several frames, so it doesn't trip this — only a sudden single-frame jump does.
    local vel = root.AssemblyLinearVelocity
    if S.flingGuard then
        local jump = vel.Magnitude - Void.lastVelY
        if jump > 220 and vel.Magnitude > 260 then
            root.AssemblyLinearVelocity = vel.Unit * 220
            vel = root.AssemblyLinearVelocity
        end
    end
    Void.lastVelY = vel.Magnitude

    -- PREDICT: catch half a second before you'd actually cross the trigger height
    local falling = vel.Y < -5
    local predictedY = root.Position.Y
    if S.voidPredict and falling then predictedY = root.Position.Y + vel.Y * 0.5 end

    local warnY = S.voidY + 40
    VoidPulse(1 - math.clamp((root.Position.Y - S.voidY) / (warnY - S.voidY), 0, 1))

    local fy = S.voidY + 20
    if predictedY < S.voidY and not (Void.part and Void.part.Parent) then
        DoCatch(root, (S.voidPredict and falling) and "predicted" or "trigger")
    end
    if Void.part and Void.part.Parent then
        Void.part.Position = Vector3.new(root.Position.X, fy, root.Position.Z)
    end
end
Track(RunService.Heartbeat:Connect(VoidStep))

table.insert(ENV.ApexShadowLoops, {Disconnect = function()
    HitRestore()
    if Void.part then pcall(function() Void.part:Destroy() end) end
end})

-- press and hold to keep stepping
local function HoldRepeat(btn, fn)
    btn.InputBegan:Connect(function(inp)
        local ut = inp.UserInputType
        if ut ~= Enum.UserInputType.MouseButton1 and ut ~= Enum.UserInputType.Touch then return end
        local alive = true
        inp.Changed:Connect(function()
            if inp.UserInputState == Enum.UserInputState.End then alive = false end
        end)
        fn()
        task.spawn(function()
            task.wait(0.4)
            while alive do fn(); task.wait(0.07) end
        end)
    end)
end
local function Stepper(labelFn, onDec, onInc)
    local r = Row(28)
    local dec = Btn(r, "-", UDim2.new(0, 34, 1, 0), UDim2.new(0, 0, 0, 0))
    local inc = Btn(r, "+", UDim2.new(0, 34, 1, 0), UDim2.new(1, -34, 0, 0))
    dec.TextSize = 15; inc.TextSize = 15
    local disp = Label(r, "", UDim2.new(1, -76, 1, 0), UDim2.new(0, 38, 0, 0), TXT, 11)
    HoldRepeat(dec, function() onDec(); Refresh() end)
    HoldRepeat(inc, function() onInc(); Refresh() end)
    Refreshers[#Refreshers + 1] = function() disp.Text = labelFn() end
end

local function ClampPos(pos, size)
    local ss = ScreenSize()
    return Vector2.new(
        math.floor(math.clamp(pos.X, 4, math.max(4, ss.X - size.X - 4)) + 0.5),
        math.floor(math.clamp(pos.Y, 4, math.max(4, ss.Y - size.Y - 4)) + 0.5))
end
local function ApplyPos()
    local size = Expanded and Panel.AbsoluteSize or Vector2.new(40, 40)
    if size.X < 10 then size = Vector2.new(PANEL_W, 380) end
    PanelPos = ClampPos(PanelPos, size)
    Panel.Position = UDim2.fromOffset(PanelPos.X, PanelPos.Y)
    Icon.Position = UDim2.fromOffset(PanelPos.X, PanelPos.Y)
    S.ux, S.uy = PanelPos.X, PanelPos.Y
end

end

local function MakeDraggable(handle, onTap)
    local dragging, moved, startInput, startMouse, startPos = false, false, nil, nil, nil
    handle.InputBegan:Connect(function(inp)
        local ut = inp.UserInputType
        if ut ~= Enum.UserInputType.MouseButton1 and ut ~= Enum.UserInputType.Touch then return end
        dragging, moved, startInput = true, false, inp
        startMouse = Vector2.new(inp.Position.X, inp.Position.Y)
        startPos = PanelPos
        inp.Changed:Connect(function()
            if inp.UserInputState == Enum.UserInputState.End and startInput == inp and dragging then
                dragging = false
                if not moved and onTap then onTap() end
            end
        end)
    end)
    Track(UserInputService.InputChanged:Connect(function(inp)
        if not dragging then return end
        if inp.UserInputType == Enum.UserInputType.MouseMovement or inp == startInput then
            local d = Vector2.new(inp.Position.X, inp.Position.Y) - startMouse
            if not moved and d.Magnitude > 6 then moved = true end
            if moved then PanelPos = startPos + d; ApplyPos() end
        end
    end))
end
MakeDraggable(Header, nil)

    end
end))
