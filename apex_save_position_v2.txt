local function CleanName(s, maxChars)
    local t = tostring(s or "")
    t = t:gsub("[<>&%c]", "")
    t = t:match("^%s*(.-)%s*$") or ""
    local len = utf8.len(t)
    if not len then return "" end
    maxChars = maxChars or 24
    if len > maxChars then t = t:sub(1, utf8.offset(t, maxChars + 1) - 1) end
    return t
end

local function FirstChars(s, n)
    local t = CleanName(s, 64)
    local len = utf8.len(t)
    if not len or len == 0 then return nil end
    if len > n then t = t:sub(1, utf8.offset(t, n + 1) - 1) end
    return t
end

local function ValidColor(c)
    return type(c) == "table" and type(c[1]) == "number" and type(c[2]) == "number" and type(c[3]) == "number"
end
local function ColorOf(c) return Color3.fromRGB(c[1], c[2], c[3]) end
local function HexOf(c) return string.format("#%02X%02X%02X", c[1], c[2], c[3]) end
local function ParseHex(s)
    local h = tostring(s or ""):match("^%s*#?(%x%x%x%x%x%x)%s*$")
    if not h then return nil end
    return {tonumber(h:sub(1, 2), 16), tonumber(h:sub(3, 4), 16), tonumber(h:sub(5, 6), 16)}
end
local function Fmt(v) return string.format("%.1f, %.1f, %.1f", v.X, v.Y, v.Z) end

-- DATA + PERSISTENCE
-- Slots carry a unique id, so sorting favorites to the top or
-- filtering by folder never breaks which slot is selected.
-- ════════════════════════════════════════════════════════════════
local function CanFiles()
    return type(writefile) == "function" and type(readfile) == "function" and type(isfile) == "function"
end

local function DefaultData()
    return {
        v = 2, slots = {}, folders = {}, nextId = 1, nextFolderId = 1,
        selectedId = nil, view = "all", holdIdx = 2, autoReturn = false,
    }
end

local function ValidSlot(s)
    if type(s) ~= "table" or type(s.c) ~= "table" or #s.c ~= 12 then return false end
    for i = 1, 12 do
        if type(s.c[i]) ~= "number" then return false end
    end
    return true
end

local function FindById(list, id)
    for i, v in ipairs(list) do
        if v.id == id then return v, i end
    end
    return nil, nil
end

-- Accepts v1 data (index-based selection, no ids/icons/colors/folders)
-- as well as v2, and always returns a clean v2 table.
local function Normalize(d)
    local out = DefaultData()

    local maxFolder = 0
    if type(d.folders) == "table" then
        for _, f in ipairs(d.folders) do
            local fid = type(f) == "table" and tonumber(f.id)
            if fid then
                fid = math.floor(fid)
                if fid > 0 and not FindById(out.folders, fid) then
                    local nm = CleanName(f.name, 16)
                    out.folders[#out.folders + 1] = {id = fid, name = (nm ~= "") and nm or ("Folder " .. fid)}
                    maxFolder = math.max(maxFolder, fid)
                end
            end
        end
    end
    out.nextFolderId = math.max(math.floor(tonumber(d.nextFolderId) or 1), maxFolder + 1)

    local seen, maxId = {}, 0
    if type(d.slots) == "table" then
        for _, s in ipairs(d.slots) do
            if ValidSlot(s) then
                local slot = {
                    name = CleanName(s.name, 24),
                    c = s.c,
                    fav = (s.fav == true),
                    icon = FirstChars(s.icon, 2) or ICON_PRESETS[1],
                    color = ValidColor(s.color) and {s.color[1], s.color[2], s.color[3]} or nil,
                    folder = tonumber(s.folder),
                }
                local id = math.floor(tonumber(s.id) or 0)
                if id > 0 and not seen[id] then
                    slot.id = id
                    seen[id] = true
                    maxId = math.max(maxId, id)
                end
                out.slots[#out.slots + 1] = slot
            end
        end
    end
    local nextId = math.max(math.floor(tonumber(d.nextId) or 1), maxId + 1)
    for i, slot in ipairs(out.slots) do
        if not slot.id then slot.id = nextId; nextId = nextId + 1 end
        if slot.name == "" then slot.name = "Pos " .. i end
        if not slot.color then
            local pc = PALETTE[(slot.id - 1) % #PALETTE + 1]
            slot.color = {pc[1], pc[2], pc[3]}
        end
        if slot.folder and not FindById(out.folders, slot.folder) then slot.folder = nil end
    end
    out.nextId = nextId

    local selId = tonumber(d.selectedId)
    if not selId and tonumber(d.selected) then
        local old = out.slots[math.floor(d.selected)]
        if old then selId = old.id end
    end
    if selId and FindById(out.slots, selId) then out.selectedId = selId end

    if d.view == "fav" or d.view == "all" then
        out.view = d.view
    elseif tonumber(d.view) and FindById(out.folders, tonumber(d.view)) then
        out.view = tonumber(d.view)
    end

    out.holdIdx = math.clamp(math.floor(tonumber(d.holdIdx) or 2), 1, #HOLD_OPTIONS)
    out.autoReturn = (d.autoReturn == true)
    out.ux, out.uy = tonumber(d.ux), tonumber(d.uy)
    return out
end

local function LoadFromFile()
    if not CanFiles() then return nil end
    local ok, raw = pcall(function()
        if isfile(FILE) then return readfile(FILE) end
        return nil
    end)
    if not ok or type(raw) ~= "string" then return nil end
    local ok2, decoded = pcall(function() return HttpService:JSONDecode(raw) end)
    if not ok2 or type(decoded) ~= "table" then return nil end
    return Normalize(decoded)
end

-- Session data survives re-running the script; the file survives rejoining.
local Data
do
    local raw = ENV.ApexSavePosData
    if type(raw) == "table" and type(raw.slots) == "table" then
        Data = Normalize(raw)
    else
        Data = LoadFromFile() or DefaultData()
    end
    ENV.ApexSavePosData = Data
end

local function Persist()
    if not CanFiles() then return end
    pcall(function()
        writefile(FILE, HttpService:JSONEncode({
            v = 2, slots = Data.slots, folders = Data.folders,
            nextId = Data.nextId, nextFolderId = Data.nextFolderId,
            selectedId = Data.selectedId, view = Data.view,
            holdIdx = Data.holdIdx, autoReturn = Data.autoReturn,
            ux = Data.ux, uy = Data.uy,
        }))
    end)
end

-- Lowest unused "Pos N": delete everything and the next save is "Pos 1".
local function NextDefaultName()
    local used = {}
    for _, s in ipairs(Data.slots) do
        local n = tonumber(string.match(s.name, "^Pos (%d+)$"))
        if n then used[n] = true end
    end
    local n = 1
    while used[n] do n = n + 1 end
    return "Pos " .. n
end

local function NextFolderName()
    local used = {}
    for _, f in ipairs(Data.folders) do
        local n = tonumber(string.match(f.name, "^Folder (%d+)$"))
        if n then used[n] = true end
    end
    local n = 1
    while used[n] do n = n + 1 end
    return "Folder " .. n
end

local function UniqueCopyName(base)
    local stem = base:gsub(" %(%d+%)$", "")
    local names = {}
    for _, s in ipairs(Data.slots) do names[s.name] = true end
    local n = 2
    local cand = stem .. " (" .. n .. ")"
    while names[cand] do
        n = n + 1
        cand = stem .. " (" .. n .. ")"
    end
    return cand
end

-- POSITION: clamped to the screen and kept off the center
-- ════════════════════════════════════════════════════════════════
local function ActiveSize()
    if Expanded then
        local s = Panel.AbsoluteSize
        if s.X < 10 or s.Y < 10 then return Vector2.new(PANEL_W, 260) end
        return s
    end
    return Vector2.new(MINI_W, MINI_H)
end
local function ClampToScreen(pos, size)
    local ss = ScreenSize()
    return Vector2.new(
        math.clamp(pos.X, 4, math.max(4, ss.X - size.X - 4)),
        math.clamp(pos.Y, 4, math.max(4, ss.Y - size.Y - 4)))
end
local function AvoidCenter(pos, size)
    local ss = ScreenSize()
    local cx, cy = ss.X / 2, ss.Y / 2
    local hw, hh = 48, 48
    if pos.X + size.X > cx - hw and pos.X < cx + hw and pos.Y + size.Y > cy - hh and pos.Y < cy + hh then
        local leftX, rightX = (cx - hw) - size.X, cx + hw
        if math.abs(pos.X - leftX) < math.abs(pos.X - rightX) then
            return Vector2.new(leftX, pos.Y)
        end
        return Vector2.new(rightX, pos.Y)
    end
    return pos
end
local function ApplyPos()
    local size = ActiveSize()
    local p = ClampToScreen(PanelPos, size)
    p = AvoidCenter(p, size)
    p = ClampToScreen(p, size)
    p = Vector2.new(math.floor(p.X + 0.5), math.floor(p.Y + 0.5))
    PanelPos = p
    Panel.Position = UDim2.fromOffset(p.X, p.Y)
    W.Mini.Position = UDim2.fromOffset(p.X, p.Y)
    Data.ux, Data.uy = p.X, p.Y
end
Track(Panel:GetPropertyChangedSignal("AbsoluteSize"):Connect(ApplyPos))

local function Collapse()
    Expanded = false
    Panel.Visible = false
    W.Mini.Visible = true
    ApplyPos()
    Bump()
end
local function Expand()
    Expanded = true
    W.Mini.Visible = false
    Panel.Visible = true
    ApplyPos()
    Bump()
end

-- Drag by the title bar (expanded) or the bar itself (collapsed).
local function MakeDraggable(handle)
    local dragging, moved = false, false
    local startInput, startMouse, startPos
    handle.InputBegan:Connect(function(inp)
        local ut = inp.UserInputType
        if ut ~= Enum.UserInputType.MouseButton1 and ut ~= Enum.UserInputType.Touch then return end
        dragging, moved = true, false
        DraggingNow = true
        startInput = inp
        startMouse = Vector2.new(inp.Position.X, inp.Position.Y)
        startPos = PanelPos
        Bump()
        inp.Changed:Connect(function()
            if inp.UserInputState == Enum.UserInputState.End and startInput == inp and dragging then
                dragging = false
                DraggingNow = false
                if moved then Persist() end
            end
        end)
    end)
    Track(UserInputService.InputChanged:Connect(function(inp)
        if not dragging then return end
        if inp.UserInputType == Enum.UserInputType.MouseMovement or inp == startInput then
            local delta = Vector2.new(inp.Position.X, inp.Position.Y) - startMouse
            if not moved and delta.Magnitude > 6 then moved = true end
            if moved then
                PanelPos = startPos + delta
                ApplyPos()
                Bump()
            end
        end
    end))
end
MakeDraggable(Header)
MakeDraggable(W.Mini)
Panel.InputBegan:Connect(function(inp)
    local ut = inp.UserInputType
    if ut == Enum.UserInputType.MouseButton1 or ut == Enum.UserInputType.Touch then Bump() end
end)

-- ════════════════════════════════════════════════════════════════
-- TELEPORT — instant, zero velocity, re-applied if the game pulls
-- you back
-- ════════════════════════════════════════════════════════════════
local function ApplyCF(cf)
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return false end
    local ok = pcall(function() char:PivotTo(cf) end)
    if not ok then root.CFrame = cf end
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
    return true
end

local function TeleportTo(cf, label)
    if not GetRoot() then
        SetStatus("No character yet", T.Red)
        return false
    end

    ENV.ApexSaveHold = ENV.ApexSaveHold + 1
    local token = ENV.ApexSaveHold

    -- A seated character stays welded to its seat, so stand up first.
    local hum = GetHum()
    if hum and hum.SeatPart then hum.Sit = false end

    ApplyCF(cf)
    SetStatus("Teleported to " .. label, T.Green)

    task.spawn(function()
        local holdSecs = HOLD_OPTIONS[Data.holdIdx] or 0
        local window = math.max(holdSecs, 0.35) -- always re-check briefly, even with HOLD off
        local t0 = os.clock()
        local fixes = 0
        local released = false
        while ENV.ApexSaveHold == token and os.clock() - t0 < window do
            RunService.Heartbeat:Wait()
            local r = GetRoot()
            if not r then break end -- respawned or removed; RETURN handles respawns
            if holdSecs > 0 and os.clock() - t0 > RELEASE_GRACE then
                local h = GetHum()
                if h and h.MoveDirection.Magnitude > 0.1 then released = true; break end
            end
            if (r.Position - cf.Position).Magnitude > PULLBACK_DIST then
                ApplyCF(cf)
                fixes = fixes + 1
            end
        end
        if ENV.ApexSaveHold == token then
            if fixes > 0 then
                SetStatus(("Held %s — countered %d pull-back%s"):format(label, fixes, fixes == 1 and "" or "s"), T.Yellow)
            elseif released then
                SetStatus("Hold released (you moved)", T.TxtSec)
            end
        end
    end)
    return true
end

local RefreshAll, RebuildList, RebuildFolders, OpenEditor, ShowPage, LoadEditor

local function SelectedSlot()
    return FindSlot(Data.selectedId) or Data.slots[#Data.slots]
end

local function GoSelected()
    local slot = SelectedSlot()
    if not slot then
        SetStatus("No saved positions yet", T.Yellow)
        Expand()
        return
    end
    TeleportTo(SlotCF(slot), slot.name)
end

-- Slots in the current view, favorites first (otherwise original order).
local function VisibleSlots()
    local favs, rest = {}, {}
    for _, s in ipairs(Data.slots) do
        local show
        if Data.view == "all" then show = true
        elseif Data.view == "fav" then show = s.fav
        else show = (s.folder == Data.view) end
        if show then
            if s.fav then favs[#favs + 1] = s else rest[#rest + 1] = s end
        end
    end
    for _, s in ipairs(rest) do favs[#favs + 1] = s end
    return favs
end

local function AddSlot(cf, name)
    local id = Data.nextId
    Data.nextId = id + 1
    local pc = PALETTE[(id - 1) % #PALETTE + 1]
    local slot = {
        id = id, name = name, c = {cf:GetComponents()},
        fav = (Data.view == "fav"),
        icon = ICON_PRESETS[1],
        color = {pc[1], pc[2], pc[3]},
        folder = (type(Data.view) == "number") and Data.view or nil,
    }
    Data.slots[#Data.slots + 1] = slot
    Data.selectedId = id
    return slot
end

local function SaveHere()
    local root = GetRoot()
    if not root then SetStatus("No character to save", T.Red); Expand(); return end
    local name = CleanName(W.Name.Text, 24)
    if name == "" then name = NextDefaultName() end
    local slot = AddSlot(root.CFrame, name)
    W.Name.Text = ""
    Persist()
    RefreshAll()
    SetStatus("Saved " .. slot.name .. "  (" .. Fmt(root.Position) .. ")", T.Green)
    task.defer(function()
        W.List.CanvasPosition = Vector2.new(0, math.max(0, W.List.AbsoluteCanvasSize.Y))
    end)
end

local function ReadTypedXYZ()
    local x, y, z = tonumber(Xy.X.Text), tonumber(Xy.Y.Text), tonumber(Xy.Z.Text)
    if not (x and y and z) then
        SetStatus("Enter numbers in X, Y and Z", T.Red)
        Expand()
        return nil
    end
    local root = GetRoot()
    -- keep the way you're currently facing
    local rot = root and root.CFrame.Rotation or CFrame.new()
    return CFrame.new(x, y, z) * rot, x, y, z
end

local function GoToTyped()
    local cf, x, y, z = ReadTypedXYZ()
    if not cf then return end
    if not GetRoot() then SetStatus("No character yet", T.Red); return end
    TeleportTo(cf, string.format("%.1f, %.1f, %.1f", x, y, z))
end

local function DuplicateSlot(slot)
    local _, idx = FindSlot(slot.id)
    local id = Data.nextId
    Data.nextId = id + 1
    local c = {}
    for i = 1, 12 do c[i] = slot.c[i] end
    local copy = {
        id = id, name = UniqueCopyName(slot.name), c = c, fav = slot.fav,
        icon = slot.icon, color = {slot.color[1], slot.color[2], slot.color[3]}, folder = slot.folder,
    }
    table.insert(Data.slots, (idx or #Data.slots) + 1, copy)
    Data.selectedId = id
    return copy
end

local function DeleteSlot(slot)
    local _, idx = FindSlot(slot.id)
    if not idx then return end
    table.remove(Data.slots, idx)
    if Data.selectedId == slot.id then
        local nxt = Data.slots[math.min(idx, #Data.slots)]
        Data.selectedId = nxt and nxt.id or nil
    end
end

local function CreateFolder(name)
    local id = Data.nextFolderId
    Data.nextFolderId = id + 1
    local nm = CleanName(name, 16)
    if nm == "" then nm = NextFolderName() end
    Data.folders[#Data.folders + 1] = {id = id, name = nm}
    Data.view = id
    return nm
end

local function CurrentFolder()
    if type(Data.view) == "number" then return (FindById(Data.folders, Data.view)) end
    return nil
end

local function CycleSelection(dir)
    local list = VisibleSlots()
    if #list == 0 then SetStatus("No saved positions in this view", T.Yellow); return end
    local idx = 0
    for i, s in ipairs(list) do
        if s.id == Data.selectedId then idx = i; break end
    end
    if idx == 0 then
        idx = (dir > 0) and 1 or #list
    else
        idx = ((idx - 1 + dir) % #list) + 1
    end
    Data.selectedId = list[idx].id
    Persist()
    RebuildList()
    SetStatus("Selected: " .. list[idx].name, T.Cyan)
end

RebuildFolders = function()
    for _, ch in ipairs(W.Folders:GetChildren()) do
        if ch:IsA("TextButton") then ch:Destroy() end
    end
    local order = 0
    local function AddChip(text, width, isSel, textCol, onClick)
        order = order + 1
        local b, st = MakeBtn({text = text, col = isSel and T.Cyan or textCol,
            bg = isSel and T.BtnSel or T.BtnBase, sz = 11,
            size = UDim2.new(0, width, 0, CHIP_H - 2), parent = W.Folders})
        b.LayoutOrder = order
        if isSel then st.Transparency = 0.05; st.Thickness = 1.75 end
        b.MouseButton1Click:Connect(function() Bump(); onClick() end)
    end

    AddChip("ALL", 40, Data.view == "all", T.TxtSec, function()
        Data.view = "all"; Persist(); RefreshAll()
    end)
    AddChip(STAR, 34, Data.view == "fav", T.Yellow, function()
        Data.view = "fav"; Persist(); RefreshAll()
    end)
    for _, f in ipairs(Data.folders) do
        local w = math.clamp((utf8.len(f.name) or 6) * 7 + 18, 44, 100)
        AddChip(f.name, w, Data.view == f.id, T.TxtSec, function()
            Data.view = f.id; Persist(); RefreshAll()
        end)
    end
    AddChip("+", 30, false, T.Green, function()
        local nm = CreateFolder("")
        Persist(); RefreshAll()
        SetStatus("Created " .. nm .. " — rename it on the MORE tab", T.Green)
    end)

    local cur = CurrentFolder()
    Mo.Box.PlaceholderText = cur and ("Rename: " .. cur.name) or "Folder name"
end

RefreshAll = function()
    RebuildFolders()
    RebuildList()
    W.Name.PlaceholderText = NextDefaultName()
    W.Count.Text = tostring(#Data.slots)
end

local EditingId = nil

local function EditingSlot()
    return EditingId and FindSlot(EditingId) or nil
end

local function FolderLabel(slot)
    if slot.folder then
        local f = FindById(Data.folders, slot.folder)
        if f then return f.name end
    end
    return "NO FOLDER"
end

local function RefreshEditorVisuals(slot)
    local col = ColorOf(slot.color)
    Ed.Fav.TextColor3 = slot.fav and T.Yellow or T.TxtDim
    Ed.IconBtn.Text = slot.icon
    Ed.IconBtn.TextColor3 = col
    Ed.Folder.Text = FolderLabel(slot)
    for _, sw in ipairs(Ed.Swatches) do
        local same = (sw.color[1] == slot.color[1] and sw.color[2] == slot.color[2] and sw.color[3] == slot.color[3])
        sw.stroke.Transparency = same and 0 or 0.8
        sw.stroke.Thickness = same and 2.25 or 1.25
    end
end

LoadEditor = function(slot)
    Ed.Name.Text = slot.name
    Ed.IconBox.Text = slot.icon
    Ed.Hex.Text = HexOf(slot.color)
    RefreshEditorVisuals(slot)
end

OpenEditor = function(id)
    local slot = FindSlot(id)
    if not slot then return end
    EditingId = id
    Data.selectedId = id
    LoadEditor(slot)
    ShowPage("edit")
end

local function CloseEditor()
    EditingId = nil
    Persist()
    RefreshAll()
    ShowPage("list")
end

-- Two-tap guard for destructive buttons.
local function MakeConfirm(btn, idleText, action)
    local armed = false
    btn.MouseButton1Click:Connect(function()
        Bump()
        if armed then
            armed = false
            btn.Text = idleText
            action()
        else
            armed = true
            btn.Text = "SURE?"
            task.delay(2, function()
                if armed then armed = false; btn.Text = idleText end
            end)
        end
    end)
end

Ed.Folder.MouseButton1Click:Connect(function()

    Bump()

    for _, f in ipairs(Data.folders) do ids[#ids + 1] = f.id end

    for i, v in ipairs(ids) do

    Persist()
    RefreshEditorVisuals(slot)
end)

-- Paste "x, y, z" into the X box and it spreads across all three.
local function ParseNumbers(str)
    local nums = {}
    for n in string.gmatch(str, "%-?%d+%.?%d*") do
        nums[#nums + 1] = tonumber(n)
    end
    return nums
end
Xy.X.FocusLost:Connect(function()
    local nums = ParseNumbers(Xy.X.Text)
    if #nums >= 3 then
        Xy.X.Text = tostring(nums[1])
        Xy.Y.Text = tostring(nums[2])
        Xy.Z.Text = tostring(nums[3])
    end
end)

-- Idle status line shows live coordinates; messages hold for a few seconds.
local posLoop = task.spawn(function()
    while true do
        task.wait(0.2)
        if os.clock() > StatusUntil then
            local r = GetRoot()
            W.Status.Text = r and ("pos " .. Fmt(r.Position) .. "  (tap to copy)") or "no character"
            W.Status.TextColor3 = T.TxtDim
        end
    end
end)
table.insert(ENV.ApexSavePosLoops, {Disconnect = function() task.cancel(posLoop) end})

-- Auto-minimize when idle (never while dragging or typing).
local idleLoop = task.spawn(function()
    while true do
        task.wait(0.5)
        if Expanded and not DraggingNow and not UserInputService:GetFocusedTextBox()
            and os.clock() - LastInteract > AUTO_COLLAPSE_AFTER then
            Collapse()
        end
    end
end)
table.insert(ENV.ApexSavePosLoops, {Disconnect = function() task.cancel(idleLoop) end})

Track(UserInputService.InputBegan:Connect(function(inp, gpe)
    if gpe then return end
    local k = inp.KeyCode
    if k == Enum.KeyCode.K then
        SaveHere()
    elseif k == Enum.KeyCode.L then
        GoSelected()
    elseif k == Enum.KeyCode.P then
        if Expanded then Collapse() else Expand() end
    elseif k == Enum.KeyCode.LeftBracket then
        CycleSelection(-1)
    elseif k == Enum.KeyCode.RightBracket then
        CycleSelection(1)
    end
end))

-- After a respawn, optionally teleport straight back to the selected slot.
Track(LP.CharacterAdded:Connect(function(char)
    ENV.ApexSaveHold = ENV.ApexSaveHold + 1 -- cancel any hold on the old character
    if not Data.autoReturn then return end
    local slot = SelectedSlot()
    if not slot then return end
    local root = char:WaitForChild("HumanoidRootPart", 10)
    if not root then return end
    task.wait(0.6) -- let the game place the spawn first
    TeleportTo(SlotCF(slot), slot.name)
end))
