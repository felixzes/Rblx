local SILVER = Color3.fromRGB(205, 225, 255)

-- a silver ghost of you, left standing where you were, fading out
local function Afterimage()
    local char = LP.Character
    if not char then return end
    local ok, clone = pcall(function()
        local prev = char.Archivable
        char.Archivable = true
        local c = char:Clone()
        char.Archivable = prev
        return c
    end)
    if not ok or not clone then return end
    clone.Name = "ApexUIAfterimage"
    for _, d in ipairs(clone:GetDescendants()) do
        if d:IsA("BaseScript") or d:IsA("Humanoid") or d:IsA("Sound") or d:IsA("ForceField")
            or d:IsA("Decal") or d:IsA("Texture") or d:IsA("SurfaceAppearance") or d:IsA("ParticleEmitter")
            or d:IsA("Trail") or d:IsA("Beam") or d:IsA("Highlight") or d:IsA("BillboardGui") then
            pcall(function() d:Destroy() end)
        end
    end
    local parts = {}
    for _, d in ipairs(clone:GetDescendants()) do
        if d:IsA("BasePart") then
            d.Anchored = true
            d.CanCollide = false; d.CanQuery = false; d.CanTouch = false; d.CastShadow = false
            if d.Name == "HumanoidRootPart" then
                d.Transparency = 1
            else
                d.Material = Enum.Material.Neon
                d.Color = SILVER
                d.Transparency = 0.45
                parts[#parts + 1] = d
            end
        end
    end
    clone.Parent = Workspace
    for _, p in ipairs(parts) do
        TweenService:Create(p, TweenInfo.new(0.5, Enum.EasingStyle.Quad), {Transparency = 1}):Play()
    end
    task.delay(0.6, function() pcall(function() clone:Destroy() end) end)
end

local function Burst(pos)
    local ring = Instance.new("Part")
    ring.Shape = Enum.PartType.Cylinder
    ring.Size = Vector3.new(0.2, 2, 2)
    ring.CFrame = CFrame.new(pos) * CFrame.Angles(0, 0, math.pi / 2)
    ring.Anchored = true; ring.CanCollide = false; ring.CanQuery = false; ring.CanTouch = false
    ring.CastShadow = false
    ring.Material = Enum.Material.Neon
    ring.Color = SILVER
    ring.Transparency = 0.3
    ring.Parent = Workspace
    TweenService:Create(ring, TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {Size = Vector3.new(0.2, 15, 15), Transparency = 1}):Play()
    task.delay(0.5, function() pcall(function() ring:Destroy() end) end)
end

local function NearestEnemy(from, maxD)
    local best, bd
    for model, e in pairs(Enemies) do
        if e.root and e.root.Parent and e.hum.Health > 0 and IsEnemy(model) then
            local d = (e.root.Position - from).Magnitude
            if d < maxD and (not bd or d < bd) then best, bd = e, d end
        end
    end
    return best
end

-- Something is about to hit you: blink behind them (and, only if STEP BACKUP is on, sidestep when
-- the blink can't happen). Nothing else ever moves you.
local function Threat(e, flat, look, kind)
    local now = os.clock()
    -- a flinch is their reaction to being hit, not an attack on you
    if kind == "anim" and now < e.dmgUntil then return end
    -- HANDS OFF mode: while you're attacking / dashing / holding a button, don't move you
    if PRESETS.attack[S.attackIdx][2] == 0 and AttackBlocked(now) then return end
    alertUntil = now + 0.25
    if S.blink and BlinkBehind(e) then return end
    if S.stepFallback then StepAside(flat, look, 1) end
end

local function OnEnemyAnimation(e, track)
    if not S.on or not IsAttackTrack(track, S.abilities) then return end
    local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    local er = e.root
    if not (root and er and IsEnemy(e.model)) then return end
    local toMe = root.Position - er.Position
    local flat = Vector3.new(toMe.X, 0, toMe.Z)
    local dist = flat.Magnitude
    if dist < 0.1 or dist > PRESETS.range[S.rangeIdx][2] + 4 then return end
    local now = os.clock()
    if now - e.cool < 0.05 then return end
    local facing = er.CFrame.LookVector:Dot(flat.Unit)
    -- REACT: AIMED ignores swings that clearly aren't aimed at you; ALL answers every swing
    local minDot = PRESETS.coverage[S.coverageIdx][2]
    if minDot > 0 and facing < minDot and dist > 6 then return end
    -- being knocked away from you (or backing off fast) isn't an attack on you
    if e.vel:Dot(flat.Unit) < -14 then return end
    -- movement-priority moves are the lower-confidence kind: only count them close up and facing you
    if not ATTACK_PRIO[track.Priority] and (dist > 10 or facing < 0.4) then return end
    e.cool = now
    Threat(e, flat, er.CFrame.LookVector, "anim")
end

local function Unregister(model)
    local e = Enemies[model]
    if not e then return end
    for _, c in ipairs(e.conns) do pcall(function() c:Disconnect() end) end
    Enemies[model] = nil
end

local function Register(model)
    if Enemies[model] or model == LP.Character then return end
    local hum = model:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local e = {model = model, hum = hum, root = nil, lastPos = nil, vel = Vector3.zero,
        cool = 0, limbCool = 0, dmgUntil = 0, limbs = {}, conns = {}}
    Enemies[model] = e
    local lastHp = hum.Health
    e.conns[#e.conns + 1] = hum.HealthChanged:Connect(function(h)
        if h < lastHp then e.dmgUntil = os.clock() + 0.5 end -- someone just hit them: what they play next is a flinch
        lastHp = h
    end)
    e.conns[#e.conns + 1] = model.AncestryChanged:Connect(function(_, parent)
        if not parent then Unregister(model) end
    end)
    task.spawn(function()
        e.root = model:WaitForChild("HumanoidRootPart", 6)
        for _, ch in ipairs(model:GetChildren()) do
            if LIMB_NAMES[ch.Name] and ch:IsA("BasePart") then e.limbs[#e.limbs + 1] = {part = ch, last = nil} end
        end
        local animator = hum:FindFirstChildOfClass("Animator") or hum:WaitForChild("Animator", 3)
        if animator and Enemies[model] == e then
            e.conns[#e.conns + 1] = animator.AnimationPlayed:Connect(function(track)
                OnEnemyAnimation(e, track)
            end)
        end
    end)
end

-- Hitbox / projectile candidates: new non-colliding or see-through parts
-- near you that aren't body parts or accessories.
local function MaybeHitbox(part)
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root or part:IsDescendantOf(char) then return end
    if (part.Position - root.Position).Magnitude > 28 then return end
    if part.Size.Magnitude < 3 or (part.Anchored and part.Size.Magnitude > 60) then return end
    if not (part.CanCollide == false or part.Transparency >= 0.5) then return end
    local parent = part.Parent
    if parent and parent:FindFirstChildOfClass("Humanoid") then return end
    if part:FindFirstAncestorOfClass("Accessory") then return end
    Hitboxes[part] = os.clock() + 2.5
end

-- Where would a crisp step of `dist` studs along `dir` land? nil when a wall is in the way
-- (it shortens the step first, and gives up under 1 stud) or there's no ground there.
local function SafeSpot(root, dir, dist)
    local origin = root.Position
    local hit = Workspace:Raycast(origin, dir * (dist + 1.3), RayParams)
    local d = hit and math.max(0, hit.Distance - 1.3) or dist
    if d < 1 then return nil end
    local spot = origin + dir * d
    if not Workspace:Raycast(spot, Vector3.new(0, -12, 0), RayParams) then return nil end
    return spot
end

-- A hitbox on you, or a fast part flying at you. It only counts as an attack when a hostile is close
-- enough to have thrown it AND you aren't the one acting — so your own effects never set it off.
local HitReact = setmetatable({}, {__mode = "k"})
local function HitboxThreats(root, now, blocked)
    if blocked then return end
    if not S.blink and not S.stepFallback then return end
    local myPos = root.Position
    local owner = NearestEnemy(myPos, 22)
    if not owner then return end
    for part, expiry in pairs(Hitboxes) do
        if not part.Parent or now > expiry then
            Hitboxes[part] = nil
        elseif now - (HitReact[part] or 0) > 0.3 then
            local rel = part.CFrame:PointToObjectSpace(myPos)
            local half = part.Size * 0.5
            local ox = half.X + 2.2 - math.abs(rel.X)
            local oy = half.Y + 2.2 - math.abs(rel.Y)
            local oz = half.Z + 2.2 - math.abs(rel.Z)
            local escape, escapeDist = nil, 0
            if ox > 0 and oy > 0 and oz > 0 then
                local localPush
                if ox <= oz then
                    localPush = Vector3.new((rel.X >= 0 and 1 or -1) * ox, 0, 0)
                else
                    localPush = Vector3.new(0, 0, (rel.Z >= 0 and 1 or -1) * oz)
                end
                local w = part.CFrame:VectorToWorldSpace(localPush)
                w = Vector3.new(w.X, 0, w.Z)
                if w.Magnitude > 0.05 then escape, escapeDist = w.Unit, math.min(w.Magnitude + 1, 18) end
            else
                local v = part.AssemblyLinearVelocity
                if v.Magnitude > 20 then
                    local toMe = myPos - part.Position
                    local t = math.clamp(toMe:Dot(v) / v:Dot(v), 0, 0.3)
                    local closest = part.Position + v * t
                    local flatV = Vector3.new(v.X, 0, v.Z)
                    if toMe:Dot(v) > 0 and flatV.Magnitude > 0.1
                        and (closest - myPos).Magnitude < math.max(half.X, half.Y, half.Z) + 2.5 then
                        local perp = flatV.Unit:Cross(Vector3.yAxis)
                        escape = (perp:Dot(toMe) >= 0) and perp or -perp
                        escapeDist = PRESETS.dodge[S.dodgeIdx][2] + 1
                    end
                end
            end
            if escape then
                HitReact[part] = now
                alertUntil = now + 0.25
                if not (S.blink and BlinkBehind(owner)) and S.stepFallback then
                    StepAside(nil, nil, 1, escape, escapeDist)
                end
            end
        end
    end
end

-- ════════════════════════════════════════════════════════════════
-- GUARD: passive by default — only watches for the game's own movement, never fights it
-- ════════════════════════════════════════════════════════════════
local Guard = {lastPos = nil, lastCF = nil, lastT = 0, framePos = nil, frameT = 0, clamping = false}

local function Revert(cf, keepVelocity)
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local vel = root.AssemblyLinearVelocity
    local ok = pcall(function() char:PivotTo(cf) end)
    if not ok then root.CFrame = cf end
    root.AssemblyLinearVelocity = keepVelocity and vel or Vector3.zero
    Guard.lastPos, Guard.lastCF, Guard.lastT = cf.Position, cf, os.clock()
end

local function NearestEnemyPos(from)
    local best, bd
    for model, e in pairs(Enemies) do
        if e.root and e.root.Parent and e.hum.Health > 0 and IsEnemy(model) then
            local d = (e.root.Position - from).Magnitude
            if d < 30 and (not bd or d < bd) then best, bd = e.root.Position, d end
        end
    end
    return best
end

local function GuardTick(isHeartbeat)
    if not S.on then return end
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not (root and hum) or hum.Health <= 0 then return end
    local now = os.clock()
    local pos = root.Position
    local vel = root.AssemblyLinearVelocity

    -- the game itself is moving you fast (dash, skill, knockback): stay out of its way
    if isHeartbeat then
        local hs = Vector3.new(vel.X, 0, vel.Z).Magnitude
        if hs > math.max(hum.WalkSpeed * 2.2, 48) then gameMoveUntil = now + 0.35 end
    end

    -- 1) ANTI-SNAP (off by default): true teleports only — a huge jump in one step
    if S.antiSnap and Guard.lastPos then
        local dt = math.clamp(now - Guard.lastT, 0, 0.25)
        if (pos - Guard.lastPos).Magnitude > 35 + vel.Magnitude * dt * 2.5 then
            Revert(Guard.lastCF, true)
            S.snaps = S.snaps + 1
            pos = Guard.lastPos
        end
    end

    -- 2) M1 LUNGE BLOCK (off by default): right after an M1 click, if the game pulls you toward
    -- an enemy while you are NOT steering and NOT holding Q, only walking speed gets through
    local clamped = false
    if isHeartbeat and S.lungeBlock and Guard.framePos and now < m1Until
        and hum.MoveDirection.Magnitude < 0.1 and not UserInputService:IsKeyDown(Enum.KeyCode.Q) then
        local fdt = math.clamp(now - Guard.frameT, 1 / 240, 0.1)
        local delta = pos - Guard.framePos
        local flat = Vector3.new(delta.X, 0, delta.Z)
        local allowed = (hum.WalkSpeed * 1.15 + 6 + totalVel.Magnitude) * fdt + 0.05
        local target = NearestEnemyPos(pos)
        if target and flat.Magnitude > allowed and flat.Magnitude < 9 then
            local toTarget = Vector3.new(target.X - pos.X, 0, target.Z - pos.Z)
            if toTarget.Magnitude > 0.1 and flat.Unit:Dot(toTarget.Unit) > 0.5 then
                local dir = flat.Unit
                local fixed = Vector3.new(Guard.framePos.X + dir.X * allowed, pos.Y, Guard.framePos.Z + dir.Z * allowed)
                Revert(CFrame.new(fixed) * root.CFrame.Rotation, true)
                pos = fixed
                clamped = true
                if not Guard.clamping then S.snaps = S.snaps + 1 end
            end
        end
    end
    Guard.clamping = clamped

    if isHeartbeat then Guard.framePos, Guard.frameT = pos, now end
    Guard.lastPos, Guard.lastCF, Guard.lastT = pos, root.CFrame, now
end

-- ════════════════════════════════════════════════════════════════
-- THE BLINK: appear behind the attacker, facing their back
-- ════════════════════════════════════════════════════════════════
BlinkBehind = function(e)
    local now = os.clock()
    if now - lastBlink < PRESETS.blinkCd[S.cdIdx][2] then return false end
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local er = e.root
    if not (root and hum and er and er.Parent and e.hum.Health > 0) or hum.Health <= 0 then return false end

    local look = er.CFrame.LookVector
    local flatLook = Vector3.new(look.X, 0, look.Z)
    if flatLook.Magnitude < 0.05 then return false end
    flatLook = flatLook.Unit

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {char, e.model}
    local overlap = OverlapParams.new()
    overlap.FilterType = Enum.RaycastFilterType.Exclude
    overlap.FilterDescendantsInstances = {char, e.model}

    local origin = er.Position
    local want = PRESETS.behindDist[S.behindIdx][2]
    local standY = (hum.RigType == Enum.HumanoidRigType.R6) and (root.Size.Y / 2 + 2)
        or (hum.HipHeight + root.Size.Y / 2)

    -- directly behind them first; if a wall, gap or solid is there, fan out to either side of "behind"
    local target
    for _, deg in ipairs({0, 35, -35, 70, -70}) do
        local dirBack = CFrame.Angles(0, math.rad(deg), 0) * (-flatLook)
        local wall = Workspace:Raycast(origin, dirBack * (want + 1.5), params)
        local dist = wall and (wall.Distance - 1.5) or want
        while dist >= 1.8 and not target do
            local spot = origin + dirBack * dist
            local floor = Workspace:Raycast(spot + Vector3.new(0, 6, 0), Vector3.new(0, -70, 0), params)
            if floor then
                local cand = Vector3.new(spot.X, floor.Position.Y + standY, spot.Z)
                local blocked = false
                for _, part in ipairs(Workspace:GetPartBoundsInRadius(cand, 1.6, overlap)) do
                    if part.CanCollide then blocked = true; break end
                end
                if not blocked then target = cand end
            end
            dist = dist - 1
        end
        if target then break end
    end
    if not target or (target - root.Position).Magnitude > 60 then return false end

    local cf = CFrame.lookAt(target, Vector3.new(origin.X, target.Y, origin.Z)) -- facing their back
    if S.afterimage and now - lastAfter > 0.25 then lastAfter = now; Afterimage() end
    local ok = pcall(function() char:PivotTo(cf) end)
    if not ok then root.CFrame = cf end
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
    impulseVel, totalVel = Vector3.zero, Vector3.zero
    lastBlink = now
    -- our own move: authorize it so ANTI-SNAP / NO LUNGE don't undo it
    Guard.lastPos, Guard.lastCF, Guard.lastT = cf.Position, cf, now
    Guard.framePos, Guard.frameT = cf.Position, now
    FaceLock.e, FaceLock.untilT = e, now + 0.3
    S.blinks = S.blinks + 1
    Burst(Vector3.new(target.X, target.Y - standY + 0.15, target.Z))
    FovPunch()
    return true
end

-- Crisp micro-teleport sidestep. No velocity is touched and nothing glides: next frame you're simply
-- somewhere safer, and you keep full control. It tries the best side, then the other, then straight
-- back, and skips any spot that's inside a wall or off a ledge.
StepAside = function(flat, look, scale, forcedDir, forcedDist)
    if scale <= 0 then return false end
    local now = os.clock()
    if now - lastStep < 0.08 then return false end
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return false end

    local want = forcedDist or (PRESETS.dodge[S.dodgeIdx][2] * scale)
    local options
    if forcedDir then
        local perp = forcedDir:Cross(Vector3.yAxis)
        options = {{forcedDir, lastSide}, {perp, 1}, {-perp, -1}}
    else
        local fl = Vector3.new(look.X, 0, look.Z)
        if fl.Magnitude < 0.05 then fl = flat end
        fl = fl.Unit
        local right = fl:Cross(Vector3.yAxis)
        local lateral = flat:Dot(right)
        local side
        if math.abs(lateral) > 0.6 then side = (lateral > 0) and 1 or -1 else side = -lastSide end
        options = {{right * side, side}, {right * -side, -side}, {flat.Unit, lastSide}}
    end
    for _, o in ipairs(options) do
        local dir = Vector3.new(o[1].X, 0, o[1].Z)
        if dir.Magnitude > 0.05 then
            local spot = SafeSpot(root, dir.Unit, want)
            if spot then
                local cf = CFrame.new(spot) * root.CFrame.Rotation
                if S.afterimage and now - lastAfter > 0.25 then lastAfter = now; Afterimage() end
                local ok = pcall(function() char:PivotTo(cf) end)
                if not ok then root.CFrame = cf end
                -- our own move: authorize it so the guards don't treat it as the game's
                Guard.lastPos, Guard.lastCF, Guard.lastT = cf.Position, cf, now
                Guard.framePos, Guard.frameT = cf.Position, now
                lastSide, lastStep = o[2], now
                S.dodges = S.dodges + 1
                return true
            end
        end
    end
    return false
end

-- ════════════════════════════════════════════════════════════════
-- MAIN STEP (before physics)
-- ════════════════════════════════════════════════════════════════
local function Step(dt)
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not S.on or not (root and hum) or hum.Health <= 0 then return end
    dt = math.clamp(dt, 1 / 240, 0.1)
    local now = os.clock()
    local blocked = AttackBlocked(now)
    local myPos = root.Position
    local range = PRESETS.range[S.rangeIdx][2]
    local senseSpeed = PRESETS.sense[S.senseIdx][2]

    for model, e in pairs(Enemies) do
        local er = e.root
        if er and er.Parent and e.hum.Health > 0 and IsEnemy(model) then
            local p = er.Position
            if e.lastPos then e.vel = e.vel:Lerp((p - e.lastPos) / dt, 0.6) end
            e.lastPos = p
            local toMe = myPos - p
            local flat = Vector3.new(toMe.X, 0, toMe.Z)
            local dist = flat.Magnitude
            if dist > 0.1 and dist < range then
                local u = flat / dist
                local look = er.CFrame.LookVector
                -- a dash / lunge at you: closing fast enough to reach you within half a second.
                -- No facing needed — dashes can be angled, which is why some used to get through.
                local closing = e.vel:Dot(u)
                if closing > 20 and dist / closing < 0.5 and now - e.cool > 0.15 then
                    e.cool = now
                    Threat(e, flat, look, "dash")
                end
                -- a hand or foot snapping toward you is a strike, whatever the move is called
                if dist < 12 then
                    for _, l in ipairs(e.limbs) do
                        local lp = l.part.Position
                        if l.last then
                            local rel = (lp - l.last) / dt - e.vel
                            local sp = rel.Magnitude
                            local toward = myPos - lp
                            if sp > senseSpeed and now - e.limbCool > 0.2
                                and toward.Magnitude < 8 and toward.Magnitude > 0.05
                                and rel:Dot(toward.Unit) > sp * 0.45 then
                                e.limbCool = now
                                Threat(e, flat, look, "strike")
                            end
                        end
                        l.last = lp
                    end
                else
                    for _, l in ipairs(e.limbs) do l.last = nil end
                end
            else
                for _, l in ipairs(e.limbs) do l.last = nil end
            end
        end
    end

    HitboxThreats(root, now, blocked)

    -- after a blink: keep you turned to their back for a moment (only while you aren't steering)
    if FaceLock.e and now < FaceLock.untilT and hum.MoveDirection.Magnitude < 0.1 then
        local er = FaceLock.e.root
        if er and er.Parent then
            local p = root.Position
            local at = Vector3.new(er.Position.X, p.Y, er.Position.Z)
            if (at - p).Magnitude > 0.5 then
                local cf = CFrame.lookAt(p, at)
                root.CFrame = cf
                Guard.lastPos, Guard.lastCF, Guard.lastT = cf.Position, cf, now
            end
        end
    end
end

-- LEAN + AURA (visual, local)
-- ════════════════════════════════════════════════════════════════
-- Lean is a REMOVABLE offset on top of whatever C0 the game has, never an absolute pose,
-- so it can't overwrite the game's own joint work. Off by default.
local Lean = {joints = {}, roll = 0, pitch = 0, applied = false}
local IDENT = CFrame.new()
function Lean.Restore()
    for _, j in pairs(Lean.joints) do
        pcall(function() j.m.C0 = j.m.C0 * j.off:Inverse() end)
        j.off = IDENT
    end
    Lean.joints = {}
    Lean.applied = false
end
function Lean.Bind(char)
    Lean.Restore()
    local waist = char:FindFirstChild("Waist", true)
    local neck = char:FindFirstChild("Neck", true)
    -- R15 only: its joints have plain axes, so a roll/pitch looks right
    if waist and waist:IsA("Motor6D") and neck and neck:IsA("Motor6D") then
        Lean.joints.waist = {m = waist, off = IDENT}
        Lean.joints.neck = {m = neck, off = IDENT}
    end
end
table.insert(ENV.ApexUI3Loops, {Disconnect = Lean.Restore})

local Aura = {h = nil}
function Aura.Remove()
    if Aura.h then pcall(function() Aura.h:Destroy() end); Aura.h = nil end
end
function Aura.Bind(char)
    Aura.Remove()
    local h = Instance.new("Highlight")
    h.FillColor = SILVER
    h.OutlineColor = Color3.fromRGB(255, 255, 255)
    h.FillTransparency = 0.85
    h.OutlineTransparency = 0.35
    h.DepthMode = Enum.HighlightDepthMode.Occluded
    h.Adornee = char
    h.Parent = char
    Aura.h = h
end
table.insert(ENV.ApexUI3Loops, {Disconnect = Aura.Remove})

local function VisualStep(dt)
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local active = S.on and root ~= nil
    local a = 1 - math.exp(-dt * 14)
    local tRoll, tPitch = 0, 0
    if active and S.lean then
        local lv = root.CFrame:VectorToObjectSpace(totalVel)
        tRoll = math.clamp(-lv.X * 0.012, -0.45, 0.45)
        tPitch = math.clamp(lv.Z * 0.008, -0.3, 0.3)
    end
    Lean.roll = Lean.roll + (tRoll - Lean.roll) * a
    Lean.pitch = Lean.pitch + (tPitch - Lean.pitch) * a
    local j = Lean.joints
    if j.waist and j.waist.m.Parent and j.neck and j.neck.m.Parent then
        local moving = S.lean or math.abs(Lean.roll) > 0.002 or math.abs(Lean.pitch) > 0.002
        if moving then
            local wOff = CFrame.Angles(Lean.pitch, 0, Lean.roll)
            local nOff = CFrame.Angles(0, 0, -Lean.roll * 0.7)
            j.waist.m.C0 = j.waist.m.C0 * j.waist.off:Inverse() * wOff
            j.neck.m.C0 = j.neck.m.C0 * j.neck.off:Inverse() * nOff
            j.waist.off, j.neck.off = wOff, nOff
            Lean.applied = true
        elseif Lean.applied then
            -- back at rest: take our offset off exactly once, then leave the joints alone
            j.waist.m.C0 = j.waist.m.C0 * j.waist.off:Inverse()
            j.neck.m.C0 = j.neck.m.C0 * j.neck.off:Inverse()
            j.waist.off, j.neck.off = IDENT, IDENT
            Lean.applied = false
        end
    end
    if Aura.h then
        Aura.h.Enabled = active and S.aura
        local flare = (os.clock() < alertUntil) and 0.25 or 0
        Aura.h.FillTransparency = 0.82 + 0.08 * math.sin(os.clock() * 3) - flare
    end
end

local pollT = 0
local function PollMyAttack(now)
    if now - pollT < 0.066 then return end
    pollT = now
    local hum = GetHum()
    local animator = hum and hum:FindFirstChildOfClass("Animator")
    if not animator then return end
    for _, tr in ipairs(animator:GetPlayingAnimationTracks()) do
        if IsAttackTrack(tr) then
            local len = (tr.Length > 0) and tr.Length or 0.5
            MarkAttack(math.min(math.max(0.15, len - tr.TimePosition), 0.7) + 0.1)
        end
    end
end

end
local heldPollT = 0
local function PollHeld(now)
    if now - heldPollT < 0.05 then return end
    heldPollT = now
    if UserInputService:GetFocusedTextBox() then heldAction = false; return end
    local held = false
    if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
        held = true
        m1Until = now + 0.25
    else
        for _, inp in ipairs(UserInputService:GetKeysPressed()) do
            if ACTION_KEYS[inp.KeyCode] then held = true; break end
        end
    end
    heldAction = held
end

local CharConns = {}
local function HookTool(tool)
    CharConns[#CharConns + 1] = tool.Activated:Connect(function() MarkAttack(0.5) end)
end

local function BindChar(char)
    for _, c in ipairs(CharConns) do pcall(function() c:Disconnect() end) end
    CharConns = {}
    RayParams.FilterDescendantsInstances = {char}
    impulseVel, totalVel = Vector3.zero, Vector3.zero
    FaceLock.e = nil
    local root = char:WaitForChild("HumanoidRootPart", 10)
    local hum = char:WaitForChild("Humanoid", 10)
    if not (root and hum) then return end
    Lean.Bind(char)
    Aura.Bind(char)
    Guard.lastPos, Guard.lastCF, Guard.lastT = root.Position, root.CFrame, os.clock()
    Guard.framePos, Guard.frameT = root.Position, os.clock()

    CharConns[#CharConns + 1] = root:GetPropertyChangedSignal("CFrame"):Connect(function() GuardTick(false) end)
    -- a body mover / constraint appearing on you means the GAME is moving you (a dash, a skill,
    -- knockback): stay out of the way for a moment — never destroy or fight it
    CharConns[#CharConns + 1] = char.DescendantAdded:Connect(function(d)
        if FORCE_CLASSES[d.ClassName] then gameMoveUntil = os.clock() + 0.6 end
    end)
    local animator = hum:FindFirstChildOfClass("Animator") or hum:WaitForChild("Animator", 3)
    if animator then
        CharConns[#CharConns + 1] = animator.AnimationPlayed:Connect(function(track)
            if IsAttackTrack(track) then
                local len = (track.Length > 0) and track.Length or 0.5
                MarkAttack(math.clamp(len, 0.3, 0.7) + 0.15)
            end
        end)
    end
    -- tools: a click on one counts as you attacking (this is how mobile swings show up)
    for _, ch in ipairs(char:GetChildren()) do
        if ch:IsA("Tool") then HookTool(ch) end
    end
    CharConns[#CharConns + 1] = char.ChildAdded:Connect(function(ch)
        if ch:IsA("Tool") then HookTool(ch) end
    end)

    -- BACKUP: if you do get hit (and weren't swinging), counter-blink behind the nearest close enemy
    local lastHealth = hum.Health
    CharConns[#CharConns + 1] = hum.HealthChanged:Connect(function(h)
        local dropped = h < lastHealth
        lastHealth = h
        if not (dropped and S.on and S.blink and S.hitBackup)
            or (PRESETS.attack[S.attackIdx][2] == 0 and AttackBlocked(os.clock())) then return end
        local r = char:FindFirstChild("HumanoidRootPart")
        if not r then return end
        local best, bd
        for model, e in pairs(Enemies) do
            if e.root and e.root.Parent and e.hum.Health > 0 and IsEnemy(model) then
                local d = (e.root.Position - r.Position).Magnitude
                if d < 14 and (not bd or d < bd) then best, bd = e, d end
            end
        end
        if best then alertUntil = os.clock() + 0.25; BlinkBehind(best) end
    end)
end

local function ClampPos(pos, size)
    local ss = ScreenSize()
    return Vector2.new(
        math.floor(math.clamp(pos.X, 4, math.max(4, ss.X - size.X - 4)) + 0.5),
        math.floor(math.clamp(pos.Y, 4, math.max(4, ss.Y - size.Y - 4)) + 0.5))
end
local function ApplyPos()
    local size = Expanded and Panel.AbsoluteSize or Vector2.new(40, 40)
    if size.X < 10 then size = Vector2.new(PANEL_W, 250) end
    PanelPos = ClampPos(PanelPos, size)
    Panel.Position = UDim2.fromOffset(PanelPos.X, PanelPos.Y)
    Icon.Position = UDim2.fromOffset(PanelPos.X, PanelPos.Y)
end
Track(Panel:GetPropertyChangedSignal("AbsoluteSize"):Connect(ApplyPos))
local function SetExpanded(v)
    Expanded = v
    Panel.Visible = v
    Icon.Visible = not v
    ApplyPos()
end
CollapseBtn.MouseButton1Click:Connect(function() SetExpanded(false) end)

local function MakeDraggable(handle, onTap)
    local dragging, moved, startInput, startMouse, startPos = false, false, nil, nil, nil
    handle.InputBegan:Connect(function(inp)
        local ut = inp.UserInputType
        if ut ~= Enum.UserInputType.MouseButton1 and ut ~= Enum.UserInputType.Touch then return end
        dragging, moved, startInput = true, false, inp
        startMouse = Vector2.new(inp.Position.X, inp.Position.Y)
        startPos = PanelPos
        inp.Changed:Connect(function()
            if inp.UserInputState == Enum.UserInputState.End and startInput == inp and dragging then
                dragging = false
                if not moved and onTap then onTap() end
            end
        end)
    end)
    Track(UserInputService.InputChanged:Connect(function(inp)
        if not dragging then return end
        if inp.UserInputType == Enum.UserInputType.MouseMovement or inp == startInput then
            local d = Vector2.new(inp.Position.X, inp.Position.Y) - startMouse
            if not moved and d.Magnitude > 6 then moved = true end
            if moved then PanelPos = startPos + d; ApplyPos() end
        end
    end))
end
MakeDraggable(Header, nil)
