local TAU = math.pi * 2
local V3  = Vector3.new

-- deterministic pseudo-random in [0, 1)
local function H(i, k)
    local v = math.sin(i * 12.9898 + k * 78.233) * 43758.5453
    return v - math.floor(v)
end

-- ════════════════════════════════════════════════════════════════
-- SHAPES — each returns a point in shape units (roughly -1..1) for
-- particle i of n. Drawn in the XY plane, a little depth in Z.
-- ════════════════════════════════════════════════════════════════
local Shapes = {}
Shapes.lanes = {-1, 1, -0.5, 0.5, 0}
Shapes.orbits = {
    CFrame.Angles(0, 0, 0),
    CFrame.Angles(math.pi / 3, 0, math.pi / 3),
    CFrame.Angles(-math.pi / 3, 0, -math.pi / 3),
}
Shapes.tilt = CFrame.Angles(math.rad(22), 0, math.rad(-18))

Shapes.SWORD = function(i, n)
    local f = (i - 1) / math.max(1, n - 1)
    if f < 0.07 then -- pommel ring
        local a = (f / 0.07) * TAU
        return V3(math.cos(a) * 0.08, -0.93 + math.sin(a) * 0.08, 0)
    elseif f < 0.15 then -- wrapped grip
        local t = (f - 0.07) / 0.08
        return V3(math.sin(t * 18) * 0.03, -0.84 + t * 0.24, math.cos(t * 18) * 0.03)
    elseif f < 0.33 then -- curved crossguard, two horns
        local u = (f - 0.15) / 0.18
        local side = (i % 2 == 0) and 1 or -1
        return V3(side * (0.05 + 0.46 * u), -0.57 + 0.18 * u * u - 0.05 * u, (H(i, 1) - 0.5) * 0.04)
    end
    -- blade: two edges, two mid-lines and a spine, tapering to a point
    local t = (f - 0.33) / 0.67
    local y = -0.52 + t * 1.5
    local w = 0.16 * (1 - t ^ 1.7)
    local lane = i % 5
    return V3(Shapes.lanes[lane + 1] * w, y, (lane == 4) and 0.03 or (H(i, 2) - 0.5) * 0.03)
end

Shapes.WINGS = function(i, n)
    local half = math.floor(n / 2)
    local side = (i <= half) and -1 or 1
    local idx = (i <= half) and (i - 1) or (i - half - 1)
    local F = 7 -- feathers per wing
    local j = idx % F
    local k = math.floor(idx / F)
    local K = math.max(2, math.ceil(half / F))
    local t = k / (K - 1)
    local ang = math.rad(78 - (j / (F - 1)) * 110)
    local len = 0.55 + 0.9 * math.sin(math.pi * (j + 0.6) / (F + 0.2))
    local dx, dy = math.cos(ang), math.sin(ang)
    local bow = 0.18 * t * t
    local lane = (idx * 7) % 3 - 1
    local x = 0.08 + dx * len * t - dy * bow - dy * lane * 0.03 * t
    local y = 0.15 + dy * len * t + dx * bow + dx * lane * 0.03 * t
    return V3(side * x, y, (H(i, 3) - 0.5) * 0.06 * t)
end

Shapes.GALAXY = function(i, n)
    local f = (i - 1) / math.max(1, n - 1)
    if f < 0.14 then -- bright core
        local r = 0.22 * math.sqrt(f / 0.14)
        local a = i * 2.39996
        return V3(math.cos(a) * r, math.sin(a) * r, (H(i, 4) - 0.5) * 0.12)
    end
    local arm = i % 3
    local u = (f - 0.14) / 0.86
    local r = 0.2 + 1.15 * u ^ 0.85
    local a = arm * (TAU / 3) + u * 5.2 + (H(i, 5) - 0.5) * 0.25
    return V3(math.cos(a) * r, math.sin(a) * r, (H(i, 6) - 0.5) * 0.1 * (1 - u))
end

Shapes.ATOM = function(i, n)
    local f = (i - 1) / math.max(1, n - 1)
    if f < 0.1 then -- nucleus
        local r = 0.14 * math.sqrt(f / 0.1)
        local a, b = i * 2.39996, i * 1.618
        return V3(math.cos(a) * math.sin(b) * r, math.sin(a) * math.sin(b) * r, math.cos(b) * r)
    end
    local o = i % 3
    local m = math.floor(i / 3)
    local t = (m / math.max(1, n * 0.3)) * TAU
    return Shapes.orbits[o + 1] * V3(1.05 * math.cos(t), 0.38 * math.sin(t), 0)
end

Shapes.HELIX = function(i, n)
    local half = math.floor(n / 2)
    local s = i % 2
    local t = math.floor((i - 1) / 2) / math.max(1, half - 1)
    local y = -1 + 2 * t
    local r = 0.38
    if i % 9 == 0 then -- rung between the two strands
        local a2 = t * 6 * math.pi
        local p1 = V3(math.cos(a2) * r, y, math.sin(a2) * r)
        local p2 = V3(math.cos(a2 + math.pi) * r, y, math.sin(a2 + math.pi) * r)
        return p1:Lerp(p2, H(i, 7))
    end
    local a = t * 6 * math.pi + s * math.pi
    return V3(math.cos(a) * r, y, math.sin(a) * r)
end

Shapes.HALO = function(i, n)
    local ring = i % 3
    local m = math.floor((i - 1) / 3)
    local a = (m / math.max(1, n / 3)) * TAU
    local r = 0.62 + ring * 0.28
    return V3(math.cos(a) * r, math.sin(a) * r, math.sin(a * 6 + ring) * 0.05)
end

Shapes.HEART = function(i, n)
    local layer = i % 3
    local m = math.floor((i - 1) / 3)
    local t = (m / math.max(1, n / 3)) * TAU
    local x = 16 * math.sin(t) ^ 3
    local y = 13 * math.cos(t) - 5 * math.cos(2 * t) - 2 * math.cos(3 * t) - math.cos(4 * t)
    local sc = ((layer == 0) and 1 or ((layer == 1) and 0.68 or 0.36)) / 17
    return V3(x * sc, (y + 2) * sc, (layer - 1) * 0.07)
end

Shapes.SATURN = function(i, n)
    local N = math.max(1, math.floor(n * 0.55))
    if i <= N then -- fibonacci sphere
        local y = 1 - 2 * ((i - 1) + 0.5) / N
        local r = math.sqrt(math.max(0, 1 - y * y))
        local a = (i - 1) * 2.39996
        return V3(math.cos(a) * r * 0.55, y * 0.55, math.sin(a) * r * 0.55)
    end
    local m = i - N
    local ring = m % 3
    local k = math.floor(m / 3)
    local a = (k / math.max(1, n * 0.15)) * TAU
    local r = 0.85 + ring * 0.14
    return Shapes.tilt * V3(math.cos(a) * r, 0, math.sin(a) * r)
end

local function PaletteColor(i, n, t, flash)
    local f = (i - 1) / math.max(1, n - 1)
    local name = PALETTES[S.palette]
    local c
    if name == "AURORA" then
        c = Color3.fromHSV((0.46 + 0.28 * f + 0.04 * math.sin(t)) % 1, 0.65, 1)
    elseif name == "EMBER" then
        c = Color3.fromHSV(0.01 + 0.11 * f, 0.9, 0.85 + 0.15 * math.sin(t * 5 + i))
    elseif name == "ICE" then
        c = Color3.fromHSV(0.52 + 0.06 * f, 0.15 + 0.5 * (1 - f), 1)
    elseif name == "VOID" then
        c = Color3.fromHSV(0.74 + 0.14 * f, 0.85, 0.7 + 0.3 * math.sin(t * 2 + f * 6) ^ 2)
    elseif name == "RAINBOW" then
        c = Color3.fromHSV((f + t * 0.12) % 1, 0.85, 1)
    else -- GOLD
        c = Color3.fromHSV(0.12 + 0.03 * f, 0.55 + 0.35 * (1 - f), 1)
    end
    if flash and flash > 0 then c = c:Lerp(Color3.new(1, 1, 1), flash) end
    return c
end
local function Accent() return PaletteColor(2, 3, 0) end

-- ════════════════════════════════════════════════════════════════
-- FORMATION STATE + MORPHING
-- ════════════════════════════════════════════════════════════════
local T0 = os.clock()
local slashT0, pulseT0, morphT0 = -10, -10, -10
local frame = 0
local src, dst, cur = {}, {}, {}
local moveList, cfs = {}, {}
local ActiveN = 0

local function Rebuild(morph)
    local n = COUNTS[S.countIdx]
    local fn = Shapes[SHAPE_NAMES[S.shape]]
    for i = 1, n do
        dst[i] = fn(i, n)
        if morph then
            src[i] = cur[i] or dst[i]
        else
            src[i], cur[i] = dst[i], dst[i]
        end
    end
    morphT0 = morph and os.clock() or -10
    ActiveN = n
    moveList = {Core}
    cfs = {CFrame.new()}
    for i = 1, MAX_PARTS do
        local p = Parts[i]
        if i <= n then
            moveList[#moveList + 1] = p
            cfs[#cfs + 1] = CFrame.new()
            p.Transparency = S.shown and 0 or 1
        else
            p.Transparency = 1
        end
    end
    for _, tr in ipairs(Trails) do
        tr.trail.Enabled = S.shown and S.trails and tr.idx <= n
    end
    Light.Enabled = S.shown
end

-- where the formation sits, by mode
local function FormationCF(t)
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local base = root.CFrame
    local size = SIZES[S.sizeIdx]
    local mode = MODES[S.mode]
    local spin = math.rad(SPINS[S.spinIdx][2]) * t
    local sway = CFrame.Angles(0, math.sin(t * 0.8) * 0.12, 0)
    if mode == "BACK" then
        return base * CFrame.new(0, 2.2 + size * 0.15, 2 + size * 0.2) * sway
    elseif mode == "AIM" then
        local cam = Workspace.CurrentCamera
        local look = cam and cam.CFrame.LookVector or base.LookVector
        local dir = Vector3.new(look.X, 0, look.Z)
        dir = (dir.Magnitude > 0.01) and dir.Unit or Vector3.new(0, 0, -1)
        local pos = base.Position + dir * (8 + size * 0.9) + Vector3.new(0, 3 + size * 0.2, 0)
        return CFrame.lookAt(pos, pos + dir) * sway
    elseif mode == "ORBIT" then
        local out = Vector3.new(math.cos(spin), 0, math.sin(spin))
        local pos = base.Position + out * (6 + size * 0.7) + Vector3.new(0, 3 + size * 0.2, 0)
        return CFrame.lookAt(pos, pos + out)
    elseif mode == "ABOVE" then
        return CFrame.new(base.Position + Vector3.new(0, 5 + size * 0.45, 0))
            * CFrame.Angles(0, spin, 0) * CFrame.Angles(-math.pi / 2, 0, 0)
    end
    -- AURA
    return CFrame.new(base.Position + Vector3.new(0, 0.5, 0)) * CFrame.Angles(0, spin, 0)
end

local function Step()
    if not S.shown or ActiveN == 0 then return end
    local now = os.clock()
    local t = now - T0
    local F = FormationCF(t)
    if not F then return end

    -- slash: swing the whole shape through an arc, with a white flash
    local flash = 0
    local slash = (now - slashT0) / 0.55
    if slash >= 0 and slash < 1 then
        local e = slash * slash * (3 - 2 * slash)
        F = F * CFrame.Angles(0, 0, math.rad(-110 + 220 * e))
            * CFrame.new(0, 0, -math.sin(math.pi * slash) * SIZES[S.sizeIdx] * 0.5)
        flash = math.sin(math.pi * slash) * 0.8
    end
    -- pulse: burst outward and back
    local scale = SIZES[S.sizeIdx]
    local pulse = (now - pulseT0) / 0.7
    if pulse >= 0 and pulse < 1 then
        local k = math.sin(math.pi * pulse)
        scale = scale * (1 + 0.45 * k)
        flash = math.max(flash, k * 0.6)
    end

    local mt = now - morphT0
    local n = ActiveN
    for i = 1, n do
        local d = dst[i]
        local a = math.clamp((mt - 0.35 * H(i, 9)) / 0.85, 0, 1)
        local pos = d
        if a < 1 then -- staggered, swirling morph
            local e = a * a * (3 - 2 * a)
            pos = src[i]:Lerp(d, e)
                + V3(math.cos(i * 1.7), math.sin(i * 2.3), math.cos(i * 3.1)) * (math.sin(math.pi * a) * 0.35)
        end
        cur[i] = pos
        cfs[i + 1] = CFrame.new(F:PointToWorldSpace(pos * scale))
    end
    cfs[1] = CFrame.new(F.Position)

    local ok = pcall(function() Workspace:BulkMoveTo(moveList, cfs, Enum.BulkMoveMode.FireCFrameChanged) end)
    if not ok then
        for i = 1, n + 1 do moveList[i].CFrame = cfs[i] end
    end

    -- colors: half the particles per frame is plenty smooth
    frame = frame + 1
    for i = 1 + (frame % 2), n, 2 do
        Parts[i].Color = PaletteColor(i, n, t, flash)
    end
    Light.Color = PaletteColor(math.floor(n / 2), n, t, flash)
    if S.trails then
        for _, tr in ipairs(Trails) do
            if tr.idx <= n then tr.trail.Color = ColorSequence.new(tr.part.Color) end
        end
    end
end
Track(RunService.RenderStepped:Connect(Step))

local function ClampPos(pos, size)
    local ss = ScreenSize()
    return Vector2.new(
        math.floor(math.clamp(pos.X, 4, math.max(4, ss.X - size.X - 4)) + 0.5),
        math.floor(math.clamp(pos.Y, 4, math.max(4, ss.Y - size.Y - 4)) + 0.5))
end
local function ApplyPos()
    local size = Expanded and Panel.AbsoluteSize or Vector2.new(40, 40)
    if size.X < 10 then size = Vector2.new(PANEL_W, 260) end
    PanelPos = ClampPos(PanelPos, size)
    Panel.Position = UDim2.fromOffset(PanelPos.X, PanelPos.Y)
    Icon.Position = UDim2.fromOffset(PanelPos.X, PanelPos.Y)
end

end

local function MakeDraggable(handle, onTap)
    local dragging, moved, startInput, startMouse, startPos = false, false, nil, nil, nil
    handle.InputBegan:Connect(function(inp)
        local ut = inp.UserInputType
        if ut ~= Enum.UserInputType.MouseButton1 and ut ~= Enum.UserInputType.Touch then return end
        dragging, moved, startInput = true, false, inp
        startMouse = Vector2.new(inp.Position.X, inp.Position.Y)
        startPos = PanelPos
        inp.Changed:Connect(function()
            if inp.UserInputState == Enum.UserInputState.End and startInput == inp and dragging then
                dragging = false
                if not moved and onTap then onTap() end
            end
        end)
    end)
    Track(UserInputService.InputChanged:Connect(function(inp)
        if not dragging then return end
        if inp.UserInputType == Enum.UserInputType.MouseMovement or inp == startInput then
            local d = Vector2.new(inp.Position.X, inp.Position.Y) - startMouse
            if not moved and d.Magnitude > 6 then moved = true end
            if moved then PanelPos = startPos + d; ApplyPos() end
        end
    end))
end

    elseif k == Enum.KeyCode.H then
        SetShown(not S.shown); RefreshUI()
    elseif k == Enum.KeyCode.Y then
        SetExpanded(not Expanded)
    end
end))
