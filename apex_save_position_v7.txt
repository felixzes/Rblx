local function CleanName(s, maxChars)
    local t = tostring(s or "")
    t = t:gsub("[<>&%c]", "")
    t = t:match("^%s*(.-)%s*$") or ""
    local len = utf8.len(t)
    if not len then return "" end
    maxChars = maxChars or 24
    if len > maxChars then t = t:sub(1, utf8.offset(t, maxChars + 1) - 1) end
    return t
end

local function FirstChars(s, n)
    local t = CleanName(s, 64)
    local len = utf8.len(t)
    if not len or len == 0 then return nil end
    if len > n then t = t:sub(1, utf8.offset(t, n + 1) - 1) end
    return t
end

local function ValidColor(c)
    return type(c) == "table" and type(c[1]) == "number" and type(c[2]) == "number" and type(c[3]) == "number"
end
local function ColorOf(c) return Color3.fromRGB(c[1], c[2], c[3]) end
local function HexOf(c) return string.format("#%02X%02X%02X", c[1], c[2], c[3]) end
local function ParseHex(s)
    local h = tostring(s or ""):match("^%s*#?(%x%x%x%x%x%x)%s*$")
    if not h then return nil end
    return {tonumber(h:sub(1, 2), 16), tonumber(h:sub(3, 4), 16), tonumber(h:sub(5, 6), 16)}
end
local function Fmt(v) return string.format("%.1f, %.1f, %.1f", v.X, v.Y, v.Z) end

local function CanFiles()
    return type(writefile) == "function" and type(readfile) == "function" and type(isfile) == "function"
end

local function DefaultData()
    return {
        v = 3, slots = {}, folders = {}, nextId = 1, nextFolderId = 1,
        selectedId = nil, view = "all", holdIdx = 3, returnMode = 0, tpMode = 1, markers = false, tourIdx = 2, tourLoop = false, autoIdx = 1,
        glass = 0.55, accent = 1, fx = true, sortMode = "order", death = nil,
    }
end

local function ValidSlot(s)
    if type(s) ~= "table" or type(s.c) ~= "table" or #s.c ~= 12 then return false end
    for i = 1, 12 do
        if type(s.c[i]) ~= "number" then return false end
    end
    return true
end

local function FindById(list, id)
    for i, v in ipairs(list) do
        if v.id == id then return v, i end
    end
    return nil, nil
end

-- Accepts v1 data (index-based selection, no ids/icons/colors/folders)
-- as well as v2, and always returns a clean v2 table.
local function Normalize(d)
    local out = DefaultData()

    local maxFolder = 0
    if type(d.folders) == "table" then
        for _, f in ipairs(d.folders) do
            local fid = type(f) == "table" and tonumber(f.id)
            if fid then
                fid = math.floor(fid)
                if fid > 0 and not FindById(out.folders, fid) then
                    local nm = CleanName(f.name, 16)
                    out.folders[#out.folders + 1] = {id = fid, name = (nm ~= "") and nm or ("Folder " .. fid)}
                    maxFolder = math.max(maxFolder, fid)
                end
            end
        end
    end
    out.nextFolderId = math.max(math.floor(tonumber(d.nextFolderId) or 1), maxFolder + 1)

    local seen, maxId = {}, 0
    if type(d.slots) == "table" then
        for _, s in ipairs(d.slots) do
            if ValidSlot(s) then
                local slot = {
                    name = CleanName(s.name, 24),
                    c = s.c,
                    fav = (s.fav == true),
                    icon = FirstChars(s.icon, 2) or ICON_PRESETS[1],
                    color = ValidColor(s.color) and {s.color[1], s.color[2], s.color[3]} or nil,
                    folder = tonumber(s.folder),
                }
                local id = math.floor(tonumber(s.id) or 0)
                if id > 0 and not seen[id] then
                    slot.id = id
                    seen[id] = true
                    maxId = math.max(maxId, id)
                end
                out.slots[#out.slots + 1] = slot
            end
        end
    end
    local nextId = math.max(math.floor(tonumber(d.nextId) or 1), maxId + 1)
    for i, slot in ipairs(out.slots) do
        if not slot.id then slot.id = nextId; nextId = nextId + 1 end
        if slot.name == "" then slot.name = "Pos " .. i end
        if not slot.color then
            local pc = PALETTE[(slot.id - 1) % #PALETTE + 1]
            slot.color = {pc[1], pc[2], pc[3]}
        end
        if slot.folder and not FindById(out.folders, slot.folder) then slot.folder = nil end
    end
    out.nextId = nextId

    local selId = tonumber(d.selectedId)
    if not selId and tonumber(d.selected) then
        local old = out.slots[math.floor(d.selected)]
        if old then selId = old.id end
    end
    if selId and FindById(out.slots, selId) then out.selectedId = selId end

    if d.view == "fav" or d.view == "all" then
        out.view = d.view
    elseif tonumber(d.view) and FindById(out.folders, tonumber(d.view)) then
        out.view = tonumber(d.view)
    end

    out.holdIdx = math.clamp(math.floor(tonumber(d.holdIdx) or 3), 1, #HOLD_OPTIONS)
    out.tpMode = math.clamp(math.floor(tonumber(d.tpMode) or 1), 1, #TP_MODES)
    out.markers = (d.markers == true)
    out.tourIdx = math.clamp(math.floor(tonumber(d.tourIdx) or 2), 1, #K.TourGaps)
    out.tourLoop = (d.tourLoop == true)
    out.autoIdx = math.clamp(math.floor(tonumber(d.autoIdx) or 1), 1, #K.Auto)
    -- v2 had a simple on/off autoReturn; it becomes returnMode 1 (selected slot)
    out.returnMode = math.clamp(math.floor(tonumber(d.returnMode) or ((d.autoReturn == true) and 1 or 0)), 0, 2)
    out.glass = math.clamp(tonumber(d.glass) or 0.55, GLASS_MIN, 1)
    out.accent = math.clamp(math.floor(tonumber(d.accent) or 1), 1, #ACCENTS)
    out.fx = (d.fx ~= false)
    local sm = d.sortMode
    out.sortMode = (sm == "near" or sm == "name" or sm == "new") and sm or "order"
    if ValidSlot(d.death) then out.death = {c = d.death.c} end
    out.ux, out.uy = tonumber(d.ux), tonumber(d.uy)
    return out
end

local function LoadFromFile()
    if not CanFiles() then return nil end
    local ok, raw = pcall(function()
        if isfile(FILE) then return readfile(FILE) end
        return nil
    end)
    if not ok or type(raw) ~= "string" then return nil end
    local ok2, decoded = pcall(function() return HttpService:JSONDecode(raw) end)
    if not ok2 or type(decoded) ~= "table" then return nil end
    return Normalize(decoded)
end

-- Lowest unused "Pos N": delete everything and the next save is "Pos 1".
local function NextDefaultName()
    local used = {}
    for _, s in ipairs(Data.slots) do
        local n = tonumber(string.match(s.name, "^Pos (%d+)$"))
        if n then used[n] = true end
    end
    local n = 1
    while used[n] do n = n + 1 end
    return "Pos " .. n
end

local function NextFolderName()
    local used = {}
    for _, f in ipairs(Data.folders) do
        local n = tonumber(string.match(f.name, "^Folder (%d+)$"))
        if n then used[n] = true end
    end
    local n = 1
    while used[n] do n = n + 1 end
    return "Folder " .. n
end

local function UniqueCopyName(base)
    local stem = base:gsub(" %(%d+%)$", "")
    local names = {}
    for _, s in ipairs(Data.slots) do names[s.name] = true end
    local n = 2
    local cand = stem .. " (" .. n .. ")"
    while names[cand] do
        n = n + 1
        cand = stem .. " (" .. n .. ")"
    end
    return cand
end

-- Two-tone gradient (lighter top-left to darker bottom-right) used for
-- the glass sheen and for button/stroke accents. Only ever put on plain
-- Frames (or on a UIStroke) — never directly on a TextLabel/TextButton,
-- since a UIGradient there also tints the text, not just the fill.
local function Grad2(c)
    return ColorSequence.new({
        ColorSequenceKeypoint.new(0, c:Lerp(Color3.new(1, 1, 1), 0.22)),
        ColorSequenceKeypoint.new(1, c:Lerp(Color3.new(0, 0, 0), 0.40)),
    })
end

-- GLASS: every translucent background registers here, so one slider
-- (MORE tab) makes the whole GUI more or less see-through. Text and
-- outlines stay solid so everything remains readable.
local GlassItems = {}
local function GlassTr(minTr, maxTr)
    return minTr + (maxTr - minTr) * (1 - Data.glass)
end
local function Glass(inst, minTr, maxTr, key, sheen)
    local entry = {inst = inst, minTr = minTr, maxTr = maxTr, key = key or "BtnBase"}
    if sheen then -- Frame-only decorative sheen; see the Grad2 note above
        local g = Instance.new("UIGradient")
        g.Rotation = (type(sheen) == "number") and sheen or 80
        g.Color = Grad2(inst.BackgroundColor3)
        g.Parent = inst
        entry.gradient = g
    end
    GlassItems[#GlassItems + 1] = entry
    inst.BackgroundTransparency = GlassTr(minTr, maxTr)
    return inst
end
local function ApplyGlass()
    local alive = {}
    for _, g in ipairs(GlassItems) do
        if g.inst.Parent then
            g.inst.BackgroundTransparency = GlassTr(g.minTr, g.maxTr)
            alive[#alive + 1] = g
        end
    end
    GlassItems = alive
end

local function MakeBtn(p)
    local b = U.Button({
        text = p.text, col = p.col or T.TxtPri, sz = p.sz or 12,
        bg = p.bg or T.BtnBase, tr = 0, size = p.size, pos = p.pos,
        z = p.z or 14, parent = p.parent,
    })
    U.Corner(b, UDim.new(0, 6))
    local st = U.Stroke(b, p.col or T.BorderGlow, 1.25, 0.5)
    do
        local g = Instance.new("UIGradient")
        g.Rotation = 90
        g.Color = Grad2(p.col or T.BorderGlow)
        g.Parent = st
    end
    if not p.solid then Glass(b, 0.15, 0.8, (p.bg == T.BtnSel) and "BtnSel" or "BtnBase") end
    return b, st
end

-- Every theme has its own animated scene. A scene lives inside a clipped
-- container (the panel's back layer, the load card) and is drawn behind
-- the controls, so it shows through the glass. A short full-screen burst
-- of the same style plays when you teleport or switch themes.
--   OCEAN waves · GALAXY stars · MATRIX binary rain · EMBER sparks
--   SAKURA cherry-blossom trees + petals · FROST snowfall
-- Turn it all off with FX on the MORE tab (the tint stays).
-- ════════════════════════════════════════════════════════════════
local Fx = {handles = {}, busy = false, BurstGui = nil}
local Builders = {}
Fx.PINKS = {
    Color3.fromRGB(255, 170, 205), Color3.fromRGB(255, 200, 222),
    Color3.fromRGB(255, 140, 185), Color3.fromRGB(255, 225, 236),
}

function Fx.Rand(a, b) return a + math.random() * (b - a) end

-- A small round dot, centered on its position.
function Fx.NewDot(parent, size, col, tr)
    local f = Instance.new("Frame")
    f.BackgroundColor3 = col
    f.BackgroundTransparency = tr or 0
    f.BorderSizePixel = 0
    f.AnchorPoint = Vector2.new(0.5, 0.5)
    f.Size = UDim2.new(0, size, 0, size)
    f.ZIndex = 2
    f.Parent = parent
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(1, 0)
    c.Parent = f
    return f
end

-- ── OCEAN: three layers of rolling waves along the bottom + bubbles
Builders.OCEAN = function(h)
    local SEG = 7
    local count = math.ceil(h.width / SEG)
    local layers = {
        {base = 30, amp = 5, wl = 80, speed = 1.0, tr = 0.80, phase = 0},
        {base = 22, amp = 4, wl = 56, speed = 1.7, tr = 0.72, phase = 2},
        {base = 15, amp = 3, wl = 40, speed = 2.4, tr = 0.62, phase = 4},
    }
    for li, L in ipairs(layers) do
        L.cols = {}
        local col = T.Accent:Lerp(Color3.new(1, 1, 1), (li - 1) * 0.12)
        for i = 1, count do
            local f = Instance.new("Frame")
            f.BackgroundColor3 = col
            f.BackgroundTransparency = L.tr
            f.BorderSizePixel = 0
            f.AnchorPoint = Vector2.new(0, 1)
            f.Position = UDim2.new(0, (i - 1) * SEG, 1, 0)
            f.Size = UDim2.new(0, SEG + 1, 0, L.base)
            f.ZIndex = 1 + li
            f.Parent = h.root
            L.cols[i] = f
        end
    end
    local bubbles = {}
    for i = 1, 7 do
        local b = Fx.NewDot(h.root, math.random(3, 6), Color3.new(1, 1, 1), 0.7)
        b.ZIndex = 6
        bubbles[i] = {f = b, x = Fx.Rand(0.05, 0.95), y = Fx.Rand(0.5, 1), vy = Fx.Rand(0.05, 0.14), ph = Fx.Rand(0, 6)}
    end
    local acc = 0
    return function(dt, t)
        acc = acc + dt
        if acc >= 1 / 30 then -- waves refresh at 30 Hz; plenty smooth, half the cost
            acc = 0
            for _, L in ipairs(layers) do
                for i, f in ipairs(L.cols) do
                    local x = (i - 1) * SEG
                    local hgt = L.base + L.amp * math.sin(x / L.wl * math.pi * 2 + t * L.speed + L.phase)
                    f.Size = UDim2.new(0, SEG + 1, 0, math.max(2, hgt))
                end
            end
        end
        for _, b in ipairs(bubbles) do
            b.y = b.y - b.vy * dt
            if b.y < -0.05 then b.y = 1.05; b.x = Fx.Rand(0.05, 0.95) end
            b.f.Position = UDim2.new(b.x + 0.02 * math.sin(t * 1.5 + b.ph), 0, b.y, 0)
        end
    end
end

-- ── GALAXY: nebula clouds, twinkling stars, an occasional shooting star
Builders.GALAXY = function(h)
    local nebulae = {}
    for i = 1, 2 do
        local n = Fx.NewDot(h.root, i == 1 and 150 or 110,
            i == 1 and T.Accent or Color3.fromRGB(80, 120, 255), 0.9)
        n.ZIndex = 1
        nebulae[i] = {f = n, x = Fx.Rand(0.2, 0.8), y = Fx.Rand(0.2, 0.8), ph = Fx.Rand(0, 6), sp = Fx.Rand(0.12, 0.25)}
    end
    local stars = {}
    for i = 1, 34 do
        local c = (math.random() < 0.5) and Color3.new(1, 1, 1) or T.Accent:Lerp(Color3.new(1, 1, 1), 0.5)
        local f = Fx.NewDot(h.root, (math.random() < 0.2) and 3 or 2, c, 0.3)
        f.ZIndex = 3
        stars[i] = {f = f, x = math.random(), y = math.random(), ph = Fx.Rand(0, 6.28), sp = Fx.Rand(1, 3), vx = Fx.Rand(-0.006, 0.006)}
    end
    local streak = Instance.new("Frame")
    streak.BackgroundColor3 = Color3.new(1, 1, 1)
    streak.BackgroundTransparency = 1
    streak.BorderSizePixel = 0
    streak.AnchorPoint = Vector2.new(0.5, 0.5)
    streak.Size = UDim2.new(0, 34, 0, 2)
    streak.Rotation = -35
    streak.ZIndex = 4
    streak.Parent = h.root
    local nextStreak = os.clock() + Fx.Rand(2, 5)
    local streakT0 = nil
    return function(dt, t)
        for _, n in ipairs(nebulae) do
            n.f.Position = UDim2.new(n.x + 0.08 * math.sin(t * n.sp + n.ph), 0,
                n.y + 0.06 * math.cos(t * n.sp * 0.8 + n.ph), 0)
        end
        for _, s in ipairs(stars) do
            s.x = s.x + s.vx * dt
            if s.x > 1.02 then s.x = -0.02 elseif s.x < -0.02 then s.x = 1.02 end
            s.f.Position = UDim2.new(s.x, 0, s.y, 0)
            s.f.BackgroundTransparency = 0.15 + 0.8 * (0.5 + 0.5 * math.sin(t * s.sp + s.ph))
        end
        local now = os.clock()
        if not streakT0 and now >= nextStreak then streakT0 = now end
        if streakT0 then
            local p = (now - streakT0) / 0.8
            if p >= 1 then
                streakT0 = nil
                nextStreak = now + Fx.Rand(5, 9)
                streak.BackgroundTransparency = 1
            else
                streak.Position = UDim2.new(1.0 - p * 1.1, 0, 0.05 + p * 0.7, 0)
                streak.BackgroundTransparency = 0.1 + 0.85 * math.abs(p - 0.5) * 2
            end
        end
    end
end

-- ── MATRIX: falling columns of 1s and 0s that fade out behind the head
Builders.MATRIX = function(h)
    local COLW, LINES, LH = 14, 15, 12
    local count = math.floor(h.width / COLW)
    local left = math.floor((h.width - count * COLW) / 2)
    local function RandText()
        local lines = {}
        for i = 1, LINES do lines[i] = (math.random() < 0.5) and "0" or "1" end
        return table.concat(lines, "\n")
    end
    local cols = {}
    for i = 1, count do
        local l = Instance.new("TextLabel")
        l.BackgroundTransparency = 1
        l.Font = Enum.Font.Code
        l.TextSize = 11
        l.TextColor3 = T.Accent
        l.TextTransparency = 0.3
        l.TextXAlignment = Enum.TextXAlignment.Center
        l.TextYAlignment = Enum.TextYAlignment.Top
        l.Text = RandText()
        l.Size = UDim2.new(0, COLW, 0, LINES * LH)
        l.ZIndex = 2
        l.Parent = h.root
        local g = Instance.new("UIGradient")
        g.Rotation = 90
        g.Transparency = NumberSequence.new({NumberSequenceKeypoint.new(0, 1), NumberSequenceKeypoint.new(1, 0.1)})
        g.Parent = l
        local c = {f = l, x = left + (i - 1) * COLW, y = -math.random(0, 220),
            sp = Fx.Rand(35, 95), nextChange = os.clock() + Fx.Rand(0.1, 0.4)}
        l.Position = UDim2.new(0, c.x, 0, c.y)
        cols[i] = c
    end
    return function(dt, t)
        local H = h.root.AbsoluteSize.Y
        local now = os.clock()
        for _, c in ipairs(cols) do
            c.y = c.y + c.sp * dt
            if c.y > H then c.y = -LINES * LH; c.sp = Fx.Rand(35, 95) end
            c.f.Position = UDim2.new(0, c.x, 0, c.y)
            if now >= c.nextChange then
                c.f.Text = RandText()
                c.nextChange = now + Fx.Rand(0.15, 0.45)
            end
        end
    end
end

-- ── SAKURA: two cherry-blossom trees that sway, plus drifting petals
function Fx.BuildTree(parent, xScale, scale)
    local W0, H0 = 76, 84
    local tree = Instance.new("Frame")
    tree.BackgroundTransparency = 1
    tree.BorderSizePixel = 0
    tree.AnchorPoint = Vector2.new(0.5, 1)
    tree.Position = UDim2.new(xScale, 0, 1, 4)
    tree.Size = UDim2.new(0, W0 * scale, 0, H0 * scale)
    tree.ZIndex = 3
    tree.Parent = parent
    local function px(v) return v * scale end

    local function Bar(cx, cy, w, hgt, rot)
        local f = Instance.new("Frame")
        f.BackgroundColor3 = Color3.fromRGB(92, 56, 52)
        f.BorderSizePixel = 0
        f.AnchorPoint = Vector2.new(0.5, 0.5)
        f.Position = UDim2.new(0, px(cx), 0, px(cy))
        f.Size = UDim2.new(0, math.max(2, px(w)), 0, px(hgt))
        f.Rotation = rot or 0
        f.ZIndex = 3
        f.Parent = tree
        return f
    end
    Bar(38, 62, 6, 44, 0) -- trunk

    -- branch: starts on the trunk at height py, length len, angle in
    -- degrees clockwise from straight up; its tip grows a blossom cluster.
    local tips = {}
    local function Branch(py, len, ang)
        local r = math.rad(ang)
        Bar(38 + (len / 2) * math.sin(r), py - (len / 2) * math.cos(r), 3, len, ang)
        tips[#tips + 1] = {x = 38 + len * math.sin(r), y = py - len * math.cos(r)}
    end
    Branch(50, 26, -42)
    Branch(46, 28, 40)
    Branch(38, 22, -12)
    Branch(34, 20, 18)
    tips[#tips + 1] = {x = 38, y = 26} -- crown

    for _, tp in ipairs(tips) do
        for k = 1, 11 do
            local a = math.random() * math.pi * 2
            local rr = math.random() * 12
            local d = Fx.NewDot(tree, px(Fx.Rand(5, 9)), Fx.PINKS[math.random(#Fx.PINKS)], Fx.Rand(0, 0.35))
            d.Position = UDim2.new(0, px(tp.x + rr * math.cos(a)), 0, px(tp.y + rr * math.sin(a) * 0.8))
            d.ZIndex = 4
        end
    end
    return tree
end

Builders.SAKURA = function(h)
    local trees = {
        {f = Fx.BuildTree(h.root, 0.14, 1.0), ph = 0},
        {f = Fx.BuildTree(h.root, 0.88, 0.8), ph = 2},
    }
    local petals = {}
    for i = 1, 18 do
        local p = Fx.NewDot(h.root, 8, Fx.PINKS[math.random(#Fx.PINKS)], 0.2)
        p.Size = UDim2.new(0, math.random(6, 9), 0, math.random(4, 6))
        p.ZIndex = 5
        petals[i] = {f = p, x = math.random(), y = Fx.Rand(-0.1, 1.0), vy = Fx.Rand(0.05, 0.12),
            ph = Fx.Rand(0, 6.28), sw = Fx.Rand(0.02, 0.05), rs = Fx.Rand(40, 120)}
    end
    return function(dt, t)
        for _, tr in ipairs(trees) do tr.f.Rotation = 1.6 * math.sin(t * 0.9 + tr.ph) end
        for _, p in ipairs(petals) do
            p.y = p.y + p.vy * dt
            if p.y > 1.05 then p.y = -0.05; p.x = math.random() end
            p.f.Position = UDim2.new(p.x + p.sw * math.sin(t * 1.3 + p.ph), 0, p.y, 0)
            p.f.Rotation = p.ph * 30 + t * p.rs
            p.f.BackgroundTransparency = 0.1 + 0.25 * (0.5 + 0.5 * math.sin(t * 2 + p.ph))
        end
    end
end

-- ── engine core
function Fx.Build(h)
    for _, ch in ipairs(h.root:GetChildren()) do ch:Destroy() end
    h.update = nil
    if not Data.fx then return end
    local th = ACCENTS[Data.accent] or ACCENTS[1]
    local builder = Builders[th.fx]
    if builder then h.update = builder(h) end
end

function Fx.Attach(root, width)
    local h = {root = root, width = width}
    Fx.handles[#Fx.handles + 1] = h
    Fx.Build(h)
    return h
end

function Fx.RebuildAll()
    for _, h in ipairs(Fx.handles) do Fx.Build(h) end
end

local fxClock = 0
Track(RunService.RenderStepped:Connect(function(dt)
    dt = math.min(dt, 0.1)
    fxClock = fxClock + dt
    local live = {}
    for _, h in ipairs(Fx.handles) do
        if h.root.Parent then
            live[#live + 1] = h
            if h.update and h.root.Visible then
                local ok, err = pcall(h.update, dt, fxClock)
                if not ok then
                    h.update = nil -- never let a scene error take the GUI down
                    warn("[ApexSavePosition] theme effect error: " .. tostring(err))
                end
            end
        end
    end
    Fx.handles = live
end))

-- ── full-screen burst: the same style, thrown across the screen for a
-- moment on each teleport or theme change. Tween-driven, so it's cheap.
function Fx.Spawn(kind)
    local parent = Fx.BurstGui
    if not parent then return end
    local acc = T.Accent
    local life = Fx.Rand(0.9, 1.6)
    local lin = TweenInfo.new(life, Enum.EasingStyle.Linear)
    local obj
    if kind == "MATRIX" then
        local x = math.random()
        obj = Instance.new("TextLabel")
        obj.BackgroundTransparency = 1
        obj.Font = Enum.Font.Code
        obj.Text = (math.random() < 0.5) and "0" or "1"
        obj.TextSize = math.random(14, 24)
        obj.TextColor3 = acc
        obj.Size = UDim2.new(0, 18, 0, 22)
        obj.AnchorPoint = Vector2.new(0.5, 0.5)
        obj.Position = UDim2.new(x, 0, -0.05, 0)
        obj.ZIndex = 5
        obj.Parent = parent
        TweenService:Create(obj, lin, {Position = UDim2.new(x, 0, 1.05, 0), TextTransparency = 1}):Play()
    elseif kind == "SAKURA" then
        local x0 = Fx.Rand(0.2, 1.1)
        obj = Fx.NewDot(parent, 8, Fx.PINKS[math.random(#Fx.PINKS)], 0.1)
        obj.Size = UDim2.new(0, math.random(8, 12), 0, math.random(5, 8))
        obj.Position = UDim2.new(x0, 0, -0.05, 0)
        obj.Rotation = math.random(0, 180)
        obj.ZIndex = 5
        TweenService:Create(obj, lin, {
            Position = UDim2.new(x0 - Fx.Rand(0.2, 0.5), 0, 1.05, 0),
            Rotation = obj.Rotation + math.random(180, 540),
            BackgroundTransparency = 0.85,
        }):Play()
    elseif kind == "OCEAN" then
        local x = math.random()
        obj = Fx.NewDot(parent, math.random(4, 11), Color3.new(1, 1, 1), 0.45)
        obj.Position = UDim2.new(x, 0, 1.05, 0)
        obj.ZIndex = 5
        TweenService:Create(obj, lin, {
            Position = UDim2.new(x + Fx.Rand(-0.05, 0.05), 0, -0.05, 0), BackgroundTransparency = 1,
        }):Play()
    elseif kind == "GALAXY" then
        obj = Fx.NewDot(parent, 2, Color3.new(1, 1, 1), 0.1)
        obj.Position = UDim2.new(math.random(), 0, math.random(), 0)
        obj.ZIndex = 5
        TweenService:Create(obj, lin, {Size = UDim2.new(0, 10, 0, 10), BackgroundTransparency = 1}):Play()
    elseif kind == "EMBER" then
        local x = math.random()
        obj = Fx.NewDot(parent, math.random(3, 6),
            (math.random() < 0.5) and acc or Color3.fromRGB(255, 110, 40), 0.1)
        obj.Position = UDim2.new(x, 0, 1.05, 0)
        obj.ZIndex = 5
        TweenService:Create(obj, lin, {
            Position = UDim2.new(x + Fx.Rand(-0.08, 0.08), 0, Fx.Rand(0.1, 0.6), 0), BackgroundTransparency = 1,
        }):Play()
    else -- FROST
        local x = math.random()
        obj = Fx.NewDot(parent, math.random(3, 7), Color3.new(1, 1, 1), 0.15)
        obj.Position = UDim2.new(x, 0, -0.05, 0)
        obj.ZIndex = 5
        TweenService:Create(obj, lin, {
            Position = UDim2.new(x + Fx.Rand(-0.1, 0.1), 0, 1.05, 0), BackgroundTransparency = 0.9,
        }):Play()
    end
    task.delay(life + 0.15, function()
        if obj then obj:Destroy() end
    end)
end

function Fx.Burst()
    if not Data.fx or Fx.busy then return end
    Fx.busy = true
    task.delay(0.7, function() Fx.busy = false end)
    local kind = (ACCENTS[Data.accent] or ACCENTS[1]).fx
    for i = 1, 28 do
        task.delay(math.random() * 0.4, function() Fx.Spawn(kind) end)
    end
end

        size = UDim2.new(1, -40, 0, 6), pos = UDim2.new(0, 20, 0, 92), z = 3, parent = card})
    U.Corner(track, UDim.new(0, 3))
    local fill = U.Frame({bg = T.Accent, tr = 0, size = UDim2.new(0, 0, 1, 0), z = 4, parent = track})
    U.Corner(fill, UDim.new(0, 3))

    local stageLabel = U.Label({text = "Starting...", col = T.TxtSec, font = T.FontBold, sz = 11,
        size = UDim2.new(1, -24, 0, 18), pos = UDim2.new(0, 12, 0, 104), z = 3, parent = card})
    local tip = U.Label({text = "TIP  " .. TIPS[math.random(#TIPS)], col = T.TxtDim, font = T.FontMed, sz = 10,
        size = UDim2.new(1, -28, 0, 30), pos = UDim2.new(0, 14, 0, 132), z = 3, parent = card})
    tip.TextWrapped = true

    local progress, target = 0, 0
    local skip = false
    local startT = os.clock()
    local conn = RunService.RenderStepped:Connect(function(dt)
        progress = progress + (target - progress) * math.min(1, dt * 7)
        fill.Size = UDim2.new(math.clamp(progress, 0, 1), 0, 1, 0)
        local t = os.clock() - startT
        for i, d in ipairs(dots) do
            d.BackgroundTransparency = 0.15 + 0.75 * (0.5 + 0.5 * math.sin(t * 6 - i * 0.9))
        end
    end)
    Track(conn)

    local function OnInput(inp)
        local ut = inp.UserInputType
        if ut == Enum.UserInputType.MouseButton1 or ut == Enum.UserInputType.Touch then skip = true end
    end
    card.InputBegan:Connect(OnInput)
    backdrop.InputBegan:Connect(OnInput)

    -- stages = { {progress 0..1, text}, ... }; shown one after another
    -- over LOAD_MIN_SECONDS, then the card fades away and onDone runs.
    function Loader.Finish(stages, onDone)
        task.spawn(function()
            local interval = LOAD_MIN_SECONDS / math.max(1, #stages)
            for _, st in ipairs(stages) do
                if skip then break end
                target = st[1]
                stageLabel.Text = st[2]
                local t0 = os.clock()
                while os.clock() - t0 < interval and not skip do task.wait(0.03) end
            end
            target = 1
            progress = 1
            fill.Size = UDim2.new(1, 0, 1, 0)
            local last = stages[#stages]
            stageLabel.Text = last and last[2] or "Ready"
            task.wait(skip and 0.05 or 0.25)
            TweenService:Create(card, TweenInfo.new(0.35), {GroupTransparency = 1}):Play()
            TweenService:Create(backdrop, TweenInfo.new(0.35), {BackgroundTransparency = 1}):Play()
            task.wait(0.4)
            pcall(function() conn:Disconnect() end)
            gui:Destroy()
            if onDone then onDone() end
        end)
    end

    -- Failsafe: never leave a tap-blocking overlay behind if init fails.
    task.delay(12, function()
        pcall(function() conn:Disconnect() end)
        if gui.Parent then gui:Destroy() end
    end)
end

local function ActiveSize()
    if Expanded then
        local s = Panel.AbsoluteSize
        if s.X < 10 or s.Y < 10 then return Vector2.new(PANEL_W, 260) end
        return s
    end
    return Vector2.new(MINI_W, MINI_H)
end
local function ClampToScreen(pos, size)
    local ss = ScreenSize()
    return Vector2.new(
        math.clamp(pos.X, 4, math.max(4, ss.X - size.X - 4)),
        math.clamp(pos.Y, 4, math.max(4, ss.Y - size.Y - 4)))
end
local function AvoidCenter(pos, size)
    local ss = ScreenSize()
    local cx, cy = ss.X / 2, ss.Y / 2
    local hw, hh = 48, 48
    if pos.X + size.X > cx - hw and pos.X < cx + hw and pos.Y + size.Y > cy - hh and pos.Y < cy + hh then
        local leftX, rightX = (cx - hw) - size.X, cx + hw
        if math.abs(pos.X - leftX) < math.abs(pos.X - rightX) then
            return Vector2.new(leftX, pos.Y)
        end
        return Vector2.new(rightX, pos.Y)
    end
    return pos
end
local function ApplyPos()

    local p = ClampToScreen(PanelPos, size)
    p = AvoidCenter(p, size)
    p = ClampToScreen(p, size)
    p = Vector2.new(math.floor(p.X + 0.5), math.floor(p.Y + 0.5))
    PanelPos = p
    Panel.Position = UDim2.fromOffset(p.X, p.Y)
    W.Mini.Position = UDim2.fromOffset(p.X, p.Y)

-- Drag by the title bar (expanded) or the bar itself (collapsed).
local function MakeDraggable(handle)
    local dragging, moved = false, false
    local startInput, startMouse, startPos
    handle.InputBegan:Connect(function(inp)
        local ut = inp.UserInputType
        if ut ~= Enum.UserInputType.MouseButton1 and ut ~= Enum.UserInputType.Touch then return end
        dragging, moved = true, false
        DraggingNow = true
        startInput = inp
        startMouse = Vector2.new(inp.Position.X, inp.Position.Y)
        startPos = PanelPos
        Bump()
        inp.Changed:Connect(function()
            if inp.UserInputState == Enum.UserInputState.End and startInput == inp and dragging then
                dragging = false
                DraggingNow = false
                if moved then Persist() end
            end
        end)
    end)
    Track(UserInputService.InputChanged:Connect(function(inp)
        if not dragging then return end
        if inp.UserInputType == Enum.UserInputType.MouseMovement or inp == startInput then
            local delta = Vector2.new(inp.Position.X, inp.Position.Y) - startMouse
            if not moved and delta.Magnitude > 6 then moved = true end
            if moved then
                PanelPos = startPos + delta
                ApplyPos()
                Bump()
            end
        end
    end))
end

-- ════════════════════════════════════════════════════════════════
-- TELEPORT + SHIELD
-- SHIELD is what stops a game putting you back, and unlike a position
-- lock it never pins you in place:
--  1. Arrival pin: for ~1.4s after a teleport your spot is re-applied
--     on every frame signal; moving the stick releases it early.
--  2. Jump rejection: every frame (Stepped, Heartbeat, RenderStepped AND
--     the instant the root's CFrame changes) your position is compared
--     with last frame's. A sudden jump you didn't cause is reversed, with
--     your momentum kept — walking, falling and vehicles are untouched.
--  3. Pull-back parts: Body movers / constraints the game adds to your
--     character are destroyed, and anchoring your root is undone.
--  4. Script hook (executors with hookmetamethod): while SHIELD is on, the
--     game's own scripts can't move your character via CFrame / PivotTo /
--     SetPrimaryPartCFrame / Model:MoveTo. Without hook support the
--     position guard (1-3) still runs.
-- STEP teleport glides to the spot with collisions off instead of one
-- big jump, which gets past anti-cheats that flag a single large move.
-- ════════════════════════════════════════════════════════════════
local LastOrigin = nil -- where you stood before the most recent teleport (for BACK)

local Shield = {on = false, untilT = 0, lastPos = nil, lastCF = nil, lastT = 0,
                arrival = nil, fixes = 0, conns = {}, self = false}
ENV.ApexSaveShield = Shield

local function DurText(sec)
    if sec < 0 then return "always" end
    if sec >= 120 then return math.floor(sec / 60) .. "m" end
    return sec .. "s"
end

-- Our own moves are marked (Shield.self) so neither the hook nor the
-- jump detector ever mistakes them for the game's.
local function ApplyCF(cf, keepVelocity)
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return false end
    local vel = root.AssemblyLinearVelocity
    Shield.self = true
    local ok = pcall(function() char:PivotTo(cf) end)
    if not ok then root.CFrame = cf end
    if keepVelocity then
        root.AssemblyLinearVelocity = vel
    else
        root.AssemblyLinearVelocity = Vector3.zero
    end
    root.AssemblyAngularVelocity = Vector3.zero
    Shield.self = false
    Shield.lastPos, Shield.lastCF, Shield.lastT = cf.Position, cf, os.clock()
    return true
end

local function IsMineInstance(inst)
    local char = LP.Character
    if not char then return false end
    local p, depth = inst, 0
    while p and depth < 12 do
        if p == char then return true end
        p = p.Parent
        depth = depth + 1
    end
    return false
end

local HookState = ENV.ApexSaveHookState
if type(HookState) ~= "table" then
    HookState = {installed = false}
    ENV.ApexSaveHookState = HookState
end
-- Installed lazily (first time SHIELD turns on) so it costs nothing when
-- unused. The hooks read ENV.ApexSaveShield, so reloading never stacks them.
local function InstallHooks()
    if HookState.installed then return end
    if type(hookmetamethod) ~= "function" or type(checkcaller) ~= "function"
        or type(newcclosure) ~= "function" or type(getnamecallmethod) ~= "function" then return end
    local BLOCK = {PivotTo = true, SetPrimaryPartCFrame = true, MoveTo = true, TranslateBy = true}
    local ROOTS = {HumanoidRootPart = true, Torso = true, UpperTorso = true}
    local ok = pcall(function()
        local oldNI
        oldNI = hookmetamethod(game, "__newindex", newcclosure(function(self, key, value)
            local sh = ENV.ApexSaveShield
            if sh and sh.on and not sh.self and (key == "CFrame" or key == "Position")
                and typeof(self) == "Instance" and ROOTS[self.Name] and not checkcaller()
                and IsMineInstance(self) then
                return
            end
            return oldNI(self, key, value)
        end))
        local oldNC
        oldNC = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
            local sh = ENV.ApexSaveShield
            if sh and sh.on and not sh.self then
                local method = getnamecallmethod()
                if BLOCK[method] and typeof(self) == "Instance" and not checkcaller()
                    and IsMineInstance(self) and (method ~= "MoveTo" or self.ClassName == "Model") then
                    return
                end
            end
            return oldNC(self, ...)
        end))
    end)
    HookState.installed = ok
end

local FORCE_CLASSES = {
    BodyPosition = true, BodyVelocity = true, BodyGyro = true, BodyForce = true, BodyThrust = true,
    BodyAngularVelocity = true, RocketPropulsion = true, AlignPosition = true, AlignOrientation = true,
    LinearVelocity = true, AngularVelocity = true, VectorForce = true, LineForce = true, Torque = true,
}

local function ShieldClearConns()
    for _, c in ipairs(Shield.conns) do pcall(function() c:Disconnect() end) end
    Shield.conns = {}
end

local function ShieldTick()
    if not Shield.on and not Shield.arrival then return end
    local root = GetRoot()
    if not root then return end
    local now = os.clock()
    local pos = root.Position

    -- 1) arrival pin
    local arr = Shield.arrival
    if arr then
        if now > arr.untilT then
            Shield.arrival = nil
        else
            local hum = GetHum()
            if arr.releaseOnMove and now - arr.t0 > ARRIVAL_GRACE and hum and hum.MoveDirection.Magnitude > 0.1 then
                Shield.arrival = nil
            elseif (pos - arr.cf.Position).Magnitude > PULLBACK_DIST then
                ApplyCF(arr.cf)
                arr.fixes = arr.fixes + 1
                Shield.fixes = Shield.fixes + 1
                pos = arr.cf.Position
            end
        end
    end

    -- 2) reject sudden jumps we didn't cause (never locks normal movement)
    if Shield.on and Shield.lastPos and not Shield.arrival then
        local dt = math.clamp(now - Shield.lastT, 0, 0.25)
        local allowed = 8 + root.AssemblyLinearVelocity.Magnitude * dt * 2.5
        if (pos - Shield.lastPos).Magnitude > allowed then
            ApplyCF(Shield.lastCF, true)
            Shield.fixes = Shield.fixes + 1
            pos = Shield.lastCF.Position
        end
    end

    Shield.lastPos, Shield.lastCF, Shield.lastT = pos, root.CFrame, now
end
Track(RunService.Stepped:Connect(ShieldTick))
Track(RunService.Heartbeat:Connect(ShieldTick))
Track(RunService.RenderStepped:Connect(ShieldTick))
table.insert(ENV.ApexSavePosLoops, {Disconnect = ShieldClearConns})

function Shield.Bind(char)
    ShieldClearConns()
    local root = char:WaitForChild("HumanoidRootPart", 10)
    if not root then return end
    local c = Shield.conns
    c[#c + 1] = root:GetPropertyChangedSignal("CFrame"):Connect(ShieldTick) -- instant reaction
    c[#c + 1] = root:GetPropertyChangedSignal("Anchored"):Connect(function()
        if Shield.on and root.Anchored then root.Anchored = false end
    end)
    c[#c + 1] = char.DescendantAdded:Connect(function(d)
        if Shield.on and FORCE_CLASSES[d.ClassName] then pcall(function() d:Destroy() end) end
    end)
    Shield.lastPos, Shield.lastCF, Shield.lastT = root.Position, root.CFrame, os.clock()
end

function Shield.Start(sec)
    if not Shield.on then Shield.fixes = 0 end
    Shield.on = true
    Shield.untilT = (sec < 0) and math.huge or (os.clock() + sec)
    local root = GetRoot()
    if root then Shield.lastPos, Shield.lastCF, Shield.lastT = root.Position, root.CFrame, os.clock() end
    InstallHooks()
end

function Shield.Stop()
    Shield.on = false
    Shield.arrival = nil
end

function Shield.Arrive(cf, pin)
    Shield.arrival = {cf = cf, t0 = os.clock(), untilT = os.clock() + (pin and ARRIVAL_SECONDS or 0.35),
        releaseOnMove = pin, fixes = 0}
end

-- Glide to the target in small steps, collisions off, instead of one jump.
local function StepTo(cf, speed, token)
    local root = GetRoot()
    local char = LP.Character
    if not (root and char) then return end
    local start, goal = root.Position, cf.Position
    local dur = math.clamp((goal - start).Magnitude / speed, 0.15, 12)
    local touched = {}
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false; touched[#touched + 1] = p end
    end
    local t0 = os.clock()
    while ENV.ApexSaveHold == token do
        local a = (os.clock() - t0) / dur
        if a >= 1 or not GetRoot() then break end
        ApplyCF(CFrame.new(start:Lerp(goal, a)) * cf.Rotation)
        RunService.Heartbeat:Wait()
    end
    for _, p in ipairs(touched) do
        if p.Parent then p.CanCollide = true end
    end
    if ENV.ApexSaveHold == token then ApplyCF(cf) end
end

local function TeleportTo(cf, label)
    local rootNow = GetRoot()
    if not rootNow then
        SetStatus("No character yet", T.Red)
        return false
    end
    LastOrigin = rootNow.CFrame

    ENV.ApexSaveHold = ENV.ApexSaveHold + 1
    local token = ENV.ApexSaveHold
    local dur = HOLD_OPTIONS[Data.holdIdx] or 0
    local mode = TP_MODES[Data.tpMode] or TP_MODES[1]

    -- A seated character stays welded to its seat, so stand up first.
    local hum = GetHum()
    if hum and hum.SeatPart then hum.Sit = false end

    if dur ~= 0 then Shield.Start(dur) end -- armed BEFORE the move

    task.spawn(function()
        if mode.speed > 0 then
            SetStatus("Gliding to " .. label .. " (" .. mode.name .. ")...", T.Yellow)
            StepTo(cf, mode.speed, token)
        else
            ApplyCF(cf)
        end
        if ENV.ApexSaveHold ~= token or not GetRoot() then return end
        Flash()
        SetStatus("Teleported to " .. label .. ((dur ~= 0) and ("  ·  shield " .. DurText(dur)) or ""), T.Green)
        Shield.Arrive(cf, dur ~= 0)
        local arr = Shield.arrival
        local t0 = os.clock()
        while ENV.ApexSaveHold == token and Shield.arrival == arr and os.clock() - t0 < ARRIVAL_SECONDS + 0.2 do
            task.wait(0.05)
        end
        if ENV.ApexSaveHold == token and arr and arr.fixes > 0 then
            SetStatus(("Countered %d pull-back%s at %s — you're free to move"):format(
                arr.fixes, arr.fixes == 1 and "" or "s", label), T.Yellow)
        end
    end)
    return true
end

-- Slots in the current view, favorites first. In NEAR mode each group
-- is ordered by distance from you; the second return is id -> studs.
local function VisibleSlots()
    local favs, rest = {}, {}
    for _, s in ipairs(Data.slots) do
        local show
        if Data.view == "all" then show = true
        elseif Data.view == "fav" then show = s.fav
        else show = (s.folder == Data.view) end
        if show then
            if s.fav then favs[#favs + 1] = s else rest[#rest + 1] = s end
        end
    end

    local dist = nil
    if Data.sortMode == "near" then
        local root = GetRoot()
        if root then
            dist = {}
            local here = root.Position
            for _, group in ipairs({favs, rest}) do
                for _, s in ipairs(group) do
                    dist[s.id] = (SlotCF(s).Position - here).Magnitude
                end
                table.sort(group, function(a, b) return dist[a.id] < dist[b.id] end)
            end
        end
    end

    if Data.sortMode == "name" then
        for _, group in ipairs({favs, rest}) do
            table.sort(group, function(a, b) return string.lower(a.name) < string.lower(b.name) end)
        end
    elseif Data.sortMode == "new" then
        for _, group in ipairs({favs, rest}) do
            table.sort(group, function(a, b) return a.id > b.id end)
        end
    end

    for _, s in ipairs(rest) do favs[#favs + 1] = s end
    return favs, dist
end

local function AddSlot(cf, name)
    local id = Data.nextId
    Data.nextId = id + 1
    local pc = PALETTE[(id - 1) % #PALETTE + 1]
    local slot = {
        id = id, name = name, c = {cf:GetComponents()},
        fav = (Data.view == "fav"),
        icon = ICON_PRESETS[1],
        color = {pc[1], pc[2], pc[3]},
        folder = (type(Data.view) == "number") and Data.view or nil,
    }
    Data.slots[#Data.slots + 1] = slot
    Data.selectedId = id
    return slot
end

local function SaveHere()
    local root = GetRoot()
    if not root then SetStatus("No character to save", T.Red); Expand(); return end
    local name = CleanName(W.Name.Text, 24)
    if name == "" then name = NextDefaultName() end
    local slot = AddSlot(root.CFrame, name)
    W.Name.Text = ""
    Persist()
    RefreshAll()
    SetStatus("Saved " .. slot.name .. "  (" .. Fmt(root.Position) .. ")", T.Green)
    task.defer(function()
        W.List.CanvasPosition = Vector2.new(0, math.max(0, W.List.AbsoluteCanvasSize.Y))
    end)
end

local function ReadTypedXYZ()
    local x, y, z = tonumber(Xy.X.Text), tonumber(Xy.Y.Text), tonumber(Xy.Z.Text)
    if not (x and y and z) then
        SetStatus("Enter numbers in X, Y and Z", T.Red)
        Expand()
        return nil
    end
    local root = GetRoot()
    -- keep the way you're currently facing
    local rot = root and root.CFrame.Rotation or CFrame.new()
    return CFrame.new(x, y, z) * rot, x, y, z
end

local function GoToTyped()
    local cf, x, y, z = ReadTypedXYZ()
    if not cf then return end
    if not GetRoot() then SetStatus("No character yet", T.Red); return end
    TeleportTo(cf, string.format("%.1f, %.1f, %.1f", x, y, z))
end

local function SaveTypedXYZ()
    local cf, x, y, z = ReadTypedXYZ()
    if not cf then return end
    local name = CleanName(W.Name.Text, 24)
    if name == "" then name = NextDefaultName() end
    local slot = AddSlot(cf, name)
    W.Name.Text = ""
    Persist()
    RefreshAll()
    SetStatus("Saved " .. slot.name .. " at " .. string.format("%.1f, %.1f, %.1f", x, y, z), T.Green)
end

local function DuplicateSlot(slot)
    local _, idx = FindSlot(slot.id)
    local id = Data.nextId
    Data.nextId = id + 1
    local c = {}
    for i = 1, 12 do c[i] = slot.c[i] end
    local copy = {
        id = id, name = UniqueCopyName(slot.name), c = c, fav = slot.fav,
        icon = slot.icon, color = {slot.color[1], slot.color[2], slot.color[3]}, folder = slot.folder,
    }
    table.insert(Data.slots, (idx or #Data.slots) + 1, copy)
    Data.selectedId = id
    return copy
end

local function DeleteSlot(slot)
    local _, idx = FindSlot(slot.id)
    if not idx then return end
    table.remove(Data.slots, idx)
    if Data.selectedId == slot.id then
        local nxt = Data.slots[math.min(idx, #Data.slots)]
        Data.selectedId = nxt and nxt.id or nil
    end
end

local function CreateFolder(name)
    local id = Data.nextFolderId
    Data.nextFolderId = id + 1
    local nm = CleanName(name, 16)
    if nm == "" then nm = NextFolderName() end
    Data.folders[#Data.folders + 1] = {id = id, name = nm}
    Data.view = id
    return nm
end

local function CurrentFolder()
    if type(Data.view) == "number" then return (FindById(Data.folders, Data.view)) end
    return nil
end

local function CycleSelection(dir)
    local list = VisibleSlots()
    if #list == 0 then SetStatus("No saved positions in this view", T.Yellow); return end
    local idx = 0
    for i, s in ipairs(list) do
        if s.id == Data.selectedId then idx = i; break end
    end
    if idx == 0 then
        idx = (dir > 0) and 1 or #list
    else
        idx = ((idx - 1 + dir) % #list) + 1
    end
    Data.selectedId = list[idx].id
    Persist()
    RebuildList()
    SetStatus("Selected: " .. list[idx].name, T.Cyan)
end

RebuildFolders = function()
    for _, ch in ipairs(W.Folders:GetChildren()) do
        if ch:IsA("TextButton") then ch:Destroy() end
    end
    local order = 0
    local function AddChip(text, width, isSel, textCol, onClick)
        order = order + 1
        local b, st = MakeBtn({text = text, col = isSel and T.Accent or textCol,
            bg = isSel and T.BtnSel or T.BtnBase, sz = 11,
            size = UDim2.new(0, width, 0, CHIP_H - 2), parent = W.Folders})

        if isSel then st.Transparency = 0.05; st.Thickness = 1.75 end

    end)

        for _, s in ipairs(Data.slots) do

        Persist(); RefreshAll()

    end)

    Mo.Box.PlaceholderText = cur and ("Rename: " .. cur.name) or "Folder name"

-- Glass slider: how see-through the whole GUI is.
local function SetGlass(v)
    Data.glass = math.clamp(v, GLASS_MIN, 1)
    local t = (Data.glass - GLASS_MIN) / (1 - GLASS_MIN)
    Mo.Fill.Size = UDim2.new(t, 0, 1, 0)
    Mo.Knob.Position = UDim2.new(t, -8, 0.5, -8)
    Mo.GlassLabel.Text = "GLASS " .. math.floor(Data.glass * 100 + 0.5) .. "%"
    ApplyGlass()
end
local function GlassFromX(x)
    local w = Mo.SliderHit.AbsoluteSize.X
    if w <= 0 then return end
    local t = math.clamp((x - Mo.SliderHit.AbsolutePosition.X - 4) / (w - 8), 0, 1)
    SetGlass(GLASS_MIN + t * (1 - GLASS_MIN))
end
local glassActive, glassInput = false, nil
Mo.SliderHit.InputBegan:Connect(function(inp)
    local ut = inp.UserInputType
    if ut ~= Enum.UserInputType.MouseButton1 and ut ~= Enum.UserInputType.Touch then return end
    glassActive = true
    glassInput = inp
    DraggingNow = true -- keeps auto-minimize from firing mid-drag
    Bump()
    GlassFromX(inp.Position.X)
    inp.Changed:Connect(function()
        if inp.UserInputState == Enum.UserInputState.End then
            glassActive = false
            DraggingNow = false
            Persist()
        end
    end)
end)
Track(UserInputService.InputChanged:Connect(function(inp)
    if not glassActive then return end
    if inp.UserInputType == Enum.UserInputType.MouseMovement or inp == glassInput then
        GlassFromX(inp.Position.X)
        Bump()
    end
end))

-- Paste "x, y, z" into the X box and it spreads across all three.
local function ParseNumbers(str)
    local nums = {}
    for n in string.gmatch(str, "%-?%d+%.?%d*") do
        nums[#nums + 1] = tonumber(n)
    end
    return nums
end
Xy.X.FocusLost:Connect(function()
    local nums = ParseNumbers(Xy.X.Text)
    if #nums >= 3 then
        Xy.X.Text = tostring(nums[1])
        Xy.Y.Text = tostring(nums[2])
        Xy.Z.Text = tostring(nums[3])
    end
end)

-- ── TOUR: visit every position in the current view, in order
function Tour.Stop()
    Tour.running = false
    ENV.ApexSaveTour = ENV.ApexSaveTour + 1
end
function Tour.Start()
    local list = VisibleSlots()
    if #list == 0 then SetStatus("Nothing to tour in this view", T.Yellow); return end
    Tour.running = true
    ENV.ApexSaveTour = ENV.ApexSaveTour + 1
    local token = ENV.ApexSaveTour
    task.spawn(function()
        local i = 1
        while Tour.running and ENV.ApexSaveTour == token do
            local slot = list[i]
            if not FindSlot(slot.id) then break end
            Data.selectedId = slot.id
            TeleportTo(SlotCF(slot), slot.name .. " (" .. i .. "/" .. #list .. ")")
            task.wait(K.TourGaps[Data.tourIdx] or 3)
            i = i + 1
            if i > #list then
                if Data.tourLoop then i = 1 else break end
            end
        end
        if ENV.ApexSaveTour == token then
            Tour.running = false
            SetStatus("Tour finished", T.TxtSec)
        end
        RefreshToggles()
        RebuildList()
    end)
end
function Tour.GoNearest()
    local root = GetRoot()
    if not root or #Data.slots == 0 then SetStatus("No saved positions", T.Yellow); return end
    local best, bd
    for _, sl in ipairs(Data.slots) do
        local d = (SlotCF(sl).Position - root.Position).Magnitude
        if d > 3 and (not bd or d < bd) then best, bd = sl, d end -- skip the spot you're standing on
    end
    if not best then SetStatus("You're already at the only position", T.Yellow); return end
    Data.selectedId = best.id
    TeleportTo(SlotCF(best), best.name .. " (nearest)")
    RebuildList()
end
-- direction (relative to your camera) and distance to the selected slot
function Tour.Compass(root)
    local slot = SelectedSlot()
    if not slot then return nil end
    local d = SlotCF(slot).Position - root.Position
    local cam = workspace.CurrentCamera
    local arrow = ""
    if cam then
        local look = cam.CFrame.LookVector
        local fwd = look.X * d.X + look.Z * d.Z
        local rgt = -look.Z * d.X + look.X * d.Z
        local idx = math.floor(math.atan2(rgt, fwd) / (math.pi / 4) + 0.5) % 8
        arrow = K.Arrows[idx + 1] .. " "
    end
    return arrow .. slot.name .. "  " .. math.floor(d.Magnitude + 0.5) .. " studs"
end

-- ── SHARE: export every position as text, import it back (or from a friend)
function Share.Export()
    local out = {}
    for _, sl in ipairs(Data.slots) do
        local fname
        if sl.folder then
            local f = FindById(Data.folders, sl.folder)
            fname = f and f.name
        end
        out[#out + 1] = {name = sl.name, c = sl.c, fav = sl.fav, icon = sl.icon, color = sl.color, folder = fname}
    end
    return "APEXPOS1:" .. HttpService:JSONEncode({slots = out})
end
function Share.Import(text)
    local t = tostring(text or "")
    t = t:gsub("^%s+", "")
    local body = t:match("^APEXPOS1:(.+)$") or t
    local ok, decoded = pcall(function() return HttpService:JSONDecode(body) end)
    if not ok or type(decoded) ~= "table" or type(decoded.slots) ~= "table" then
        return 0, "That doesn't look like exported positions"
    end
    local added = 0
    for _, sl in ipairs(decoded.slots) do
        if ValidSlot(sl) then
            local folderId
            local fn = CleanName(sl.folder, 16)
            if fn ~= "" then
                for _, f in ipairs(Data.folders) do
                    if f.name == fn then folderId = f.id end
                end
                if not folderId then
                    folderId = Data.nextFolderId
                    Data.nextFolderId = folderId + 1
                    Data.folders[#Data.folders + 1] = {id = folderId, name = fn}
                end
            end
            local id = Data.nextId
            Data.nextId = id + 1
            local pc = PALETTE[(id - 1) % #PALETTE + 1]
            local nm = CleanName(sl.name, 24)
            Data.slots[#Data.slots + 1] = {
                id = id, name = (nm ~= "") and nm or NextDefaultName(), c = {table.unpack(sl.c, 1, 12)},
                fav = (sl.fav == true), icon = FirstChars(sl.icon, 2) or ICON_PRESETS[1],
                color = ValidColor(sl.color) and {sl.color[1], sl.color[2], sl.color[3]} or {pc[1], pc[2], pc[3]},
                folder = folderId,
            }
            added = added + 1
        end
    end
    return added, nil
end

-- ── AUTO checkpoint: one rolling slot named "Auto", refreshed while you're
-- standing on something (so it never records a mid-air or void spot)
local autoLoop = task.spawn(function()
    local last = os.clock()
    while true do
        task.wait(2)
        local secs = K.Auto[Data.autoIdx] or 0
        local hum, root = GetHum(), GetRoot()
        if secs > 0 and os.clock() - last >= secs and hum and root
            and hum.Health > 0 and hum.FloorMaterial ~= Enum.Material.Air then
            last = os.clock()
            local comps = {root.CFrame:GetComponents()}
            local slot
            for _, sl in ipairs(Data.slots) do
                if sl.name == "Auto" then slot = sl; break end
            end
            if slot then
                slot.c = comps
            else
                local id = Data.nextId
                Data.nextId = id + 1
                Data.slots[#Data.slots + 1] = {id = id, name = "Auto", c = comps, fav = false,
                    icon = "A", color = {160, 170, 190}, folder = nil}
            end
            Persist()
            if not Expanded or os.clock() - LastInteract > 3 then RefreshAll() end
            SetStatus("Auto checkpoint saved", T.TxtSec, 2)
        end
    end
end)
table.insert(ENV.ApexSavePosLoops, {Disconnect = function() task.cancel(autoLoop) end})
