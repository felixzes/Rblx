-- VIEWPORT SCENE BUILDER
-- Clones nearby BaseParts into a ViewportFrame for photo rendering
-- ════════════════════════════════════════════════════════════════
local function BuildViewportScene(vp, capturedCF)
    -- remove old scene container
    for _, ch in ipairs(vp:GetChildren()) do
        if ch:IsA("Folder") then ch:Destroy() end
    end

    local origin    = capturedCF.Position
    local container = Instance.new("Folder")
    container.Name  = "_Scene"
    container.Parent = vp

    local count = 0
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if count >= CLONE_LIMIT then break end
        if obj:IsA("BasePart")
        and obj.Transparency < 0.95
        and not obj:IsDescendantOf(GetChar() or Instance.new("Model"))
        then
            local dist = (obj.Position - origin).Magnitude
            if dist <= CLONE_RADIUS then
                local ok, clone = pcall(function() return obj:Clone() end)
                if ok and clone then
                    -- strip scripts from clone
                    for _, ch in ipairs(clone:GetDescendants()) do
                        if ch:IsA("LuaSourceContainer") then ch:Destroy() end
                    end
                    clone.Anchored   = true
                    clone.CanCollide = false
                    clone.CastShadow = false
                    clone.Parent     = container
                    count            = count + 1
                end
            end
        end
    end
    return count
end

-- ════════════════════════════════════════════════════════════════
-- PHOTO SLOT MANAGEMENT
-- ════════════════════════════════════════════════════════════════
local function RefreshSlot(idx, photo)
    local s = PhotoSlots[idx]
    if not s then return end

    -- clear old click button
    if s.clickBtn then
        s.clickBtn:Destroy()
        s.clickBtn = nil
    end

    if not photo then
        s.frame.BackgroundTransparency = 0.25
        U.Stroke(s.frame, T.Border, 1, 0.6)
        s.emptyLbl.Visible = true
        -- clear viewport scene
        for _, ch in ipairs(s.vp:GetChildren()) do
            if ch:IsA("Folder") then ch:Destroy() end
        end
        return
    end

    s.emptyLbl.Visible = false
    s.frame.BackgroundTransparency = 0.05
    U.Stroke(s.frame, T.Violet, 2, 0.1)

    -- render thumbnail
    task.spawn(function()
        BuildViewportScene(s.vp, photo.capturedCF)
        s.vpCam.CFrame = photo.capturedCF
    end)

    -- click area → enter place mode
    local btn = U.Button({
        bg=Color3.new(0,0,0), tr=1,
        size=UDim2.new(1,0,1,0), z=14,
        parent=s.frame,
    })
    btn.MouseButton1Click:Connect(function()
        PlacingPhoto          = photo
        PlaceMode             = true
        PlaceHint.Visible     = true
        CameraEquipped        = false
        CrosshairRoot.Visible = false
        U.Stroke(CameraSlot, T.BorderGlow, 1.5, 0.35)
    end)
    s.clickBtn = btn
end

local function RefreshAllSlots()
    for i = 1, 4 do
        RefreshSlot(i, Photos[i])
    end
    CountLabel.Text = #Photos.." / 4 photos"
end

-- ════════════════════════════════════════════════════════════════
-- TAKE PHOTO
-- ════════════════════════════════════════════════════════════════
local function TakePhoto()
    if #Photos >= 4 then
        -- flash red to signal full
        Flash.BackgroundColor3 = T.Red
        DoFlash()
        Flash.BackgroundColor3 = Color3.new(1,1,1)
        return nil
    end

    DoFlash()

    -- shutter sound
    pcall(function()
        local snd = Instance.new("Sound")
        snd.SoundId = "rbxassetid://3200504521"
        snd.Volume  = 0.5
        snd.Parent  = Cam
        snd:Play()
        Debris:AddItem(snd, 3)
    end)

    local photo = {
        id          = NextId,
        capturedCF  = Cam.CFrame,
        portalModel = nil,
        portalFrame = nil,
        touchConn   = nil,
    }
    NextId = NextId + 1
    table.insert(Photos, photo)
    RefreshAllSlots()
    return photo
end

-- ════════════════════════════════════════════════════════════════
-- BUILD PORTAL PART
-- Creates the physical portal in Workspace at the given CFrame
-- ════════════════════════════════════════════════════════════════
local function BuildPortal(photo, placeCF)
    -- destroy old portal if re-placing
    if photo.touchConn   then photo.touchConn:Disconnect()    end
    if photo.portalModel then photo.portalModel:Destroy()     end
    photo.touchConn   = nil
    photo.portalModel = nil
    photo.portalFrame = nil

    local model   = Instance.new("Model")
    model.Name    = "ApexPortal_"..photo.id
    model.Parent  = Workspace

    -- Glass panel (the actual walkable portal surface)
    local panel = Instance.new("Part")
    panel.Name         = "PortalPanel"
    panel.Size         = Vector3.new(PORTAL_WIDTH, PORTAL_HEIGHT, PORTAL_DEPTH)
    panel.CFrame       = placeCF
    panel.Anchored     = true
    panel.CanCollide   = false
    panel.CastShadow   = false
    panel.Material     = Enum.Material.Glass
    panel.Transparency = 0.08
    panel.Color        = Color3.fromRGB(5, 5, 18)
    panel.Parent       = model

    -- Neon frame edges
    local edgeDefs = {
        { sz=Vector3.new(PORTAL_WIDTH+0.4, 0.25, 0.25), off=Vector3.new(0,  PORTAL_HEIGHT/2 + 0.1, 0) },
        { sz=Vector3.new(PORTAL_WIDTH+0.4, 0.25, 0.25), off=Vector3.new(0, -PORTAL_HEIGHT/2 - 0.1, 0) },
        { sz=Vector3.new(0.25, PORTAL_HEIGHT+0.4, 0.25), off=Vector3.new(-PORTAL_WIDTH/2 - 0.1, 0, 0) },
        { sz=Vector3.new(0.25, PORTAL_HEIGHT+0.4, 0.25), off=Vector3.new( PORTAL_WIDTH/2 + 0.1, 0, 0) },
    }
    local edgeParts = {}
    for _, ed in ipairs(edgeDefs) do
        local e = Instance.new("Part")
        e.Size        = ed.sz
        e.CFrame      = placeCF * CFrame.new(ed.off)
        e.Anchored    = true
        e.CanCollide  = false
        e.CastShadow  = false
        e.Material    = Enum.Material.Neon
        e.Color       = T.Violet
        e.Parent      = model
        table.insert(edgeParts, e)
    end

    -- Corner accent diamonds
    local cornerOffsets = {
        Vector3.new(-PORTAL_WIDTH/2, -PORTAL_HEIGHT/2, 0),
        Vector3.new( PORTAL_WIDTH/2, -PORTAL_HEIGHT/2, 0),
        Vector3.new(-PORTAL_WIDTH/2,  PORTAL_HEIGHT/2, 0),
        Vector3.new( PORTAL_WIDTH/2,  PORTAL_HEIGHT/2, 0),
    }
    for _, coff in ipairs(cornerOffsets) do
        local d = Instance.new("Part")
        d.Size       = Vector3.new(0.5, 0.5, 0.5)
        d.CFrame     = placeCF * CFrame.new(coff)
        d.Anchored   = true
        d.CanCollide = false
        d.CastShadow = false
        d.Material   = Enum.Material.Neon
        d.Color      = T.Cyan
        d.Shape      = Enum.PartType.Ball
        d.Parent     = model
    end

    -- SurfaceGui with ViewportFrame on front face
    local sg = Instance.new("SurfaceGui")
    sg.Adornee        = panel
    sg.Face           = Enum.NormalId.Front
    sg.SizingMode     = Enum.SurfaceGuiSizingMode.PixelsPerStud
    sg.PixelsPerStud  = 60
    sg.AlwaysOnTop    = false
    sg.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    sg.Parent         = panel

    local vp = Instance.new("ViewportFrame")
    vp.BackgroundColor3       = Color3.new(0,0,0)
    vp.BackgroundTransparency = 0
    vp.Size                   = UDim2.new(1,0,1,-20)
    vp.Ambient                = Color3.new(0.7, 0.7, 0.7)
    vp.LightDirection         = Vector3.new(-1,-2,-1)
    vp.Parent                 = sg

    local vpCam = Instance.new("Camera")
    vpCam.CFrame     = photo.capturedCF
    vpCam.FieldOfView = 70
    vpCam.Parent     = vp
    vp.CurrentCamera = vpCam

    -- Photo info strip at bottom of portal
    local strip = Instance.new("Frame")
    strip.BackgroundColor3       = T.GlassMid
    strip.BackgroundTransparency = 0.25
    strip.BorderSizePixel        = 0
    strip.Size                   = UDim2.new(1,0,0,20)
    strip.Position               = UDim2.new(0,0,1,-20)
    strip.ZIndex                 = 2
    strip.Parent                 = sg

    local stripLbl = Instance.new("TextLabel")
    stripLbl.BackgroundTransparency = 1
    stripLbl.Text      = "PHOTO #"..photo.id.."   WALK THROUGH TO PORTAL"
    stripLbl.TextColor3 = T.Cyan
    stripLbl.Font      = Enum.Font.GothamBold
    stripLbl.TextSize  = 9
    stripLbl.Size      = UDim2.new(1,0,1,0)
    stripLbl.ZIndex    = 3
    stripLbl.Parent    = strip

    -- Clone scene asynchronously
    task.spawn(function()
        local count = BuildViewportScene(vp, photo.capturedCF)
        vpCam.CFrame    = photo.capturedCF
        stripLbl.Text   = "PHOTO #"..photo.id.."  |  "..count.." parts  |  WALK THROUGH"
    end)

    -- Rainbow neon pulse on edges
    task.spawn(function()
        local hue = 0
        while model and model.Parent do
            hue = (hue + 0.007) % 1
            local col = Color3.fromHSV(hue, 0.75, 1)
            for _, ep in ipairs(edgeParts) do
                pcall(function() ep.Color = col end)
            end
            task.wait(0.05)
        end
    end)

    photo.portalModel = model
    photo.portalFrame = panel

    -- ── TOUCH PORTAL LOGIC
    -- Debounce table per player
    local debounce = {}

    photo.touchConn = panel.Touched:Connect(function(hit)
        local char = GetChar()
        if not char then return end
        if not hit:IsDescendantOf(char) then return end

        -- per-character debounce
        local userId = tostring(LP.UserId)
        if debounce[userId] then return end
        debounce[userId] = true

        local root = GetRoot()
        if not root then
            debounce[userId] = nil
            return
        end

        DoFade(function()
            local r = GetRoot()
            if not r then return end
            -- place player slightly behind the captured camera position
            -- so they appear at that spot facing the same direction
            local targetCF = photo.capturedCF
            local spawnPos = targetCF.Position
                + targetCF.LookVector * (-3.5)
                + Vector3.new(0, -3, 0)
            r.CFrame = CFrame.new(spawnPos)
                * CFrame.Angles(0, math.atan2(-targetCF.LookVector.X, -targetCF.LookVector.Z), 0)
        end)

        task.delay(1.5, function()
            debounce[userId] = nil
        end)
    end)

    return model
end

-- ════════════════════════════════════════════════════════════════
-- DELETE PHOTO
-- ════════════════════════════════════════════════════════════════
local function DeletePhoto(idx)
    local photo = Photos[idx]
    if not photo then return end
    if photo.touchConn   then photo.touchConn:Disconnect()   end
    if photo.portalModel then photo.portalModel:Destroy()    end
    table.remove(Photos, idx)
    RefreshAllSlots()
end

-- ════════════════════════════════════════════════════════════════
-- PLACEMENT PREVIEW
-- A ghost portal that follows the camera raycast
-- ════════════════════════════════════════════════════════════════
local PreviewModel = nil

local function GetPlaceCF()
    local vpSize = Cam.ViewportSize
    local ray    = Cam:ScreenPointToRay(vpSize.X/2, vpSize.Y/2)

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local char = GetChar()
    if char then
        params.FilterDescendantsInstances = {char}
    end

    -- also exclude existing portals
    local excludeList = char and {char} or {}
    for _, photo in ipairs(Photos) do
        if photo.portalModel then
            table.insert(excludeList, photo.portalModel)
        end
    end
    params.FilterDescendantsInstances = excludeList

    local result = Workspace:Raycast(ray.Origin, ray.Direction * PLACE_DIST, params)

    local hitPos, hitNorm
    if result then
        hitPos  = result.Position
        hitNorm = result.Normal
    else
        hitPos  = ray.Origin + ray.Direction * (PLACE_DIST * 0.6)
        hitNorm = -ray.Direction
    end

    -- build upright CFrame oriented to hit normal
    local up    = Vector3.new(0,1,0)
    local right = hitNorm:Cross(up)
    if right.Magnitude < 0.001 then
        right = hitNorm:Cross(Vector3.new(1,0,0))
    end
    right = right.Unit
    local trueUp = right:Cross(hitNorm).Unit

    local baseCF = CFrame.fromMatrix(hitPos, right, trueUp, -hitNorm)
        * CFrame.new(0, PORTAL_HEIGHT/2 + 0.1, 0)
        * CFrame.Angles(0, math.rad(PlaceRotation), 0)

    return baseCF
end

local function UpdatePreview()
    local cf = GetPlaceCF()

    if not PreviewModel then
        PreviewModel      = Instance.new("Model")
        PreviewModel.Name = "ApexPortalPreview"
        PreviewModel.Parent = Workspace

        local pp = Instance.new("Part")
        pp.Name        = "PreviewPanel"
        pp.Size        = Vector3.new(PORTAL_WIDTH, PORTAL_HEIGHT, PORTAL_DEPTH)
        pp.Anchored    = true
        pp.CanCollide  = false
        pp.CastShadow  = false
        pp.Material    = Enum.Material.Glass
        pp.Transparency = 0.55
        pp.Color       = T.Violet
        pp.Parent      = PreviewModel

        local edgeDefs2 = {
            { sz=Vector3.new(PORTAL_WIDTH+0.4,0.2,0.2),  off=Vector3.new(0,  PORTAL_HEIGHT/2+0.1, 0) },
            { sz=Vector3.new(PORTAL_WIDTH+0.4,0.2,0.2),  off=Vector3.new(0, -PORTAL_HEIGHT/2-0.1, 0) },
            { sz=Vector3.new(0.2,PORTAL_HEIGHT+0.4,0.2), off=Vector3.new(-PORTAL_WIDTH/2-0.1,0, 0)   },
            { sz=Vector3.new(0.2,PORTAL_HEIGHT+0.4,0.2), off=Vector3.new( PORTAL_WIDTH/2+0.1,0, 0)   },
        }
        for _, ed in ipairs(edgeDefs2) do
            local e = Instance.new("Part")
            e.Size        = ed.sz
            e.Anchored    = true
            e.CanCollide  = false
            e.CastShadow  = false
            e.Material    = Enum.Material.Neon
            e.Color       = T.VioletBr
            e.Transparency = 0.3
            e.Parent      = PreviewModel
        end
    end

    local parts = PreviewModel:GetChildren()
    if parts[1] then parts[1].CFrame = cf end

    local offsets = {
        Vector3.new(0,  PORTAL_HEIGHT/2+0.1, 0),
        Vector3.new(0, -PORTAL_HEIGHT/2-0.1, 0),
        Vector3.new(-PORTAL_WIDTH/2-0.1, 0, 0),
        Vector3.new( PORTAL_WIDTH/2+0.1, 0, 0),
    }
    for i, off in ipairs(offsets) do
        if parts[i+1] then
            parts[i+1].CFrame = cf * CFrame.new(off)
        end
    end
end

local function DestroyPreview()
    if PreviewModel then
        PreviewModel:Destroy()
        PreviewModel = nil
    end
end

-- ════════════════════════════════════════════════════════════════
-- PLACE PHOTO
-- ════════════════════════════════════════════════════════════════
local function PlacePhoto(photo)
    local cf = GetPlaceCF()
    DestroyPreview()
    BuildPortal(photo, cf)

    PlaceMode             = false
    PlacingPhoto          = nil
    PlaceRotation         = 0
    PlaceHint.Visible     = false
end

-- ════════════════════════════════════════════════════════════════
-- INPUT HANDLING
-- ════════════════════════════════════════════════════════════════
CameraSlot.MouseButton1Click:Connect(function()
    CameraEquipped        = not CameraEquipped
    CrosshairRoot.Visible = CameraEquipped

    if CameraEquipped then
        PlaceMode         = false
        PlacingPhoto      = nil
        PlaceHint.Visible = false
        DestroyPreview()
        U.Stroke(CameraSlot, T.Cyan, 2, 0)
    else
        U.Stroke(CameraSlot, T.BorderGlow, 1.5, 0.35)
    end
end)

UserInputService.InputBegan:Connect(function(inp, gpe)
    if gpe then return end

    -- ESC: cancel everything
    if inp.KeyCode == Enum.KeyCode.Escape then
        if PlaceMode then
            PlaceMode         = false
            PlacingPhoto      = nil
            PlaceHint.Visible = false
            PlaceRotation     = 0
            DestroyPreview()
        elseif CameraEquipped then
            CameraEquipped        = false
            CrosshairRoot.Visible = false
            U.Stroke(CameraSlot, T.BorderGlow, 1.5, 0.35)
        end
        return
    end

    -- R: rotate portal preview
    if inp.KeyCode == Enum.KeyCode.R and PlaceMode then
        PlaceRotation = (PlaceRotation + 45) % 360
        return
    end

    -- G: delete last portal
    if inp.KeyCode == Enum.KeyCode.G then
        if #Photos > 0 then
            DeletePhoto(#Photos)
        end
        return
    end

    -- Number keys 1-4: select photo slot → place mode
    local numMap = {
        [Enum.KeyCode.One]=1, [Enum.KeyCode.Two]=2,
        [Enum.KeyCode.Three]=3, [Enum.KeyCode.Four]=4,
    }
    local idx = numMap[inp.KeyCode]
    if idx then
        if Photos[idx] then
            PlacingPhoto          = Photos[idx]
            PlaceMode             = true
            PlaceHint.Visible     = true
            CameraEquipped        = false
            CrosshairRoot.Visible = false
            U.Stroke(CameraSlot, T.BorderGlow, 1.5, 0.35)
        end
        return
    end

    -- LMB
    if inp.UserInputType == Enum.UserInputType.MouseButton1 then
        if PlaceMode and PlacingPhoto then
            PlacePhoto(PlacingPhoto)
            return
        end
        if CameraEquipped then
            TakePhoto()
            return
        end
    end
end)

        DestroyPreview()

end)
table.insert(ENV.ApexPhotoLoops, renderConn)
