╠══════════════════════════════════════════════════════════════╣
║   One small panel replaces every old on-screen widget (the big   ║
║   tablet bezel, count/status pills, hint box, loose buttons).     ║
║                                                                     ║
║   • Drag it by the title bar. Drag position is remembered.          ║

║     fades after a few more seconds so it stays out of the way.           ║
║   • It never parks over the screen center (crosshair zone) and           ║
║     is clamped to the screen edges.                                       ║
║   • Camera controls (pan/tilt, flashlight, exit) live in the same         ║
║     small panel while you're viewing a camera. Hold a pan/tilt             ║
║     button to keep panning.                                                 ║
║   • PC keys: P toggle panel, arrows pan/aim, Esc exit/cancel,                ║
║     G delete last, H show inventory.                                          ║
║                                                                                 ║
║   Also fixed: pan/tilt directions were inverted in earlier versions              ║
║   (left button turned right, up looked down). And UIGradient tints                ║
║   button TEXT as well as the background, so the dark gradients added               ║
║   in v5 were likely dimming labels — gradients are gone.                            ║
║                                                                                       ║
║   The camera model (BuildCCTVGeometry) and the tablet handle are                       ║
║   copied over unchanged.                                                                ║

-- A small circular indicator light that blinks via Transparency.
function U.PulseDot(parent, col, size, pos)
    local dot = Instance.new("Frame")
    dot.BackgroundColor3 = col
    dot.BorderSizePixel = 0
    dot.Size = UDim2.new(0, size or 10, 0, size or 10)
    dot.Position = pos or UDim2.new(0, 0, 0, 0)
    dot.ZIndex = 20
    U.Corner(dot, UDim.new(1, 0))
    dot.Parent = parent
    local alive = true
    task.spawn(function()
        while alive and dot.Parent do
            dot.BackgroundTransparency = 0
            task.wait(0.5)
            if not (alive and dot.Parent) then break end
            dot.BackgroundTransparency = 0.75
            task.wait(0.5)
        end
    end)
    return dot, function() alive = false end
end

-- ── Opacity: every translucent background registers here so one
-- slider controls them all. Text and outlines stay fully opaque.
local GlassItems = {}
local function Glass(inst, minTr, maxTr)
    GlassItems[#GlassItems + 1] = {inst = inst, minTr = minTr, maxTr = maxTr}
    inst.BackgroundTransparency = minTr + (maxTr - minTr) * (1 - Opacity)
    return inst
end
local function ApplyOpacity()
    local alive = {}
    for _, g in ipairs(GlassItems) do
        if g.inst.Parent then
            g.inst.BackgroundTransparency = g.minTr + (g.maxTr - g.minTr) * (1 - Opacity)
            alive[#alive + 1] = g
        end
    end
    GlassItems = alive
end

-- Press-and-hold repeat: one step immediately, then a smaller step
-- repeated while the button stays pressed.
local function MakeRepeatButton(btn, tapFn, holdFn)
    local pressId = 0
    local active = false
    btn.InputBegan:Connect(function(inp)
        local ut = inp.UserInputType
        if ut ~= Enum.UserInputType.MouseButton1 and ut ~= Enum.UserInputType.Touch then return end
        pressId = pressId + 1
        local myId = pressId
        active = true
        Bump()
        tapFn()
        inp.Changed:Connect(function()
            if inp.UserInputState == Enum.UserInputState.End and pressId == myId then
                active = false
            end
        end)
        task.spawn(function()
            task.wait(0.35)
            while active and pressId == myId do
                holdFn()
                Bump()
                task.wait(0.04)
            end
        end)
    end)
end

-- ── Position handling: clamp to screen and keep clear of the center
local function ActiveFrameSize()
    if PanelExpanded then
        local s = Panel.AbsoluteSize
        if s.X < 10 or s.Y < 10 then return Vector2.new(PANEL_W, 170) end
        return s
    end
    return Vector2.new(ICON_SIZE, ICON_SIZE)
end

local function ClampToScreen(pos, size)
    local ss = ScreenSize()
    return Vector2.new(
        math.clamp(pos.X, 4, math.max(4, ss.X - size.X - 4)),
        math.clamp(pos.Y, 4, math.max(4, ss.Y - size.Y - 4)))
end

-- The middle of the screen is where you aim; never park over it.
local function AvoidCenter(pos, size)
    local ss = ScreenSize()
    local cx, cy = ss.X / 2, ss.Y / 2
    local hw, hh = 48, 48
    if pos.X + size.X > cx - hw and pos.X < cx + hw
        and pos.Y + size.Y > cy - hh and pos.Y < cy + hh then
        local leftX = (cx - hw) - size.X
        local rightX = cx + hw
        if math.abs(pos.X - leftX) < math.abs(pos.X - rightX) then
            return Vector2.new(leftX, pos.Y)
        end
        return Vector2.new(rightX, pos.Y)
    end
    return pos
end

local function ApplyPos()
    local size = ActiveFrameSize()
    local p = ClampToScreen(PanelPos, size)
    p = AvoidCenter(p, size)
    p = ClampToScreen(p, size)
    p = Vector2.new(math.floor(p.X + 0.5), math.floor(p.Y + 0.5))
    PanelPos = p
    Panel.Position = UDim2.fromOffset(p.X, p.Y)
    Icon.Position = UDim2.fromOffset(p.X, p.Y)
    Saved.x, Saved.y = p.X, p.Y
end
Track(Panel:GetPropertyChangedSignal("AbsoluteSize"):Connect(ApplyPos))

-- ── Dragging (title bar and icon). A tap that barely moves counts as a tap.
local function MakeDraggable(handle, onTap)
    local dragging, moved = false, false
    local startInput, startMouse, startPos
    handle.InputBegan:Connect(function(inp)
        local ut = inp.UserInputType
        if ut ~= Enum.UserInputType.MouseButton1 and ut ~= Enum.UserInputType.Touch then return end
        dragging, moved = true, false
        DraggingNow = true
        startInput = inp
        startMouse = Vector2.new(inp.Position.X, inp.Position.Y)
        startPos = PanelPos
        Bump()
        inp.Changed:Connect(function()
            if inp.UserInputState == Enum.UserInputState.End and startInput == inp and dragging then
                dragging = false
                DraggingNow = false
                if not moved and onTap then onTap() end
            end
        end)
    end)
    Track(UserInputService.InputChanged:Connect(function(inp)
        if not dragging then return end
        if inp.UserInputType == Enum.UserInputType.MouseMovement or inp == startInput then
            local delta = Vector2.new(inp.Position.X, inp.Position.Y) - startMouse
            if not moved and delta.Magnitude > 6 then moved = true end
            if moved then
                PanelPos = startPos + delta
                ApplyPos()
                Bump()
            end
        end
    end))
end

-- ── Opacity slider
local SliderT = (Opacity - MIN_OPACITY) / (1 - MIN_OPACITY)
local function SetSliderT(t)
    SliderT = math.clamp(t, 0, 1)
    Opacity = MIN_OPACITY + SliderT * (1 - MIN_OPACITY)
    Saved.opacity = Opacity
    SliderFill.Size = UDim2.new(SliderT, 0, 1, 0)
    SliderKnob.Position = UDim2.new(SliderT, -8, 0.5, -8)
    OpacityLabel.Text = "OPACITY  " .. math.floor(Opacity * 100 + 0.5) .. "%"
    ApplyOpacity()
end
local function SliderFromX(x)
    local w = SliderHit.AbsoluteSize.X
    if w <= 0 then return end
    SetSliderT((x - SliderHit.AbsolutePosition.X) / w)
end
SetSliderT(SliderT)

    end

local function RebuildChips()

    for i, camData in ipairs(Cameras) do
        local selected = (ActivePOVCam == camData)
        local b, st = MakeBtn({
            text = tostring(camData.id),
            col = selected and T.Cyan or (usable and T.TxtPri or T.TxtDim),

                EnterCameraPOV(camData)

-- ── Auto-minimize: collapse to the icon when idle, then fade the icon.
-- Never while aiming a mount; the dragging/slider guards stop it
-- collapsing under your finger.
local AUTO_COLLAPSE = {idle = 8, tablet = 25, place = 14, pov = 7}
local ICON_DIM_AFTER = 6
local uiLoop = task.spawn(function()
    while true do
        task.wait(0.5)
        local idleFor = os.clock() - LastInteract
        local mode = CurrentMode()
        -- Safety net: if a "finger lifted" event was ever missed, don't
        -- let a stuck flag block auto-minimize forever.
        if idleFor > 10 then DraggingNow = false; SliderActive = false end
        if PanelExpanded then
            local limit = AUTO_COLLAPSE[mode]
            if limit and idleFor > limit and not DraggingNow and not SliderActive then
                Collapse()
            end
        elseif not IconDim and mode ~= "pov" and idleFor > ICON_DIM_AFTER then
            SetIconDim(true)
        end
    end
end)
table.insert(ENV.ApexSurvLoops, {Disconnect = function() task.cancel(uiLoop) end})

-- RAYCAST HELPERS
-- ════════════════════════════════════════════════════════════════
local function IsDescendantOfAny(obj, list)
    for _, m in ipairs(list) do
        if m and obj:IsDescendantOf(m) then return true end
    end
    return false
end

local function BuildExcludeList()
    local excludeList = {}
    local char = GetChar()
    if char then table.insert(excludeList, char) end
    for _, cam in ipairs(Cameras) do
        if cam.model then table.insert(excludeList, cam.model) end
    end
    if PreviewModel then table.insert(excludeList, PreviewModel) end
    return excludeList
end

local function RaycastFromCamera(maxDist)
    local camera = GetCamera()
    local vpSize = camera.ViewportSize
    local ray = camera:ScreenPointToRay(vpSize.X / 2, vpSize.Y / 2)

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = BuildExcludeList()
    params.IgnoreWater = true

    local result = Workspace:Raycast(ray.Origin, ray.Direction * maxDist, params)

    print(("[ApexSurveillance] Raycast  origin=%s  dir=%s  hit=%s"):format(
        tostring(ray.Origin), tostring(ray.Direction), result and tostring(result.Position) or "NONE"))

    return result, ray, camera
end

local function CrosshairSilentRaycast()
    local camera = GetCamera()
    local vpSize = camera.ViewportSize
    local ray = camera:ScreenPointToRay(vpSize.X / 2, vpSize.Y / 2)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = BuildExcludeList()
    params.IgnoreWater = true
    return Workspace:Raycast(ray.Origin, ray.Direction * RAY_MAX_DIST, params), camera
end

-- Crosshair turns green over a mountable surface, red otherwise. The
-- distance readout lives in the panel's mode line, not on screen.
local function UpdateCrosshairFeedback()
    if not CrosshairRoot.Visible then return end
    local hit, camera = CrosshairSilentRaycast()
    local col = hit and T.Green or T.Red
    for _, p in ipairs(CrosshairPieces) do p.BackgroundColor3 = col end

    local mode = CurrentMode()
    if mode == "place" or mode == "preview" then
        local head = (mode == "place") and "PLACER" or "AIMING"
        local txt
        if hit then
            local d = (hit.Position - camera.CFrame.Position).Magnitude
            txt = head .. "  ·  " .. string.format("%.0f studs", d)
            if mode == "preview" then txt = txt .. "  ·  tap fire" end
        else
            txt = head .. "  ·  no surface"
        end
        if ModeText.Text ~= txt then ModeText.Text = txt end
        ModeText.TextColor3 = col
    end
end

-- CCTV CAMERA MODEL (unchanged from v5)
-- ════════════════════════════════════════════════════════════════
local function BuildCCTVGeometry(mode, container, baseCFrame)
    local bodyColor  = Color3.fromRGB(42, 46, 54)
    local bodyDark   = Color3.fromRGB(22, 24, 30)
    local metalColor = Color3.fromRGB(152, 157, 167)
    local darkMetal  = Color3.fromRGB(32, 34, 40)
    local lensGlass  = Color3.fromRGB(8, 20, 35)
    local ledRed     = Color3.fromRGB(255, 40, 40)
    local ledIR      = Color3.fromRGB(95, 22, 22)

    baseCFrame = baseCFrame or CFrame.new(0,0,0)
    local parts = {}

    local function NewPart(name, kind, size, localCF, material, color, extraProps)
        local p
        if kind == "cylinder" or kind == "box" then
            p = Instance.new("Part")
            if kind == "cylinder" then p.Shape = Enum.PartType.Cylinder end
        elseif kind == "ball" then
            p = Instance.new("Part")
            p.Shape = Enum.PartType.Ball
        end
        p.Name = name
        p.Size = size
        p.Anchored = (mode == "fixture")
        p.CanCollide = false
        p.CastShadow = true
        p.Material = material
        p.Color = color
        if extraProps then
            for k, v in pairs(extraProps) do p[k] = v end
        end

        if mode == "fixture" then
            p.CFrame = baseCFrame * localCF
            p.Parent = container
        else
            p.CFrame = localCF
            p.Parent = container
        end

        table.insert(parts, p)
        return p
    end

    -- Root wall plate (also the Tool's required "Handle" in tool mode).
    -- Z is the thin dimension — the plate sits flush against the wall,
    -- with everything else extending forward along +Z from here.
    local root = NewPart(mode=="tool" and "Handle" or "MountPlate", "box",
        Vector3.new(0.9, 1.1, 0.18), CFrame.new(0,0,0), Enum.Material.Metal, darkMetal)

    -- corner rivets (on the forward-facing side of the plate)
    for _, off in ipairs({
        Vector3.new(0.35, 0.45, 0.1), Vector3.new(-0.35, 0.45, 0.1),
        Vector3.new(0.35, -0.45, 0.1), Vector3.new(-0.35, -0.45, 0.1),
    }) do
        NewPart("Rivet", "ball", Vector3.new(0.08,0.08,0.08), CFrame.new(off), Enum.Material.Metal, metalColor)
    end

    -- swivel joints + angled mount arm (reference: articulated bracket).
    -- The Arm cylinder's own Size.X is its length regardless of which
    -- world axis it's rotated to point along — rotating it ~90° around
    -- Y converts its local length axis to point along Z (forward),
    -- with a small extra tilt for a slightly bent, less robotic look.
    NewPart("Joint1", "ball", Vector3.new(0.34,0.34,0.34),
        CFrame.new(0, 0, 0.25), Enum.Material.Metal, metalColor)
    NewPart("Arm", "cylinder", Vector3.new(1.1, 0.2, 0.2),
        CFrame.new(0, -0.05, 0.75) * CFrame.Angles(math.rad(10), math.rad(90), 0),
        Enum.Material.Metal, metalColor)
    NewPart("Joint2", "ball", Vector3.new(0.3,0.3,0.3),
        CFrame.new(0, -0.1, 1.35), Enum.Material.Metal, metalColor)

    -- bullet housing — long dimension now along Z
    NewPart("Housing", "box", Vector3.new(0.68, 0.68, 1.9),
        CFrame.new(CCTV_HOUSING_OFFSET), Enum.Material.SmoothPlastic, bodyColor)
    NewPart("HousingPanel", "box", Vector3.new(0.55, 0.55, 0.5),
        CFrame.new(CCTV_HOUSING_OFFSET + Vector3.new(0,0,0.85)), Enum.Material.SmoothPlastic, bodyDark)

    -- lens — cylinders rotated to point along Z, positioned along Z;
    -- these now actually agree with each other
    NewPart("LensRing", "cylinder", Vector3.new(0.18, 0.5, 0.5),
        CFrame.new(CCTV_HOUSING_OFFSET + Vector3.new(0,0,1.15)) * CFrame.Angles(0, math.rad(90), 0),
        Enum.Material.Metal, darkMetal)
    NewPart("LensGlass", "cylinder", Vector3.new(0.08, 0.4, 0.4),
        CFrame.new(CCTV_HOUSING_OFFSET + Vector3.new(0,0,1.28)) * CFrame.Angles(0, math.rad(90), 0),
        Enum.Material.Glass, lensGlass, {Transparency=0.2})

    -- IR LED ring around the lens — ring now sits in the XY plane at a
    -- fixed Z depth, perpendicular to the forward axis as intended
    for i = 0, 5 do
        local ang = math.rad(i * 60)
        local off = Vector3.new(math.sin(ang)*0.22, math.cos(ang)*0.22, 1.2)
        NewPart("IRLed", "ball", Vector3.new(0.06,0.06,0.06),
            CFrame.new(CCTV_HOUSING_OFFSET + off), Enum.Material.Neon, ledIR)
    end

    -- status/recording LED (caller animates its Transparency to blink)
    local statusLed = NewPart("StatusLED", "ball", Vector3.new(0.1,0.1,0.1),
        CFrame.new(CCTV_HOUSING_OFFSET + Vector3.new(0, 0.3, 0.3)), Enum.Material.Neon, ledRed)

    if mode == "tool" then
        for _, p in ipairs(parts) do
            if p ~= root then WeldTo(root, p) end
        end
    end

    return root, parts, statusLed
end

local function StartLedBlink(led, getAlive)
    task.spawn(function()
        while getAlive() do
            led.Transparency = 0
            task.wait(0.6)
            if not getAlive() then break end
            led.Transparency = 0.7
            task.wait(0.6)
        end
    end)
end

-- ════════════════════════════════════════════════════════════════
local function GetCameraPlaceCF()
    local result, ray, camera = RaycastFromCamera(RAY_MAX_DIST)
    local hitPos, hitNorm
    if result then
        hitPos = result.Position
        hitNorm = result.Normal
    else
        hitPos = ray.Origin + ray.Direction * (RAY_MAX_DIST * 0.05)
        hitNorm = -ray.Direction
        warn("[ApexSurveillance] Placement raycast missed — using fallback point in front of camera.")
    end

    -- Mount flush to the surface, lens pointing outward along its
    -- normal (walls, ceilings, or a floor-standing mount), then let
    -- the player fine-tune with yaw/pitch.
    local baseCF = CFrame.new(hitPos, hitPos + hitNorm)
    local adjustedCF = baseCF * CFrame.Angles(math.rad(PlacePitch), math.rad(PlaceYaw), 0)
    return adjustedCF, result ~= nil
end

        lens.Size = Vector3.new(0.2, 0.5, 0.5)
        lens.Anchored = true; lens.CanCollide = false; lens.CastShadow = false
        lens.Material = Enum.Material.Neon; lens.Transparency = 0.3; lens.Color = T.Cyan
        lens.Parent = PreviewModel
    end

    local parts = PreviewModel:GetChildren()
    local box, lens = parts[1], parts[2]
    if box then box.CFrame = cf * CFrame.new(CCTV_HOUSING_OFFSET) end
    if lens then
        lens.CFrame = cf * CFrame.new(CCTV_HOUSING_OFFSET + Vector3.new(0, 0, 1.15)) * CFrame.Angles(0, math.rad(90), 0)
    end
end

local function DestroyPreview()
    if PreviewModel then PreviewModel:Destroy(); PreviewModel = nil end
end

local function BuildCameraFixture(camData, placeCF)
    if camData.model then camData.model:Destroy() end
    local model = Instance.new("Model")
    model.Name = "ApexCam_" .. camData.id
    model.Parent = Workspace

    local root, parts, led = BuildCCTVGeometry("fixture", model, placeCF)
    camData.model = model
    camData.mountCFrame = placeCF
    camData.root = root

    StartLedBlink(led, function() return model and model.Parent ~= nil end)

    print(("[ApexSurveillance] Camera #%d mounted at %s facing %s")
        :format(camData.id, tostring(placeCF.Position), tostring(placeCF.LookVector)))
end

local function ConfirmCameraPlacement()
    local cf = GetCameraPlaceCF()
    DestroyPreview()

    local camData = {
        id = NextCamId,
        model = nil,
        mountCFrame = nil,
        root = nil,
        povYaw = 0,
        povPitch = 0,
    }
    NextCamId = NextCamId + 1
    table.insert(Cameras, camData)

    BuildCameraFixture(camData, cf)

    PlaceYaw = 0
    PlacePitch = 0
    SetStatus("Camera #" .. camData.id .. " mounted", T.Green)
end

CancelPreview = function()
    PlacingPreview = false
    PlaceYaw = 0
    PlacePitch = 0
    DestroyPreview()
    RefreshUI()
end

DeleteLastCamera = function()
    local camData = table.remove(Cameras)
    if not camData then return end
    if ActivePOVCam == camData then ExitCameraPOV() end
    if camData.model then camData.model:Destroy() end
    SetStatus("Camera #" .. camData.id .. " removed", T.TxtSec)
    RefreshUI()
end

-- ════════════════════════════════════════════════════════════════
-- PAN / TILT (shared by camera view and mount aiming)
-- Positive pitch looks UP and positive yaw turns LEFT (CFrame.Angles
-- convention) — earlier versions had both inverted.
-- ════════════════════════════════════════════════════════════════
PadAction = function(dir, hold)
    if ActivePOVCam then
        local c = ActivePOVCam
        local ps = hold and POV_PAN_HOLD or POV_PAN_TAP
        local ts = hold and POV_TILT_HOLD or POV_TILT_TAP
        if dir == "left" then c.povYaw = (c.povYaw + ps) % 360
        elseif dir == "right" then c.povYaw = (c.povYaw - ps) % 360
        elseif dir == "up" then c.povPitch = math.clamp(c.povPitch + ts, -80, 80)
        elseif dir == "down" then c.povPitch = math.clamp(c.povPitch - ts, -80, 80)
        end
    elseif PlacingPreview then
        local ys = hold and 3 or 15
        local ts2 = hold and 2 or 10
        if dir == "left" then PlaceYaw = (PlaceYaw + ys) % 360
        elseif dir == "right" then PlaceYaw = (PlaceYaw - ys) % 360
        elseif dir == "up" then PlacePitch = math.clamp(PlacePitch + ts2, -85, 85)
        elseif dir == "down" then PlacePitch = math.clamp(PlacePitch - ts2, -85, 85)
        end
    end
end

-- ════════════════════════════════════════════════════════════════
-- CAMERA POV — moves the REAL player camera to the CCTV's lens, so
-- the view is exactly as good as normal gameplay rendering.
-- ════════════════════════════════════════════════════════════════
local function ComputePovCFrame(camData)
    local lensCF = camData.mountCFrame * CFrame.new(CCTV_LENS_TIP_LOCAL)
    return lensCF * CFrame.Angles(math.rad(camData.povPitch), math.rad(camData.povYaw), 0)
end

-- A Scriptable camera has no light of its own, so dark areas stay dark.
-- This is a real SpotLight on an invisible part that tracks the POV
-- CFrame every frame (see the render loop below).
local FlashlightOn   = false
local FlashlightPart = nil
local FlashlightLamp = nil

local function EnsureFlashlight()
    if FlashlightPart then return end
    local p = Instance.new("Part")
    p.Name = "ApexFlashlight"
    p.Size = Vector3.new(0.2, 0.2, 0.2)
    p.Transparency = 1
    p.CanCollide = false
    p.Anchored = true
    p.CastShadow = false
    p.Parent = Workspace

    local lamp = Instance.new("SpotLight")
    lamp.Face = Enum.NormalId.Front -- local -Z, same direction as CFrame.LookVector
    lamp.Brightness = 3.5
    lamp.Range = 60
    lamp.Angle = 55
    lamp.Color = Color3.fromRGB(255, 244, 214)
    lamp.Enabled = false
    lamp.Parent = p

    FlashlightPart = p
    FlashlightLamp = lamp
end

local function SetFlashlightVisual(on)
    FlashBtn.TextColor3 = on and T.Yellow or T.TxtSec
    FlashStroke.Color = on and T.Yellow or T.BorderGlow
    FlashStroke.Transparency = on and 0.1 or 0.5
    FlashDot.BackgroundColor3 = on and T.Yellow or Color3.fromRGB(70, 64, 36)
end
SetFlashlightVisual(false)

ToggleFlashlight = function()
    EnsureFlashlight()
    FlashlightOn = not FlashlightOn
    FlashlightLamp.Enabled = FlashlightOn
    SetFlashlightVisual(FlashlightOn)
end

EnterCameraPOV = function(camData)
    local camera = GetCamera()

    if not ActivePOVCam then
        -- first entry this session: remember what to restore on exit
        SavedCameraType = camera.CameraType
        SavedFOV = camera.FieldOfView
        local hum = GetHum()
        if hum then
            SavedWalkSpeed = hum.WalkSpeed
            hum.WalkSpeed = 0 -- no walking blind while watching a feed
        end
    end

    ActivePOVCam = camData
    camera.CameraType = Enum.CameraType.Scriptable
    camera.FieldOfView = 70
    camera.CFrame = ComputePovCFrame(camData)

    RefreshUI()
    print("[ApexSurveillance] Entered POV of camera #" .. camData.id)
end

ExitCameraPOV = function()
    if not ActivePOVCam then return end
    local camera = GetCamera()
    local exitedId = ActivePOVCam.id

    -- Turn the flashlight off so it doesn't keep burning in the world.
    if FlashlightOn then
        FlashlightOn = false
        if FlashlightLamp then FlashlightLamp.Enabled = false end
        SetFlashlightVisual(false)
    end

    camera.CameraType = SavedCameraType or Enum.CameraType.Custom
    if SavedFOV then camera.FieldOfView = SavedFOV end

    local hum = GetHum()
    if hum and SavedWalkSpeed then hum.WalkSpeed = SavedWalkSpeed end
    SavedWalkSpeed = nil

    ActivePOVCam = nil
    RefreshUI()
    print("[ApexSurveillance] Exited POV of camera #" .. exitedId)
end

local function BindToolToBackpack(tool, attempt)
    attempt = attempt or 1
    local bp = GetBackpack()
    tool.Parent = bp
    task.wait(0.1)
    if tool.Parent == bp then
        SetStatus("Inventory OK (" .. #bp:GetChildren() .. " items)", T.Green)
        return true
    end
    if attempt < 5 then
        task.wait(0.3)
        return BindToolToBackpack(tool, attempt + 1)
    end
    SetStatus("Backpack bind failed — tap INVENTORY", T.Red)
    warn("[ApexSurveillance] Failed to parent tool to Backpack after 5 attempts: " .. tool.Name)
    return false
end

local PlacerTool, TabletTool

local function CreateCameraPlacerTool()
    local tool = Instance.new("Tool")
    tool.Name = "Apex Camera Placer"
    tool.RequiresHandle = true
    tool.CanBeDropped = false
    tool.ToolTip = "Click to preview a mounting spot, click again to mount a camera there"
    table.insert(ENV.ApexSurvTools, tool)

    local _, _, led = BuildCCTVGeometry("tool", tool, nil)
    StartLedBlink(led, function() return tool and tool.Parent ~= nil end)

    -- Roblox's default grip treats local -Z as "away from player," but
    -- the lens stack runs along local +Z. Flipping Forward AND Right
    -- together is a clean 180° turn (not a mirror).
    tool.GripForward = Vector3.new(0, 0, 1)
    tool.GripRight   = Vector3.new(-1, 0, 0)
    tool.GripUp      = Vector3.new(0, 1, 0)
    tool.GripPos     = Vector3.new(0, -0.15, -0.3)

    tool.Equipped:Connect(function()
        PlacerEquipped = true
        CrosshairRoot.Visible = true
        RefreshUI()
    end)

    tool.Unequipped:Connect(function()
        PlacerEquipped = false
        CrosshairRoot.Visible = false
        if PlacingPreview then CancelPreview() else RefreshUI() end
    end)

    tool.Activated:Connect(function()
        if not PlacingPreview then
            if #Cameras >= MAX_CAMERAS then
                Notify("Camera limit reached (" .. MAX_CAMERAS .. ")", T.Red)
                return
            end
            local result = RaycastFromCamera(RAY_MAX_DIST)
            if not result then
                Notify("No surface in range", T.Red)
                return
            end
            PlacingPreview = true
            PlaceYaw = 0
            PlacePitch = 0
            RefreshUI()
        else
            ConfirmCameraPlacement()
            PlacingPreview = false
            RefreshUI()
        end
    end)

    BindToolToBackpack(tool)
    return tool
end

local function CreateTabletTool()
    local tool = Instance.new("Tool")
    tool.Name = "Apex Tablet"
    tool.RequiresHandle = true
    tool.CanBeDropped = false
    tool.ToolTip = "Equip, then tap a camera number in the panel to view it"
    table.insert(ENV.ApexSurvTools, tool)

    BuildTabletHandle(tool)
    tool.GripForward = Vector3.new(1, 0, 0)
    tool.GripRight   = Vector3.new(0, 0, -1)
    tool.GripUp      = Vector3.new(0, 1, 0)
    tool.GripPos     = Vector3.new(0, -0.1, 0)

    tool.Equipped:Connect(function()
        TabletOpen = true
        RefreshUI()
    end)

    tool.Unequipped:Connect(function()
        TabletOpen = false
        if ActivePOVCam then ExitCameraPOV() else RefreshUI() end
    end)

    BindToolToBackpack(tool)
    return tool
end

-- ════════════════════════════════════════════════════════════════
-- PC KEYBOARD
-- ════════════════════════════════════════════════════════════════
Track(UserInputService.InputBegan:Connect(function(inp, gpe)
    if gpe then return end
    local k = inp.KeyCode

    if k == Enum.KeyCode.Escape then
        if ActivePOVCam then ExitCameraPOV()
        elseif PlacingPreview then CancelPreview() end
        return
    end

    if ActivePOVCam or PlacingPreview then
        local dir
        if k == Enum.KeyCode.Left then dir = "left"
        elseif k == Enum.KeyCode.Right then dir = "right"
        elseif k == Enum.KeyCode.Up then dir = "up"
        elseif k == Enum.KeyCode.Down then dir = "down" end
        if dir then
            PadAction(dir, false)
            Bump()
            return
        end
    end

    if k == Enum.KeyCode.G then
        if #Cameras > 0 then DeleteLastCamera() end
        return
    end

    if k == Enum.KeyCode.H then
        ForceBackpackVisible()
        SetStatus("Inventory forced visible", T.Green)
        return
    end

    if k == Enum.KeyCode.P then
        if PanelExpanded then Collapse() else Expand() end
        return
    end
end))

-- ════════════════════════════════════════════════════════════════
-- RENDER LOOP — while viewing, the real camera CFrame is pinned here
-- every frame so pan/tilt apply immediately and nothing can knock the
-- Scriptable camera out of position.
-- ════════════════════════════════════════════════════════════════
Track(RunService.RenderStepped:Connect(function()
    if ActivePOVCam then
        local povCF = ComputePovCFrame(ActivePOVCam)
        GetCamera().CFrame = povCF
        if FlashlightPart and FlashlightOn then
            FlashlightPart.CFrame = povCF
        end
    elseif PlacingPreview then
        UpdateCameraPreview()
    elseif PreviewModel then
        DestroyPreview()
    end
    UpdateCrosshairFeedback()
end))

-- ════════════════════════════════════════════════════════════════
-- RESPAWN — a respawn replaces the Backpack, which can take local
-- tools with it, so re-bind them or rebuild any that are gone.
-- ════════════════════════════════════════════════════════════════
local function RestoreTool(tool, factory)
    if tool then
        local bp = GetBackpack()
        if tool.Parent == bp or tool.Parent == GetChar() then return tool end
        local ok = pcall(function() tool.Parent = bp end)
        if ok and tool.Parent == bp then return tool end
        pcall(function() tool:Destroy() end)
    end
    return factory()
end

Track(LP.CharacterAdded:Connect(function()
    if ActivePOVCam then ExitCameraPOV() end
    if PlacingPreview then CancelPreview() end
    TabletOpen = false
    PlacerEquipped = false
    CrosshairRoot.Visible = false
    RefreshUI()
    task.wait(0.5)
    ForceBackpackVisible()
    PlacerTool = RestoreTool(PlacerTool, CreateCameraPlacerTool)
    TabletTool = RestoreTool(TabletTool, CreateTabletTool)
    print("[ApexSurveillance] Character respawned.")
end))
