-- ════════════════════════════════════════════════════════════════
-- VIEWPORT SCENE BUILDER
-- ════════════════════════════════════════════════════════════════
local function BuildViewportScene(vp, capturedCF)
    for _, ch in ipairs(vp:GetChildren()) do
        if ch:IsA("Folder") then ch:Destroy() end
    end
    local origin = capturedCF.Position
    local container = Instance.new("Folder")
    container.Name = "_Scene"
    container.Parent = vp

    local count = 0
    local char = GetChar()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if count >= CLONE_LIMIT then break end
        if obj:IsA("BasePart") and obj.Transparency < 0.95
        and not (char and obj:IsDescendantOf(char)) then
            local dist = (obj.Position - origin).Magnitude
            if dist <= CLONE_RADIUS then
                local ok, clone = pcall(function() return obj:Clone() end)
                if ok and clone then
                    for _, ch in ipairs(clone:GetDescendants()) do
                        if ch:IsA("LuaSourceContainer") then ch:Destroy() end
                    end
                    clone.Anchored=true; clone.CanCollide=false; clone.CastShadow=false
                    clone.Parent=container
                    count = count + 1
                end
            end
        end
    end
    return count
end

-- BUILD PORTAL
-- ════════════════════════════════════════════════════════════════
local function BuildPortal(photo, placeCF)
    if photo.touchConn   then photo.touchConn:Disconnect() end
    if photo.portalModel then photo.portalModel:Destroy()  end
    photo.touchConn = nil
    photo.portalModel = nil
    photo.portalFrame = nil

    local model = Instance.new("Model")
    model.Name  = "ApexPortal_"..photo.id
    model.Parent = Workspace

    local panel = Instance.new("Part")
    panel.Name = "PortalPanel"
    panel.Size = Vector3.new(PORTAL_WIDTH, PORTAL_HEIGHT, PORTAL_DEPTH)
    panel.CFrame = placeCF
    panel.Anchored = true
    panel.CanCollide = false
    panel.CastShadow = false
    panel.Material = Enum.Material.Glass
    panel.Transparency = 0.08
    panel.Color = Color3.fromRGB(5,5,18)
    panel.Parent = model

    local edgeDefs = {
        { sz=Vector3.new(PORTAL_WIDTH+0.4,0.25,0.25), off=Vector3.new(0, PORTAL_HEIGHT/2+0.1,0) },
        { sz=Vector3.new(PORTAL_WIDTH+0.4,0.25,0.25), off=Vector3.new(0,-PORTAL_HEIGHT/2-0.1,0) },
        { sz=Vector3.new(0.25,PORTAL_HEIGHT+0.4,0.25), off=Vector3.new(-PORTAL_WIDTH/2-0.1,0,0) },
        { sz=Vector3.new(0.25,PORTAL_HEIGHT+0.4,0.25), off=Vector3.new( PORTAL_WIDTH/2+0.1,0,0) },
    }
    local edgeParts = {}
    for _, ed in ipairs(edgeDefs) do
        local e=Instance.new("Part")
        e.Size=ed.sz; e.CFrame=placeCF*CFrame.new(ed.off)
        e.Anchored=true; e.CanCollide=false; e.CastShadow=false
        e.Material=Enum.Material.Neon; e.Color=T.Violet
        e.Parent=model
        table.insert(edgeParts, e)
    end

    local cornerOffsets = {
        Vector3.new(-PORTAL_WIDTH/2,-PORTAL_HEIGHT/2,0), Vector3.new(PORTAL_WIDTH/2,-PORTAL_HEIGHT/2,0),
        Vector3.new(-PORTAL_WIDTH/2, PORTAL_HEIGHT/2,0), Vector3.new(PORTAL_WIDTH/2, PORTAL_HEIGHT/2,0),
    }
    for _, coff in ipairs(cornerOffsets) do
        local d=Instance.new("Part")
        d.Size=Vector3.new(0.5,0.5,0.5); d.CFrame=placeCF*CFrame.new(coff)
        d.Anchored=true; d.CanCollide=false; d.CastShadow=false
        d.Material=Enum.Material.Neon; d.Color=T.Cyan; d.Shape=Enum.PartType.Ball
        d.Parent=model
    end

    local sg = Instance.new("SurfaceGui")
    sg.Adornee=panel; sg.Face=Enum.NormalId.Front
    sg.SizingMode=Enum.SurfaceGuiSizingMode.PixelsPerStud
    sg.PixelsPerStud=60; sg.Parent=panel

    local vp = Instance.new("ViewportFrame")
    vp.BackgroundColor3=Color3.new(0,0,0); vp.BackgroundTransparency=0
    vp.Size=UDim2.new(1,0,1,-20)
    vp.Ambient=Color3.new(0.7,0.7,0.7); vp.LightDirection=Vector3.new(-1,-2,-1)
    vp.Parent=sg

    local vpCam = Instance.new("Camera")
    vpCam.CFrame=photo.capturedCF; vpCam.FieldOfView=70
    vpCam.Parent=vp; vp.CurrentCamera=vpCam

    local strip=Instance.new("Frame")
    strip.BackgroundColor3=T.GlassMid; strip.BackgroundTransparency=0.25
    strip.BorderSizePixel=0; strip.Size=UDim2.new(1,0,0,20)
    strip.Position=UDim2.new(0,0,1,-20); strip.ZIndex=2; strip.Parent=sg

    local stripLbl=Instance.new("TextLabel")
    stripLbl.BackgroundTransparency=1
    stripLbl.Text="PHOTO #"..photo.id
    stripLbl.TextColor3=T.Cyan; stripLbl.Font=Enum.Font.GothamBold; stripLbl.TextSize=9
    stripLbl.Size=UDim2.new(1,0,1,0); stripLbl.ZIndex=3; stripLbl.Parent=strip

    task.spawn(function()
        local count = BuildViewportScene(vp, photo.capturedCF)
        vpCam.CFrame = photo.capturedCF
        stripLbl.Text = "PHOTO #"..photo.id.."  |  "..count.." parts"
    end)

    task.spawn(function()
        local hue=0
        while model and model.Parent do
            hue=(hue+0.007)%1
            local col=Color3.fromHSV(hue,0.75,1)
            for _, ep in ipairs(edgeParts) do pcall(function() ep.Color=col end) end
            task.wait(0.05)
        end
    end)

    photo.portalModel=model
    photo.portalFrame=panel

    local debounce = {}
    photo.touchConn = panel.Touched:Connect(function(hit)
        local char = GetChar()
        if not char then return end
        if not hit:IsDescendantOf(char) then return end
        local uid = tostring(LP.UserId)
        if debounce[uid] then return end
        debounce[uid]=true

        DoFade(function()
            local r = GetRoot()
            if not r then return end
            local targetCF = photo.capturedCF
            local spawnPos = targetCF.Position
                + targetCF.LookVector*(-3.5)
                + Vector3.new(0,-3,0)
            r.CFrame = CFrame.new(spawnPos)
                * CFrame.Angles(0, math.atan2(-targetCF.LookVector.X,-targetCF.LookVector.Z), 0)
        end)

        task.delay(1.5, function() debounce[uid]=nil end)
    end)

    return model
end

-- ════════════════════════════════════════════════════════════════
-- PLACEMENT PREVIEW
-- ════════════════════════════════════════════════════════════════
local function GetPlaceCF()
    local vpSize = Cam.ViewportSize
    local ray = Cam:ScreenPointToRay(vpSize.X/2, vpSize.Y/2)

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local excludeList = {}
    local char = GetChar()
    if char then table.insert(excludeList, char) end
    for _, photo in ipairs(Photos) do
        if photo.portalModel then table.insert(excludeList, photo.portalModel) end
    end
    params.FilterDescendantsInstances = excludeList

    local result = Workspace:Raycast(ray.Origin, ray.Direction*PLACE_DIST, params)
    local hitPos, hitNorm
    if result then
        hitPos=result.Position; hitNorm=result.Normal
    else
        hitPos=ray.Origin+ray.Direction*(PLACE_DIST*0.6); hitNorm=-ray.Direction
    end

    local up=Vector3.new(0,1,0)
    local right=hitNorm:Cross(up)
    if right.Magnitude<0.001 then right=hitNorm:Cross(Vector3.new(1,0,0)) end
    right=right.Unit
    local trueUp=right:Cross(hitNorm).Unit

    return CFrame.fromMatrix(hitPos, right, trueUp, -hitNorm)
        * CFrame.new(0, PORTAL_HEIGHT/2+0.1, 0)
        * CFrame.Angles(0, math.rad(PlaceRotation), 0)
end

            local e=Instance.new("Part")
            e.Size=ed.sz; e.Anchored=true; e.CanCollide=false; e.CastShadow=false
            e.Material=Enum.Material.Neon; e.Color=T.VioletBr; e.Transparency=0.3
            e.Parent=PreviewModel
        end
    end

    local parts = PreviewModel:GetChildren()
    if parts[1] then parts[1].CFrame = cf end
    local offsets = {
        Vector3.new(0,PORTAL_HEIGHT/2+0.1,0), Vector3.new(0,-PORTAL_HEIGHT/2-0.1,0),
        Vector3.new(-PORTAL_WIDTH/2-0.1,0,0), Vector3.new(PORTAL_WIDTH/2+0.1,0,0),
    }
    for i, off in ipairs(offsets) do
        if parts[i+1] then parts[i+1].CFrame = cf * CFrame.new(off) end
    end
end

local function DestroyPreview()
    if PreviewModel then PreviewModel:Destroy(); PreviewModel=nil end
end

-- ════════════════════════════════════════════════════════════════
-- PLACEMENT FLOW CONTROL
-- ════════════════════════════════════════════════════════════════
local function CancelPlacement()
    ActivePlacingPhoto = nil
    PlaceRotation = 0
    PlaceHint.Visible = false
    DestroyPreview()
    RotateBtn.Visible = false
end

local function ConfirmPlacement()
    if not ActivePlacingPhoto then return end
    local cf = GetPlaceCF()
    DestroyPreview()
    BuildPortal(ActivePlacingPhoto, cf)
    CancelPlacement()
end

-- ════════════════════════════════════════════════════════════════
-- DELETE PHOTO / PORTAL
-- ════════════════════════════════════════════════════════════════
local function DeletePhoto(idx)
    local photo = Photos[idx]
    if not photo then return end
    if photo.touchConn   then photo.touchConn:Disconnect() end
    if photo.portalModel then photo.portalModel:Destroy()  end
    if photo.tool        then photo.tool:Destroy()         end
    if ActivePlacingPhoto == photo then CancelPlacement() end
    table.remove(Photos, idx)
    CountLabel.Text = #Photos.." / "..MAX_PHOTOS.." photos"
end

DeleteBtn.MouseButton1Click:Connect(function()
    if #Photos > 0 then DeletePhoto(#Photos) end
end)

-- ════════════════════════════════════════════════════════════════
-- TOOL FACTORY — CAMERA
-- Retry-safe: waits for a real Backpack instance, verifies the
-- parenting actually stuck, and reports status on-screen.
-- ════════════════════════════════════════════════════════════════
local CreatePhotoTool -- forward declare

local function BindToolToBackpack(tool, attempt)
    attempt = attempt or 1
    local bp = GetBackpack()
    tool.Parent = bp
    task.wait(0.1)
    if tool.Parent == bp then
        StatusLabel.Text = "Inventory bound OK ("..#bp:GetChildren().." items)"
        StatusLabel.TextColor3 = T.Green
        return true
    end
    if attempt < 5 then
        task.wait(0.3)
        return BindToolToBackpack(tool, attempt+1)
    end
    StatusLabel.Text = "Backpack bind FAILED — tap SHOW INV"
    StatusLabel.TextColor3 = T.Red
    warn("[ApexPhotoPortal] Failed to parent tool to Backpack after 5 attempts: "..tool.Name)
    return false
end

local function CreateCameraTool()
    local tool = Instance.new("Tool")
    tool.Name            = "Apex Camera"
    tool.RequiresHandle  = false
    tool.CanBeDropped    = false
    tool.ToolTip         = "Click to capture a photo"
    table.insert(ENV.ApexPhotoTools, tool)

    tool.Equipped:Connect(function()
        CrosshairRoot.Visible = true
        CancelPlacement()
    end)

    tool.Unequipped:Connect(function()
        CrosshairRoot.Visible = false
    end)

    tool.Activated:Connect(function()
        if #Photos >= MAX_PHOTOS then
            DoFlash(T.Red)
            return
        end

        DoFlash()
        pcall(function()
            local snd = Instance.new("Sound")
            snd.SoundId = "rbxassetid://3200504521"
            snd.Volume  = 0.5
            snd.Parent  = Cam
            snd:Play()
            Debris:AddItem(snd, 3)
        end)

        local photo = {
            id          = NextId,
            capturedCF  = Cam.CFrame,
            tool        = nil,
            portalModel = nil,
            portalFrame = nil,
            touchConn   = nil,
        }
        NextId = NextId + 1
        table.insert(Photos, photo)
        CountLabel.Text = #Photos.." / "..MAX_PHOTOS.." photos"

        CreatePhotoTool(photo)
    end)

    BindToolToBackpack(tool)
    return tool
end

-- ════════════════════════════════════════════════════════════════
-- TOOL FACTORY — PHOTO
-- ════════════════════════════════════════════════════════════════
CreatePhotoTool = function(photo)
    local tool = Instance.new("Tool")
    tool.Name            = "Apex Photo #"..photo.id
    tool.RequiresHandle  = false
    tool.CanBeDropped    = false
    tool.ToolTip         = "Click to preview placement, click again to place"
    photo.tool            = tool
    table.insert(ENV.ApexPhotoTools, tool)

    local placing = false

    tool.Equipped:Connect(function()
        placing = false
    end)

    tool.Unequipped:Connect(function()
        placing = false
        if ActivePlacingPhoto == photo then CancelPlacement() end
    end)

    tool.Activated:Connect(function()
        if not placing then
            ActivePlacingPhoto = photo
            placing = true
            PlaceRotation = 0
            PlaceHint.Visible = true
            RotateBtn.Visible = true
        else
            ConfirmPlacement()
            placing = false
        end
    end)

    BindToolToBackpack(tool)
    return tool
end

-- ════════════════════════════════════════════════════════════════
-- PC KEYBOARD SUPPORT
-- ════════════════════════════════════════════════════════════════
UserInputService.InputBegan:Connect(function(inp, gpe)
    if gpe then return end

    if inp.KeyCode == Enum.KeyCode.Escape then
        if ActivePlacingPhoto then
            CancelPlacement()
            local hum = GetHum()
            if hum then hum:UnequipTools() end
        end
        return
    end

    if inp.KeyCode == Enum.KeyCode.R and ActivePlacingPhoto then
        PlaceRotation = (PlaceRotation + 45) % 360
        return
    end

    if inp.KeyCode == Enum.KeyCode.G then
        if #Photos > 0 then DeletePhoto(#Photos) end
        return
    end

    if inp.KeyCode == Enum.KeyCode.H then
        ForceBackpackVisible()
        StatusLabel.Text = "Forced inventory visible"
        StatusLabel.TextColor3 = T.Green
        return
    end
end)
