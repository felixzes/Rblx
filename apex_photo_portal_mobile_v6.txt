-- SHARED RAYCAST HELPER

local function IsDescendantOfAny(obj, list)
    for _, m in ipairs(list) do
        if m and obj:IsDescendantOf(m) then return true end
    end
    return false
end

local function BuildExcludeList()
    local excludeList = {}
    local char = GetChar()
    if char then table.insert(excludeList, char) end
    for _, photo in ipairs(Photos) do
        if photo.portalModel then table.insert(excludeList, photo.portalModel) end
    end
    if PreviewModel then table.insert(excludeList, PreviewModel) end
    return excludeList
end

local function RaycastFromCamera(maxDist)
    local camera = GetCamera()
    local vpSize = camera.ViewportSize
    local ray = camera:ScreenPointToRay(vpSize.X/2, vpSize.Y/2)

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = BuildExcludeList()
    params.IgnoreWater = true

    local result = Workspace:Raycast(ray.Origin, ray.Direction * maxDist, params)

    print(("[ApexPhotoPortal] Raycast  origin=%s  dir=%s  hit=%s"):format(
        tostring(ray.Origin), tostring(ray.Direction),
        result and tostring(result.Position) or "NONE"
    ))

    return result, ray, camera
end

local function SafeClone(obj)
    local wasArchivable = obj.Archivable
    if not wasArchivable then
        local okSet = pcall(function() obj.Archivable = true end)
        if not okSet then return nil end
    end
    local okClone, clone = pcall(function() return obj:Clone() end)
    if not wasArchivable then
        pcall(function() obj.Archivable = false end)
    end
    if okClone then return clone end
    return nil
end

local function CloneTerrainInto(container, originPos)
    local radius = 56
    local okTerrain = pcall(function()
        local region = Region3.new(
            originPos - Vector3.new(radius, radius, radius),
            originPos + Vector3.new(radius, radius, radius)
        ):ExpandToGrid(4)
        local resolution = 4
        local materials, occupancies = Workspace.Terrain:ReadVoxels(region, resolution)
        local newTerrain = Instance.new("Terrain")
        newTerrain.Parent = container
        newTerrain:WriteVoxels(region, resolution, materials, occupancies)
    end)
    return okTerrain
end

local function AddGuaranteedBackdrop(container, originPos)
    -- Always present, regardless of whether real cloning or terrain
    -- capture succeed. This is the fix for "just black" — there is
    -- now always SOMETHING in the scene to render: a ground plane,
    -- a sky-colored ceiling plane, and four horizon walls forming
    -- an open box around the origin. Real captured geometry (below)
    -- layers on top of this when it succeeds; when it doesn't, you
    -- still see a colored room instead of a void.
    local ground = Instance.new("Part")
    ground.Name = "_BackdropGround"
    ground.Size = Vector3.new(500, 2, 500)
    ground.CFrame = CFrame.new(originPos.X, originPos.Y - 4, originPos.Z)
    ground.Anchored = true; ground.CanCollide = false; ground.CastShadow = false
    ground.Material = Enum.Material.SmoothPlastic
    ground.Color = Color3.fromRGB(70, 74, 86)
    ground.Parent = container

    local sky = Instance.new("Part")
    sky.Name = "_BackdropSky"
    sky.Size = Vector3.new(500, 2, 500)
    sky.CFrame = CFrame.new(originPos.X, originPos.Y + 90, originPos.Z)
    sky.Anchored = true; sky.CanCollide = false; sky.CastShadow = false
    sky.Material = Enum.Material.SmoothPlastic
    sky.Color = Color3.fromRGB(120, 168, 215)
    sky.Parent = container

    local wallDefs = {
        {size=Vector3.new(500,160,2), off=Vector3.new(0,40,250)},
        {size=Vector3.new(500,160,2), off=Vector3.new(0,40,-250)},
        {size=Vector3.new(2,160,500), off=Vector3.new(250,40,0)},
        {size=Vector3.new(2,160,500), off=Vector3.new(-250,40,0)},
    }
    for i, wd in ipairs(wallDefs) do
        local w = Instance.new("Part")
        w.Name = "_BackdropWall"..i
        w.Size = wd.size
        w.CFrame = CFrame.new(originPos + wd.off)
        w.Anchored = true; w.CanCollide = false; w.CastShadow = false
        w.Material = Enum.Material.SmoothPlastic
        w.Color = Color3.fromRGB(95, 130, 165)
        w.Parent = container
    end
end

local function BuildViewportScene(vp, originPos, excludeModels)
    for _, ch in ipairs(vp:GetChildren()) do
        if ch:IsA("Folder") then ch:Destroy() end
    end
    local container = Instance.new("Folder")
    container.Name = "_Scene"
    container.Parent = vp

    AddGuaranteedBackdrop(container, originPos)

    local count = 0
    local char = GetChar()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if count >= CLONE_LIMIT then break end
        if obj:IsA("BasePart") and obj.Transparency < 0.95
        and not (char and obj:IsDescendantOf(char))
        and not IsDescendantOfAny(obj, excludeModels or {}) then
            local dist = (obj.Position - originPos).Magnitude
            if dist <= CLONE_RADIUS then
                local clone = SafeClone(obj)
                if clone then
                    for _, ch in ipairs(clone:GetDescendants()) do
                        if ch:IsA("LuaSourceContainer") then ch:Destroy() end
                    end
                    clone.Anchored=true; clone.CanCollide=false; clone.CastShadow=false
                    clone.Parent=container
                    count = count + 1
                end
            end
        end
    end

    local terrainOK = CloneTerrainInto(container, originPos)

    return count, terrainOK
end

local function ComputeViewCFrame(photo)
    -- Fix: the old version sat in front of the captured surface and
    -- looked straight back at it (a point-blank wall texture). This
    -- version sits near the destination, at eye height, and looks
    -- ONWARD in the same direction you were aiming — the same
    -- vantage point you'll actually be teleported into.
    local lookDir = photo.camLookDir
    local horiz = Vector3.new(lookDir.X, 0, lookDir.Z)
    if horiz.Magnitude < 0.001 then horiz = Vector3.new(0, 0, -1) end
    horiz = horiz.Unit

    local liftPos = photo.targetPos + Vector3.new(0, VIEW_EYE_HEIGHT, 0) - horiz * 1.5
    return CFrame.new(liftPos, liftPos + horiz)
end

local function FindSafeGroundPosition(targetPos, targetNormal)
    local castOrigin = targetPos + targetNormal * 2 + Vector3.new(0, 8, 0)

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local char = GetChar()
    params.FilterDescendantsInstances = char and {char} or {}

    local result = Workspace:Raycast(castOrigin, Vector3.new(0, -60, 0), params)

    if result then
        local safePos = result.Position + Vector3.new(0, GROUND_CLEARANCE, 0)
        print(("[ApexPhotoPortal] Ground found below target at %s -> safe pos %s"):format(
            tostring(result.Position), tostring(safePos)))
        return safePos, true
    end

    local fallback = targetPos + targetNormal * 3 + Vector3.new(0, GROUND_CLEARANCE, 0)
    warn("[ApexPhotoPortal] No ground found below target — using ungrounded fallback: "..tostring(fallback))
    return fallback, false
end

local function TeleportToPhoto(photo)
    local r = GetRoot()
    if not r then
        warn("[ApexPhotoPortal] Teleport aborted: no HumanoidRootPart.")
        return false
    end

    local safePos, grounded = FindSafeGroundPosition(photo.targetPos, photo.targetNormal)

    local lookDir = photo.camLookDir
    local horiz = Vector3.new(lookDir.X, 0, lookDir.Z)
    if horiz.Magnitude < 0.001 then horiz = Vector3.new(0, 0, -1) end
    horiz = horiz.Unit

    r.CFrame = CFrame.new(safePos, safePos + horiz)

    print(("[ApexPhotoPortal] TELEPORT photo#%d  target=%s  safe=%s  grounded=%s")
        :format(photo.id, tostring(photo.targetPos), tostring(safePos), tostring(grounded)))

    return true
end

-- ════════════════════════════════════════════════════════════════
-- PORTAL PLACEMENT — now yaw + pitch + flip, not just one axis
-- ════════════════════════════════════════════════════════════════
local function GetPlaceCF()
    local result, ray, camera = RaycastFromCamera(RAY_MAX_DIST)

    local hitPos, hitNorm
    if result then
        hitPos = result.Position
        hitNorm = result.Normal
    else
        hitPos = ray.Origin + ray.Direction * (RAY_MAX_DIST * 0.05)
        hitNorm = -ray.Direction
        warn("[ApexPhotoPortal] Placement raycast missed — using fallback point in front of camera.")
    end

    local lookDir
    if math.abs(hitNorm.Y) < 0.7 then
        lookDir = Vector3.new(hitNorm.X, 0, hitNorm.Z)
    else
        local camLook = camera.CFrame.LookVector
        lookDir = Vector3.new(camLook.X, 0, camLook.Z)
    end
    if lookDir.Magnitude < 0.001 then
        lookDir = Vector3.new(0, 0, -1)
    end
    lookDir = lookDir.Unit

    local placePos = hitPos + Vector3.new(0, PORTAL_HEIGHT/2 + 0.1, 0)
    local baseCF = CFrame.new(placePos, placePos + lookDir)

    local adjustedCF = baseCF * CFrame.Angles(math.rad(PlacePitch), math.rad(PlaceYaw), 0)
    if PlaceFlipped then
        adjustedCF = adjustedCF * CFrame.Angles(0, math.rad(180), 0)
    end

    return adjustedCF, result ~= nil
end

-- BUILD PORTAL
-- Stores vp/vpCam/stripLbl on the photo table now, so the refresh
-- loop below can re-render them without rebuilding from scratch.
-- ════════════════════════════════════════════════════════════════
local function BuildPortal(photo, placeCF)
    if photo.portalModel then photo.portalModel:Destroy() end
    photo.portalModel = nil
    photo.portalFrame = nil
    photo.inCooldown  = false
    photo.vp          = nil
    photo.vpCam        = nil
    photo.stripLbl     = nil

    local model = Instance.new("Model")
    model.Name  = "ApexPortal_"..photo.id
    model.Parent = Workspace

    local panel = Instance.new("Part")
    panel.Name = "PortalPanel"
    panel.Size = Vector3.new(PORTAL_WIDTH, PORTAL_HEIGHT, PORTAL_DEPTH)
    panel.CFrame = placeCF
    panel.Anchored = true
    panel.CanCollide = false
    panel.CastShadow = false
    panel.Material = Enum.Material.Glass
    panel.Transparency = 0.08
    panel.Color = Color3.fromRGB(5,5,18)
    panel.Parent = model

    local edgeDefs = {
        { sz=Vector3.new(PORTAL_WIDTH+0.4,0.25,0.25), off=Vector3.new(0, PORTAL_HEIGHT/2+0.1,0) },
        { sz=Vector3.new(PORTAL_WIDTH+0.4,0.25,0.25), off=Vector3.new(0,-PORTAL_HEIGHT/2-0.1,0) },
        { sz=Vector3.new(0.25,PORTAL_HEIGHT+0.4,0.25), off=Vector3.new(-PORTAL_WIDTH/2-0.1,0,0) },
        { sz=Vector3.new(0.25,PORTAL_HEIGHT+0.4,0.25), off=Vector3.new( PORTAL_WIDTH/2+0.1,0,0) },
    }
    local edgeParts = {}
    for _, ed in ipairs(edgeDefs) do
        local e=Instance.new("Part")
        e.Size=ed.sz; e.CFrame=placeCF*CFrame.new(ed.off)
        e.Anchored=true; e.CanCollide=false; e.CastShadow=false
        e.Material=Enum.Material.Neon; e.Color=T.Violet
        e.Parent=model
        table.insert(edgeParts, e)
    end

    local cornerOffsets = {
        Vector3.new(-PORTAL_WIDTH/2,-PORTAL_HEIGHT/2,0), Vector3.new(PORTAL_WIDTH/2,-PORTAL_HEIGHT/2,0),
        Vector3.new(-PORTAL_WIDTH/2, PORTAL_HEIGHT/2,0), Vector3.new(PORTAL_WIDTH/2, PORTAL_HEIGHT/2,0),
    }
    for _, coff in ipairs(cornerOffsets) do
        local d=Instance.new("Part")
        d.Size=Vector3.new(0.5,0.5,0.5); d.CFrame=placeCF*CFrame.new(coff)
        d.Anchored=true; d.CanCollide=false; d.CastShadow=false
        d.Material=Enum.Material.Neon; d.Color=T.Cyan; d.Shape=Enum.PartType.Ball
        d.Parent=model
    end

    sg.Adornee=panel; sg.Face=Enum.NormalId.Front
    sg.SizingMode=Enum.SurfaceGuiSizingMode.PixelsPerStud
    sg.PixelsPerStud=60; sg.Parent=panel

    local vp = Instance.new("ViewportFrame")
    vp.BackgroundColor3=Color3.new(0,0,0); vp.BackgroundTransparency=0
    vp.Size=UDim2.new(1,0,1,-20)
    vp.Ambient=Color3.fromRGB(205,205,205); vp.LightDirection=Vector3.new(-1,-2,-1)
    vp.Parent=sg

    local viewCF = ComputeViewCFrame(photo)

    local vpCam = Instance.new("Camera")
    vpCam.CFrame=viewCF; vpCam.FieldOfView=70
    vpCam.Parent=vp; vp.CurrentCamera=vpCam

    local strip=Instance.new("Frame")
    strip.BackgroundColor3=T.GlassMid; strip.BackgroundTransparency=0.25
    strip.BorderSizePixel=0; strip.Size=UDim2.new(1,0,0,20)
    strip.Position=UDim2.new(0,0,1,-20); strip.ZIndex=2; strip.Parent=sg

    local stripLbl=Instance.new("TextLabel")
    stripLbl.BackgroundTransparency=1
    stripLbl.Text="PHOTO #"..photo.id.."  |  rendering..."

    stripLbl.Size=UDim2.new(1,0,1,0); stripLbl.ZIndex=3; stripLbl.Parent=strip

    photo.portalFrame=panel
    photo.inCooldown  = false
    photo.vp          = vp
    photo.vpCam        = vpCam
    photo.stripLbl     = stripLbl

    task.spawn(function()
        local excludeModels = {model}
        for _, p2 in ipairs(Photos) do
            if p2.portalModel and p2.portalModel ~= model then
                table.insert(excludeModels, p2.portalModel)
            end
        end
        if PreviewModel then table.insert(excludeModels, PreviewModel) end

        local count, terrainOK = BuildViewportScene(vp, viewCF.Position, excludeModels)
        vpCam.CFrame = viewCF
        stripLbl.Text = "PHOTO #"..photo.id.."  |  backdrop+"..count.." parts"
            ..(terrainOK and "  |  terrain OK" or "")

        print(("[ApexPhotoPortal] Portal #%d viewport camera CFrame=%s  parts=%d  terrain=%s")
            :format(photo.id, tostring(viewCF), count, tostring(terrainOK)))
    end)

    task.spawn(function()
        local hue=0
        while model and model.Parent do
            hue=(hue+0.007)%1
            local col=Color3.fromHSV(hue,0.75,1)
            for _, ep in ipairs(edgeParts) do pcall(function() ep.Color=col end) end
            task.wait(0.05)
        end
    end)

    return model
end

-- ════════════════════════════════════════════════════════════════
-- TELEPORT: HEARTBEAT PROXIMITY CHECK
-- ════════════════════════════════════════════════════════════════
local function CheckPortalProximity()
    local root = GetRoot()
    if not root then return end

    for _, photo in ipairs(Photos) do
        if photo.portalFrame and not photo.inCooldown then
            local panel = photo.portalFrame
            local ok, rel = pcall(function()
                return panel.CFrame:PointToObjectSpace(root.Position)
            end)
            if ok then
                local inDepth  = math.abs(rel.Z) <= (PORTAL_DEPTH/2 + TRIGGER_DEPTH_BUFFER)
                local inWidth  = math.abs(rel.X) <= (PORTAL_WIDTH/2  - TRIGGER_EDGE_INSET)
                local inHeight = math.abs(rel.Y) <= (PORTAL_HEIGHT/2 - TRIGGER_EDGE_INSET)

                if inDepth and inWidth and inHeight then
                    photo.inCooldown = true
                    DoFade(function()
                        local success = TeleportToPhoto(photo)
                        if not success then
                            warn("[ApexPhotoPortal] Teleport failed for photo #"..photo.id)
                        end
                    end)
                    task.delay(1.75, function() photo.inCooldown = false end)
                end
            end
        end
    end
end

-- ════════════════════════════════════════════════════════════════
-- PORTAL LIVE-REFRESH LOOP
-- Re-renders one portal's scene at a time, round-robin, instead of
-- a single frozen snapshot — keeps what you see through a portal
-- reasonably current without re-cloning everything every frame.
-- ════════════════════════════════════════════════════════════════
local RefreshCursor = 0

local function RefreshOnePortal()
    if #Photos == 0 then return end
    RefreshCursor = (RefreshCursor % #Photos) + 1
    local photo = Photos[RefreshCursor]
    if not (photo and photo.portalModel and photo.vp) then return end

    local excludeModels = {photo.portalModel}
    for _, p2 in ipairs(Photos) do
        if p2.portalModel and p2.portalModel ~= photo.portalModel then
            table.insert(excludeModels, p2.portalModel)
        end
    end
    if PreviewModel then table.insert(excludeModels, PreviewModel) end

    local viewCF = ComputeViewCFrame(photo)
    local count, terrainOK = BuildViewportScene(photo.vp, viewCF.Position, excludeModels)
    if photo.vpCam then photo.vpCam.CFrame = viewCF end
    if photo.stripLbl then
        photo.stripLbl.Text = "PHOTO #"..photo.id.."  |  backdrop+"..count.." parts"
            ..(terrainOK and "  |  terrain OK" or "").."  |  live"
    end
end

local refreshLoop = task.spawn(function()
    while true do
        local n = math.max(1, #Photos)
        task.wait(PORTAL_REFRESH_INTERVAL / n)
        local ok, err = pcall(RefreshOnePortal)
        if not ok then warn("[ApexPhotoPortal] Refresh loop error: "..tostring(err)) end
    end
end)
table.insert(ENV.ApexPhotoLoops, {Disconnect=function() task.cancel(refreshLoop) end})

    task.wait(0.1)
    if tool.Parent == bp then
        StatusLabel.Text = "Inventory bound OK ("..#bp:GetChildren().." items)"
        StatusLabel.TextColor3 = T.Green
        return true
    end
    if attempt < 5 then
        task.wait(0.3)
        return BindToolToBackpack(tool, attempt+1)
    end
    StatusLabel.Text = "Backpack bind FAILED — tap SHOW INV"
    StatusLabel.TextColor3 = T.Red
    warn("[ApexPhotoPortal] Failed to parent tool to Backpack after 5 attempts: "..tool.Name)
    return false
end

local function CreateCameraTool()
    local tool = Instance.new("Tool")
    tool.Name            = "Apex Camera"
    tool.RequiresHandle  = true
    tool.CanBeDropped    = false
    tool.ToolTip         = "Click to capture a photo of what you're aiming at"
    table.insert(ENV.ApexPhotoTools, tool)

    BuildCameraModel(tool)

    tool.Equipped:Connect(function()
        CrosshairRoot.Visible = true
        CancelPlacement()
    end)

    tool.Unequipped:Connect(function()
        CrosshairRoot.Visible = false
    end)

    tool.Activated:Connect(function()
        if #Photos >= MAX_PHOTOS then
            DoFlash(T.Red)
            return
        end

        local result, ray, camera = RaycastFromCamera(RAY_MAX_DIST)
        if not result then
            DoFlash(T.Red)
            warn("[ApexPhotoPortal] Photo capture failed: raycast found no surface within "..RAY_MAX_DIST.." studs.")
            return
        end

        DoFlash()
        pcall(function()
            local snd = Instance.new("Sound")
            snd.SoundId = "rbxassetid://3200504521"
            snd.Volume  = 0.5
            snd.Parent  = camera
            snd:Play()
            Debris:AddItem(snd, 3)
        end)

        local photo = {
            id          = NextId,
            targetPos   = result.Position,
            targetNormal= result.Normal,
            camLookDir  = camera.CFrame.LookVector,
            tool        = nil,
            portalModel = nil,
            portalFrame = nil,
            inCooldown  = false,
            vp          = nil,
            vpCam       = nil,
            stripLbl    = nil,
        }
        NextId = NextId + 1
        table.insert(Photos, photo)
        CountLabel.Text = #Photos.." / "..MAX_PHOTOS.." photos"

        print(("[ApexPhotoPortal] Captured photo #%d  target=%s  normal=%s  dist=%.1f")
            :format(photo.id, tostring(photo.targetPos), tostring(photo.targetNormal),
                (photo.targetPos - camera.CFrame.Position).Magnitude))

        CreatePhotoTool(photo)
    end)

    BindToolToBackpack(tool)
    return tool
end

-- TOOL FACTORY — PHOTO
-- ════════════════════════════════════════════════════════════════
CreatePhotoTool = function(photo)
    local tool = Instance.new("Tool")
    tool.Name            = "Apex Photo #"..photo.id
    tool.RequiresHandle  = true
    tool.CanBeDropped    = false
    tool.ToolTip         = "Click to preview placement, click again to place"
    photo.tool            = tool
    table.insert(ENV.ApexPhotoTools, tool)

    BuildPolaroidHandle(tool)

    local placing = false

    tool.Equipped:Connect(function()
        placing = false
    end)

    tool.Unequipped:Connect(function()
        placing = false
        if ActivePlacingPhoto == photo then CancelPlacement() end
    end)

    tool.Activated:Connect(function()
        if not placing then
            ActivePlacingPhoto = photo
            placing = true
            PlaceYaw = 0
            PlacePitch = 0
            PlaceFlipped = false
            PlaceHint.Visible = true
            AimPad.Visible = true
        else
            ConfirmPlacement()
            placing = false
        end
    end)

    BindToolToBackpack(tool)
    return tool
end

-- ════════════════════════════════════════════════════════════════
-- PC KEYBOARD SUPPORT
-- ════════════════════════════════════════════════════════════════
UserInputService.InputBegan:Connect(function(inp, gpe)
    if gpe then return end

    if inp.KeyCode == Enum.KeyCode.Escape then
        if ActivePlacingPhoto then
            CancelPlacement()
            local hum = GetHum()
            if hum then hum:UnequipTools() end
        end
        return
    end

    if ActivePlacingPhoto then
        if inp.KeyCode == Enum.KeyCode.Left  or inp.KeyCode == Enum.KeyCode.R then
            AdjustYaw(-45); return
        elseif inp.KeyCode == Enum.KeyCode.Right then
            AdjustYaw(45); return
        elseif inp.KeyCode == Enum.KeyCode.Up then
            AdjustPitch(-15); return
        elseif inp.KeyCode == Enum.KeyCode.Down then
            AdjustPitch(15); return
        elseif inp.KeyCode == Enum.KeyCode.F then
            ToggleFlip(); return
        end
    end

    if inp.KeyCode == Enum.KeyCode.G then
        if #Photos > 0 then DeletePhoto(#Photos) end
        return
    end

    if inp.KeyCode == Enum.KeyCode.H then
        ForceBackpackVisible()
        StatusLabel.Text = "Forced inventory visible"
        StatusLabel.TextColor3 = T.Green
        return
    end
end)
