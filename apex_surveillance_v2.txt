local function IsDescendantOfAny(obj, list)
    for _, m in ipairs(list) do
        if m and obj:IsDescendantOf(m) then return true end
    end
    return false
end

local function BuildExcludeList()
    local excludeList = {}
    local char = GetChar()
    if char then table.insert(excludeList, char) end
    for _, cam in ipairs(Cameras) do
        if cam.model then table.insert(excludeList, cam.model) end
    end
    if PreviewModel then table.insert(excludeList, PreviewModel) end
    return excludeList
end

local function RaycastFromCamera(maxDist)
    local camera = GetCamera()
    local vpSize = camera.ViewportSize
    local ray = camera:ScreenPointToRay(vpSize.X/2, vpSize.Y/2)

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = BuildExcludeList()
    params.IgnoreWater = true

    local result = Workspace:Raycast(ray.Origin, ray.Direction * maxDist, params)

    print(("[ApexSurveillance] Raycast  origin=%s  dir=%s  hit=%s"):format(
        tostring(ray.Origin), tostring(ray.Direction), result and tostring(result.Position) or "NONE"))

    return result, ray, camera
end

local function CrosshairSilentRaycast()
    local camera = GetCamera()
    local vpSize = camera.ViewportSize
    local ray = camera:ScreenPointToRay(vpSize.X/2, vpSize.Y/2)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = BuildExcludeList()
    params.IgnoreWater = true
    return Workspace:Raycast(ray.Origin, ray.Direction * RAY_MAX_DIST, params), camera
end

-- ════════════════════════════════════════════════════════════════
-- SAFE CLONE + TERRAIN + GUARANTEED BACKDROP
-- (same pattern proven out in the earlier photo-portal project)
-- ════════════════════════════════════════════════════════════════
local function SafeClone(obj)
    local wasArchivable = obj.Archivable
    if not wasArchivable then
        local okSet = pcall(function() obj.Archivable = true end)
        if not okSet then return nil end
    end
    local okClone, clone = pcall(function() return obj:Clone() end)
    if not wasArchivable then
        pcall(function() obj.Archivable = false end)
    end
    if okClone then return clone end
    return nil
end

local function CloneTerrainInto(container, originPos)
    local radius = 50
    local okTerrain = pcall(function()
        local region = Region3.new(
            originPos - Vector3.new(radius, radius, radius),
            originPos + Vector3.new(radius, radius, radius)
        ):ExpandToGrid(4)
        local resolution = 4
        local materials, occupancies = Workspace.Terrain:ReadVoxels(region, resolution)
        local newTerrain = Instance.new("Terrain")
        newTerrain.Parent = container
        newTerrain:WriteVoxels(region, resolution, materials, occupancies)
    end)
    return okTerrain
end

local function CloneCharactersInto(container, originPos)
    -- Unlike a one-shot photo, a security camera SHOULD see players —
    -- that's the entire point. Every character within range is cloned
    -- fresh on every refresh, so movement reads as (choppy, refresh-
    -- rate-limited) live activity rather than a frozen pose.
    local count = 0
    for _, p in ipairs(Players:GetPlayers()) do
        local c = p.Character
        if c then
            local ok, pivot = pcall(function() return c:GetPivot() end)
            if ok and (pivot.Position - originPos).Magnitude <= CAM_CLONE_RADIUS then
                for _, part in ipairs(c:GetDescendants()) do
                    if part:IsA("BasePart") and part.Transparency < 0.95 then
                        local clone = SafeClone(part)
                        if clone then
                            clone.Anchored=true; clone.CanCollide=false; clone.CastShadow=false
                            clone.Parent=container
                            count = count + 1
                        end
                    end
                end
            end
        end
    end
    return count
end

-- CAMERA SCENE BUILDER ("recording")
-- ════════════════════════════════════════════════════════════════
local function BuildCameraScene(camData)
    if not camData.vp then return 0, 0, false end

    for _, ch in ipairs(camData.vp:GetChildren()) do
        if ch:IsA("Folder") then ch:Destroy() end
    end
    local container = Instance.new("Folder")
    container.Name = "_Scene"
    container.Parent = camData.vp

    local originPos = camData.mountCFrame.Position
    AddGuaranteedBackdrop(container, originPos)

    local excludeModels = {}
    for _, cam in ipairs(Cameras) do
        if cam.model then table.insert(excludeModels, cam.model) end
    end
    if PreviewModel then table.insert(excludeModels, PreviewModel) end

    local partCount = 0
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if partCount >= CAM_CLONE_LIMIT then break end
        if obj:IsA("BasePart") and obj.Transparency < 0.95
        and not IsDescendantOfAny(obj, excludeModels) then
            -- character parts are handled separately below so they can
            -- be refreshed every cycle regardless of this loop's limit
            local isCharPart = false
            for _, p in ipairs(Players:GetPlayers()) do
                if p.Character and obj:IsDescendantOf(p.Character) then isCharPart = true; break end
            end
            if not isCharPart then
                local dist = (obj.Position - originPos).Magnitude
                if dist <= CAM_CLONE_RADIUS then
                    local clone = SafeClone(obj)
                    if clone then
                        for _, ch in ipairs(clone:GetDescendants()) do
                            if ch:IsA("LuaSourceContainer") then ch:Destroy() end
                        end
                        clone.Anchored=true; clone.CanCollide=false; clone.CastShadow=false
                        clone.Parent=container
                        partCount = partCount + 1
                    end
                end
            end
        end
    end

    local charCount = CloneCharactersInto(container, originPos)
    local terrainOK = CloneTerrainInto(container, originPos)

    local lensPos = camData.mountCFrame * CCTV_LENS_TIP_LOCAL
    camData.vpCam.CFrame = CFrame.new(lensPos, lensPos + camData.mountCFrame.LookVector)

    return partCount, charCount, terrainOK
end

-- ════════════════════════════════════════════════════════════════
-- CCTV GEOMETRY — shared by the held placer tool AND placed fixtures
-- so investing effort here pays off in both places. mode="tool":
-- unanchored, welded to a Handle, parented to a Tool. mode="fixture":
-- anchored, positioned via baseCFrame, parented to a Model.
-- ════════════════════════════════════════════════════════════════
local function BuildCCTVGeometry(mode, container, baseCFrame)
    local bodyColor  = Color3.fromRGB(42, 46, 54)
    local bodyDark   = Color3.fromRGB(22, 24, 30)
    local metalColor = Color3.fromRGB(152, 157, 167)
    local darkMetal  = Color3.fromRGB(32, 34, 40)
    local lensGlass  = Color3.fromRGB(8, 20, 35)
    local ledRed     = Color3.fromRGB(255, 40, 40)
    local ledIR      = Color3.fromRGB(95, 22, 22)

    baseCFrame = baseCFrame or CFrame.new(0,0,0)
    local parts = {}

    local function NewPart(name, kind, size, localCF, material, color, extraProps)
        local p
        if kind == "cylinder" or kind == "box" then
            p = Instance.new("Part")
            if kind == "cylinder" then p.Shape = Enum.PartType.Cylinder end
        elseif kind == "ball" then
            p = Instance.new("Part")
            p.Shape = Enum.PartType.Ball
        end
        p.Name = name
        p.Size = size
        p.Anchored = (mode == "fixture")
        p.CanCollide = false
        p.CastShadow = true
        p.Material = material
        p.Color = color
        if extraProps then
            for k, v in pairs(extraProps) do p[k] = v end
        end

        if mode == "fixture" then
            p.CFrame = baseCFrame * localCF
            p.Parent = container
        else
            p.CFrame = localCF
            p.Parent = container
        end

        table.insert(parts, p)
        return p
    end

    -- Root wall plate (also the Tool's required "Handle" in tool mode).
    -- Z is the thin dimension — the plate sits flush against the wall,
    -- with everything else extending forward along +Z from here.
    local root = NewPart(mode=="tool" and "Handle" or "MountPlate", "box",
        Vector3.new(0.9, 1.1, 0.18), CFrame.new(0,0,0), Enum.Material.Metal, darkMetal)

    -- corner rivets (on the forward-facing side of the plate)
    for _, off in ipairs({
        Vector3.new(0.35, 0.45, 0.1), Vector3.new(-0.35, 0.45, 0.1),
        Vector3.new(0.35, -0.45, 0.1), Vector3.new(-0.35, -0.45, 0.1),
    }) do
        NewPart("Rivet", "ball", Vector3.new(0.08,0.08,0.08), CFrame.new(off), Enum.Material.Metal, metalColor)
    end

    -- swivel joints + angled mount arm (reference: articulated bracket).
    -- The Arm cylinder's own Size.X is its length regardless of which
    -- world axis it's rotated to point along — rotating it ~90° around
    -- Y converts its local length axis to point along Z (forward),
    -- with a small extra tilt for a slightly bent, less robotic look.
    NewPart("Joint1", "ball", Vector3.new(0.34,0.34,0.34),
        CFrame.new(0, 0, 0.25), Enum.Material.Metal, metalColor)
    NewPart("Arm", "cylinder", Vector3.new(1.1, 0.2, 0.2),
        CFrame.new(0, -0.05, 0.75) * CFrame.Angles(math.rad(10), math.rad(90), 0),
        Enum.Material.Metal, metalColor)
    NewPart("Joint2", "ball", Vector3.new(0.3,0.3,0.3),
        CFrame.new(0, -0.1, 1.35), Enum.Material.Metal, metalColor)

    -- bullet housing — long dimension now along Z
    NewPart("Housing", "box", Vector3.new(0.68, 0.68, 1.9),
        CFrame.new(CCTV_HOUSING_OFFSET), Enum.Material.SmoothPlastic, bodyColor)
    NewPart("HousingPanel", "box", Vector3.new(0.55, 0.55, 0.5),
        CFrame.new(CCTV_HOUSING_OFFSET + Vector3.new(0,0,0.85)), Enum.Material.SmoothPlastic, bodyDark)

    -- lens — cylinders rotated to point along Z, positioned along Z;
    -- these now actually agree with each other
    NewPart("LensRing", "cylinder", Vector3.new(0.18, 0.5, 0.5),
        CFrame.new(CCTV_HOUSING_OFFSET + Vector3.new(0,0,1.15)) * CFrame.Angles(0, math.rad(90), 0),
        Enum.Material.Metal, darkMetal)
    NewPart("LensGlass", "cylinder", Vector3.new(0.08, 0.4, 0.4),
        CFrame.new(CCTV_HOUSING_OFFSET + Vector3.new(0,0,1.28)) * CFrame.Angles(0, math.rad(90), 0),
        Enum.Material.Glass, lensGlass, {Transparency=0.2})

    -- IR LED ring around the lens — ring now sits in the XY plane at a
    -- fixed Z depth, perpendicular to the forward axis as intended
    for i = 0, 5 do
        local ang = math.rad(i * 60)
        local off = Vector3.new(math.sin(ang)*0.22, math.cos(ang)*0.22, 1.2)
        NewPart("IRLed", "ball", Vector3.new(0.06,0.06,0.06),
            CFrame.new(CCTV_HOUSING_OFFSET + off), Enum.Material.Neon, ledIR)
    end

    -- status/recording LED (caller animates its Transparency to blink)
    local statusLed = NewPart("StatusLED", "ball", Vector3.new(0.1,0.1,0.1),
        CFrame.new(CCTV_HOUSING_OFFSET + Vector3.new(0, 0.3, 0.3)), Enum.Material.Neon, ledRed)

    if mode == "tool" then
        for _, p in ipairs(parts) do
            if p ~= root then WeldTo(root, p) end
        end
    end

    return root, parts, statusLed
end

local function StartLedBlink(led, getAlive)
    task.spawn(function()
        while getAlive() do
            led.Transparency = 0
            task.wait(0.6)
            if not getAlive() then break end
            led.Transparency = 0.7
            task.wait(0.6)
        end
    end)
end

-- ════════════════════════════════════════════════════════════════
-- PLACEMENT: raycast target + yaw/pitch, ghost preview, confirm
-- ════════════════════════════════════════════════════════════════
local function GetCameraPlaceCF()
    local result, ray, camera = RaycastFromCamera(RAY_MAX_DIST)
    local hitPos, hitNorm
    if result then
        hitPos = result.Position
        hitNorm = result.Normal
    else
        hitPos = ray.Origin + ray.Direction * (RAY_MAX_DIST * 0.05)
        hitNorm = -ray.Direction
        warn("[ApexSurveillance] Placement raycast missed — using fallback point in front of camera.")
    end

    -- Mount flush to the surface, lens pointing outward along its
    -- normal (works the same for walls, ceilings, or a floor-standing
    -- mount), then let the player fine-tune with yaw/pitch.
    local baseCF = CFrame.new(hitPos, hitPos + hitNorm)
    local adjustedCF = baseCF * CFrame.Angles(math.rad(PlacePitch), math.rad(PlaceYaw), 0)
    return adjustedCF, result ~= nil
end

local function UpdateCameraPreview()
    local cf = GetCameraPlaceCF()

    if not PreviewModel then
        PreviewModel = Instance.new("Model")
        PreviewModel.Name = "ApexCamPreview"
        PreviewModel.Parent = Workspace

        local box = Instance.new("Part")
        box.Name = "PreviewHousing"
        box.Size = Vector3.new(1.9, 0.68, 0.68)
        box.Anchored = true; box.CanCollide = false; box.CastShadow = false
        box.Material = Enum.Material.Glass; box.Transparency = 0.5; box.Color = T.Violet
        box.Parent = PreviewModel

        local lens = Instance.new("Part")
        lens.Name = "PreviewLens"
        lens.Shape = Enum.PartType.Cylinder
        lens.Size = Vector3.new(0.2, 0.5, 0.5)
        lens.Anchored = true; lens.CanCollide = false; lens.CastShadow = false
        lens.Material = Enum.Material.Neon; lens.Transparency = 0.3; lens.Color = T.Cyan
        lens.Parent = PreviewModel
    end

    local parts = PreviewModel:GetChildren()
    local box, lens = parts[1], parts[2]
    if box then box.CFrame = cf * CFrame.new(CCTV_HOUSING_OFFSET) end
    if lens then
        lens.CFrame = cf * CFrame.new(CCTV_HOUSING_OFFSET + Vector3.new(0,0,1.15)) * CFrame.Angles(0, math.rad(90), 0)
    end
end

local function DestroyPreview()
    if PreviewModel then PreviewModel:Destroy(); PreviewModel=nil end
end

local function EnsureCameraViewport(camData)
    if camData.vp then return end
    local vp = Instance.new("ViewportFrame")
    vp.BackgroundTransparency = 1
    vp.Ambient = Color3.fromRGB(205,205,205)
    vp.LightDirection = Vector3.new(-1,-2,-1)

    local vpCam = Instance.new("Camera")
    vpCam.FieldOfView = 85
    vpCam.Parent = vp
    vp.CurrentCamera = vpCam

    camData.vp = vp
    camData.vpCam = vpCam
end

local function BuildCameraFixture(camData, placeCF)
    if camData.model then camData.model:Destroy() end
    local model = Instance.new("Model")
    model.Name = "ApexCam_"..camData.id
    model.Parent = Workspace

    local root, parts, led = BuildCCTVGeometry("fixture", model, placeCF)
    camData.model = model
    camData.mountCFrame = placeCF
    camData.root = root

    StartLedBlink(led, function() return model and model.Parent ~= nil end)

    EnsureCameraViewport(camData)
    print(("[ApexSurveillance] Camera #%d mounted at %s facing %s")
        :format(camData.id, tostring(placeCF.Position), tostring(placeCF.LookVector)))

    -- Fix: this used to be left to the next staggered refresh cycle,
    -- which only runs while the Tablet is open — a freshly mounted
    -- camera could sit completely empty (not even the backdrop) until
    -- its turn came up. Populate it right away instead.
    task.spawn(function()
        local pc, cc, tOK = BuildCameraScene(camData)
        print(("[ApexSurveillance] Camera #%d initial scene: %d parts, %d characters, terrain=%s")
            :format(camData.id, pc, cc, tostring(tOK)))
        if camData.tileLbl then
            camData.tileLbl.Text = ("CAM #%d  |  %dp %dc%s"):format(camData.id, pc, cc, tOK and " T" or "")
        end
    end)
end

local function ConfirmCameraPlacement()
    local cf = GetCameraPlaceCF()
    DestroyPreview()

    local camData = {
        id = NextCamId,
        model = nil,
        mountCFrame = nil,
        root = nil,
        vp = nil,
        vpCam = nil,
        tileLbl = nil,
    }
    NextCamId = NextCamId + 1
    table.insert(Cameras, camData)
    CountLabel.Text = #Cameras.." / "..MAX_CAMERAS.." cameras"

    BuildCameraFixture(camData, cf)

    PlaceYaw = 0
    PlacePitch = 0
end

    local camData = table.remove(Cameras)

    if camData.model then camData.model:Destroy() end
    if camData.vp then camData.vp:Destroy() end
    CountLabel.Text = #Cameras.." / "..MAX_CAMERAS.." cameras"

end

-- ════════════════════════════════════════════════════════════════
-- STAGGERED REFRESH LOOP — "recording." Only runs while the Tablet
-- is actually open, so idle cameras don't burn performance when
-- nobody's watching.
-- ════════════════════════════════════════════════════════════════
local CamRefreshCursor = 0

local function RefreshOneCamera()
    if #Cameras == 0 then return end
    CamRefreshCursor = (CamRefreshCursor % #Cameras) + 1
    local camData = Cameras[CamRefreshCursor]
    if not camData then return end
    EnsureCameraViewport(camData)
    local partCount, charCount, terrainOK = BuildCameraScene(camData)
    local txt = ("CAM #%d  |  %dp %dc%s"):format(camData.id, partCount, charCount, terrainOK and " T" or "")
    if camData.tileLbl then
        camData.tileLbl.Text = txt
    end
    if CurrentExpandedCam == camData and ExpandedStatusLbl then
        ExpandedStatusLbl.Text = txt
    end
end

-- Fix: a newly opened Tablet used to just wait for whichever camera
-- happened to be next in the staggered round-robin — with several
-- cameras placed, some feeds could sit stale for a while after
-- opening. Force every camera to refresh immediately on open instead.
local function RefreshAllCamerasNow()
    for _, camData in ipairs(Cameras) do
        EnsureCameraViewport(camData)
        task.spawn(function()
            local partCount, charCount, terrainOK = BuildCameraScene(camData)
            local txt = ("CAM #%d  |  %dp %dc%s"):format(camData.id, partCount, charCount, terrainOK and " T" or "")
            if camData.tileLbl then camData.tileLbl.Text = txt end
            if CurrentExpandedCam == camData and ExpandedStatusLbl then
                ExpandedStatusLbl.Text = txt
            end
        end)
    end
end

local camRefreshLoop = task.spawn(function()
    while true do
        local n = math.max(1, #Cameras)
        task.wait(CAM_REFRESH_INTERVAL / n)
        if TabletOpen then
            local ok, err = pcall(RefreshOneCamera)
            if not ok then warn("[ApexSurveillance] Camera refresh error: "..tostring(err)) end
        end
    end
end)
table.insert(ENV.ApexSurvLoops, {Disconnect=function() task.cancel(camRefreshLoop) end})

-- TOOL FACTORY — CAMERA PLACER (single reusable tool)

local function BindToolToBackpack(tool, attempt)
    attempt = attempt or 1
    local bp = GetBackpack()
    tool.Parent = bp
    task.wait(0.1)
    if tool.Parent == bp then
        StatusLabel.Text = "Inventory bound OK ("..#bp:GetChildren().." items)"
        StatusLabel.TextColor3 = T.Green
        return true
    end
    if attempt < 5 then
        task.wait(0.3)
        return BindToolToBackpack(tool, attempt+1)
    end
    StatusLabel.Text = "Backpack bind FAILED — tap SHOW INV"
    StatusLabel.TextColor3 = T.Red
    warn("[ApexSurveillance] Failed to parent tool to Backpack after 5 attempts: "..tool.Name)
    return false
end

local function CreateCameraPlacerTool()
    local tool = Instance.new("Tool")
    tool.Name = "Apex Camera Placer"
    tool.RequiresHandle = true
    tool.CanBeDropped = false
    tool.ToolTip = "Click to preview a mounting spot, click again to mount a camera there"
    table.insert(ENV.ApexSurvTools, tool)

    local _, _, led = BuildCCTVGeometry("tool", tool, nil)
    StartLedBlink(led, function() return tool and tool.Parent ~= nil end)

    -- Applied up front this time: Roblox's default grip treats local
    -- -Z as "away from player," but the lens stack here runs along
    -- local +Z. Flipping Forward AND Right together is a clean 180°
    -- turn (not a mirror), so the model holds facing the right way.
    tool.GripForward = Vector3.new(0, 0, 1)
    tool.GripRight   = Vector3.new(-1, 0, 0)
    tool.GripUp      = Vector3.new(0, 1, 0)
    tool.GripPos     = Vector3.new(0, -0.15, -0.3)

    local placing = false

    tool.Equipped:Connect(function()
        CrosshairRoot.Visible = true
    end)

    tool.Unequipped:Connect(function()
        CrosshairRoot.Visible = false
        if placing then
            PlaceYaw = 0; PlacePitch = 0
            PlaceHint.Visible = false
            AimPad.Visible = false
            DestroyPreview()
            placing = false
        end
    end)

    tool.Activated:Connect(function()
        if #Cameras >= MAX_CAMERAS then
            DoFlash(T.Red)
            warn("[ApexSurveillance] Camera limit reached ("..MAX_CAMERAS..").")
            return
        end

        if not placing then
            local result = RaycastFromCamera(RAY_MAX_DIST)
            if not result then
                DoFlash(T.Red)
                return
            end
            placing = true
            PlaceYaw = 0; PlacePitch = 0
            PlaceHint.Visible = true
            AimPad.Visible = true
        else
            ConfirmCameraPlacement()
            PlaceHint.Visible = false
            AimPad.Visible = false
            placing = false
        end
    end)

    BindToolToBackpack(tool)
    return tool
end

-- TOOL FACTORY — TABLET
-- ════════════════════════════════════════════════════════════════
local function CreateTabletTool()
    local tool = Instance.new("Tool")
    tool.Name = "Apex Tablet"
    tool.RequiresHandle = true
    tool.CanBeDropped = false
    tool.ToolTip = "Click to view all placed camera feeds"
    table.insert(ENV.ApexSurvTools, tool)

    BuildTabletHandle(tool)
    tool.GripForward = Vector3.new(1, 0, 0)
    tool.GripRight   = Vector3.new(0, 0, -1)
    tool.GripUp      = Vector3.new(0, 1, 0)
    tool.GripPos     = Vector3.new(0, -0.1, 0)

    tool.Equipped:Connect(function()
        TabletOpen = true
        TabletGui.Enabled = true
        CloseExpandedView()
        RefreshAllCamerasNow()
    end)

    tool.Unequipped:Connect(function()
        TabletOpen = false
        TabletGui.Enabled = false
    end)

    BindToolToBackpack(tool)
    return tool
end

-- ════════════════════════════════════════════════════════════════
-- PC KEYBOARD SUPPORT
-- ════════════════════════════════════════════════════════════════
UserInputService.InputBegan:Connect(function(inp, gpe)
    if gpe then return end

    if inp.KeyCode == Enum.KeyCode.Escape then
        if PlaceHint.Visible then
            PlaceYaw = 0; PlacePitch = 0
            PlaceHint.Visible = false
            AimPad.Visible = false
            DestroyPreview()
            local hum = GetHum()
            if hum then hum:UnequipTools() end
        end
        return
    end

    if PlaceHint.Visible then
        if inp.KeyCode == Enum.KeyCode.Left then AdjustYaw(-30); return
        elseif inp.KeyCode == Enum.KeyCode.Right then AdjustYaw(30); return
        elseif inp.KeyCode == Enum.KeyCode.Up then AdjustPitch(-15); return
        elseif inp.KeyCode == Enum.KeyCode.Down then AdjustPitch(15); return
        end
    end

    if inp.KeyCode == Enum.KeyCode.G then
        if #Cameras > 0 then DeleteLastCamera() end
        return
    end

    if inp.KeyCode == Enum.KeyCode.H then
        ForceBackpackVisible()
        StatusLabel.Text = "Forced inventory visible"
        StatusLabel.TextColor3 = T.Green
        return
    end
end)

-- ════════════════════════════════════════════════════════════════
-- CHARACTER RESPAWN HANDLING
-- ════════════════════════════════════════════════════════════════
LP.CharacterAdded:Connect(function()
    task.wait(0.5)
    ForceBackpackVisible()
    local bp = GetBackpack()
    for _, t in pairs(ENV.ApexSurvTools) do
        if t and t.Parent ~= bp and t.Parent ~= GetChar() then
            pcall(function() t.Parent = bp end)
        end
    end
    print("[ApexSurveillance] Character respawned — camera will be re-fetched live on next use.")
end)
